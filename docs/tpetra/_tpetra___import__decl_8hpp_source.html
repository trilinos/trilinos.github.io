501template <class LocalOrdinal, class GlobalOrdinal, class Node>
502Teuchos::RCP<const Import<LocalOrdinal, GlobalOrdinal, Node>>
503createImport(const Teuchos::RCP<const Map<LocalOrdinal, GlobalOrdinal, Node>>& src,
504 const Teuchos::RCP<const Map<LocalOrdinal, GlobalOrdinal, Node>>& tgt) {
505 if (src == tgt) {
506 return Teuchos::null;
507 }
508#ifdef HAVE_TPETRA_DEBUG
509 TEUCHOS_TEST_FOR_EXCEPTION(
510 src == Teuchos::null || tgt == Teuchos::null, std::runtime_error,
511 "Tpetra::createImport: Neither source nor target Map may be null.");
512#endif // HAVE_TPETRA_DEBUG
513 using import_type = Import<LocalOrdinal, GlobalOrdinal, Node>;
514 return Teuchos::rcp(new import_type(src, tgt));
515}

527template <class LocalOrdinal, class GlobalOrdinal, class Node>
528Teuchos::RCP<const Import<LocalOrdinal, GlobalOrdinal, Node>>
529createImport(const Teuchos::RCP<const Map<LocalOrdinal, GlobalOrdinal, Node>>& src,
530 const Teuchos::RCP<const Map<LocalOrdinal, GlobalOrdinal, Node>>& tgt,
531 const Teuchos::RCP<Teuchos::ParameterList>& plist) {
532 if (src == tgt) {
533 return Teuchos::null;
534 }
535#ifdef HAVE_TPETRA_DEBUG

537 src == Teuchos::null || tgt == Teuchos::null, std::runtime_error,
538 "Tpetra::createImport(): neither source nor target map may be null:"
539 << std::endl
540 << "source: " << src << std::endl
541 << "target: " << tgt
542 << std::endl);
543#endif // HAVE_TPETRA_DEBUG
544 typedef Import<LocalOrdinal, GlobalOrdinal, Node> import_type;
545 return Teuchos::rcp(new import_type(src, tgt, plist));
546}
