500template <class LocalOrdinal, class GlobalOrdinal, class Node>
501Teuchos::RCP<const Import<LocalOrdinal, GlobalOrdinal, Node>>
502createImport(const Teuchos::RCP<const Map<LocalOrdinal, GlobalOrdinal, Node>>& src,
503 const Teuchos::RCP<const Map<LocalOrdinal, GlobalOrdinal, Node>>& tgt) {
504 if (src == tgt) {
505 return Teuchos::null;
506 }
507#ifdef HAVE_TPETRA_DEBUG
508 TEUCHOS_TEST_FOR_EXCEPTION(
509 src == Teuchos::null || tgt == Teuchos::null, std::runtime_error,
510 "Tpetra::createImport: Neither source nor target Map may be null.");
511#endif // HAVE_TPETRA_DEBUG
512 using import_type = Import<LocalOrdinal, GlobalOrdinal, Node>;
513 return Teuchos::rcp(new import_type(src, tgt));
514}

526template <class LocalOrdinal, class GlobalOrdinal, class Node>
527Teuchos::RCP<const Import<LocalOrdinal, GlobalOrdinal, Node>>
528createImport(const Teuchos::RCP<const Map<LocalOrdinal, GlobalOrdinal, Node>>& src,
529 const Teuchos::RCP<const Map<LocalOrdinal, GlobalOrdinal, Node>>& tgt,
530 const Teuchos::RCP<Teuchos::ParameterList>& plist) {
531 if (src == tgt) {
532 return Teuchos::null;
533 }
534#ifdef HAVE_TPETRA_DEBUG
535 TEUCHOS_TEST_FOR_EXCEPTION(
536 src == Teuchos::null || tgt == Teuchos::null, std::runtime_error,
537 "Tpetra::createImport(): neither source nor target map may be null:"
538 << std::endl
539 << "source: " << src << std::endl
540 << "target: " << tgt
541 << std::endl);
542#endif // HAVE_TPETRA_DEBUG
543 typedef Import<LocalOrdinal, GlobalOrdinal, Node> import_type;
544 return Teuchos::rcp(new import_type(src, tgt, plist));
545}
