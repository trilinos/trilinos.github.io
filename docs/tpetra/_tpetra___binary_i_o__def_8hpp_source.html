47template <class LocalOrdinal, class GlobalOrdinal, class Node>
48Teuchos::RCP<const Tpetra::Map<LocalOrdinal, GlobalOrdinal, Node>>
49makeColumnMapFromGlobalColumns(const Kokkos::View<GlobalOrdinal*, Kokkos::HostSpace>& globalColumns,
50 const Teuchos::RCP<const Tpetra::Map<LocalOrdinal, GlobalOrdinal, Node>>& domainMap) {
51 using map_type = Tpetra::Map<LocalOrdinal, GlobalOrdinal, Node>;
52 using memory_space = typename Node::memory_space;
53 const size_t localNnz = globalColumns.extent(0);
54
55 int hasLocalEntries = localNnz == 0 ? 0 : 1;
56 int allHaveLocalEntries = 0;
57 Teuchos::reduceAll(*domainMap->getComm(), Teuchos::REDUCE_MIN, 1, &hasLocalEntries, &allHaveLocalEntries);
58 if (allHaveLocalEntries != 0) {
59 Kokkos::View<GlobalOrdinal*, memory_space> globalColumnsDefaultMemory("Tpetra::BinaryIO::globalColumnsForColMap", localNnz);
60 Kokkos::deep_copy(globalColumnsDefaultMemory, globalColumns);
61
62 Teuchos::RCP<const map_type> colMap;
63 std::ostringstream errStrm;
64 const int err = Tpetra::Details::makeColMap<LocalOrdinal, GlobalOrdinal, Node>(colMap,
65 domainMap,
66 globalColumnsDefaultMemory,
67 &errStrm);
68 TEUCHOS_TEST_FOR_EXCEPTION(err != 0 || colMap.is_null(),
69 std::runtime_error,
70 "Tpetra::BinaryIO: Failed to construct column map from sparse matrix file. " << errStrm.str());
71 return colMap;
72 }
73
74 using host_execution_space = Kokkos::DefaultHostExecutionSpace;
75 Kokkos::View<GlobalOrdinal*, Kokkos::HostSpace> colGids("Tpetra::BinaryIO::colGids", localNnz);
76 Kokkos::parallel_for(
77 "Tpetra::BinaryIO::copyColumnGidsForMap",
78 Kokkos::RangePolicy<host_execution_space>(0, localNnz),
79 KOKKOS_LAMBDA(const size_t i) {
80 colGids(i) = globalColumns(i);
81 });
82 if (localNnz > 1) {
83 Kokkos::sort(colGids);
84 }
85
86 size_t uniqueCount = 0;
87 Kokkos::parallel_scan(
88 "Tpetra::BinaryIO::uniqueColumnGidsForMap",
89 Kokkos::RangePolicy<host_execution_space>(0, localNnz),
90 KOKKOS_LAMBDA(const size_t i, size_t& count, const bool finalPass) {
91 const bool isUnique = i == 0 || colGids(i) != colGids(i - 1);
92 if (isUnique) {
93 if (finalPass) {
94 colGids(count) = colGids(i);
95 }
96 ++count;
97 }
98 },
99 uniqueCount);
100
101 const LocalOrdinal localCount = binaryIOCheckedLocalOrdinalCount<LocalOrdinal>(uniqueCount, "column-map local entry count");
102 using device_type = typename map_type::device_type;
103 Kokkos::View<GlobalOrdinal*, device_type> colGidsDevice("Tpetra::BinaryIO::colGidsDevice", static_cast<size_t>(localCount));
104 Kokkos::deep_copy(colGidsDevice, Kokkos::subview(colGids, Kokkos::make_pair(size_t(0), static_cast<size_t>(localCount))));
105 return Teuchos::rcp(new map_type(Teuchos::OrdinalTraits<Tpetra::global_size_t>::invalid(),
106 colGidsDevice,
107 domainMap->getIndexBase(),
108 domainMap->getComm()));
109}

111template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
112Teuchos::RCP<Tpetra::CrsMatrix<Scalar, LocalOrdinal, GlobalOrdinal, Node>>
113buildSparseMatrixFromLocalCrsViews(const Teuchos::RCP<const Tpetra::Map<LocalOrdinal, GlobalOrdinal, Node>>& rowMap,
114 const Teuchos::RCP<const Tpetra::Map<LocalOrdinal, GlobalOrdinal, Node>>& inputColMap,
115 const Teuchos::RCP<const Tpetra::Map<LocalOrdinal, GlobalOrdinal, Node>>& domainMap,
116 const Kokkos::View<unsigned long long*, Kokkos::HostSpace>& rowPtrHost,
117 const Kokkos::View<GlobalOrdinal*, Kokkos::HostSpace>& globalColumnsHost,
118 const Kokkos::View<Scalar*, Kokkos::HostSpace>& valuesHost) {
119 using matrix_type = Tpetra::CrsMatrix<Scalar, LocalOrdinal, GlobalOrdinal, Node>;
120 using local_graph_type = typename matrix_type::local_graph_device_type;
121 using rowptr_type = typename local_graph_type::row_map_type::non_const_type;
122 using colidx_type = typename local_graph_type::entries_type::non_const_type;
123 using values_type = typename matrix_type::local_matrix_device_type::values_type::non_const_type;
124 using impl_scalar_type = typename matrix_type::impl_scalar_type;
125 using device_type = typename matrix_type::device_type;
126 using execution_space = typename device_type::execution_space;
127 using host_execution_space = Kokkos::DefaultHostExecutionSpace;
128
129 const size_t localNumRows = rowMap->getLocalNumElements();
130 const size_t localNnz = globalColumnsHost.extent(0);
131
132 Teuchos::RCP<const Tpetra::Map<LocalOrdinal, GlobalOrdinal, Node>> colMap = inputColMap;
133 if (colMap.is_null()) {
134 colMap = makeColumnMapFromGlobalColumns<LocalOrdinal, GlobalOrdinal, Node>(globalColumnsHost, domainMap);
135 }
136
137 rowptr_type rowPtrDevice("Tpetra::BinaryIO::rowPtr", localNumRows + 1);
138 auto rowPtrDeviceHost = Kokkos::create_mirror_view(rowPtrDevice);
139 using rowptr_value_type = typename rowptr_type::non_const_value_type;
140 unsigned long long rowPtrOverflowCount = 0;
141 Kokkos::parallel_reduce(
142 "Tpetra::BinaryIO::checkRowPtrRange",
143 Kokkos::RangePolicy<host_execution_space>(0, localNumRows + 1),
144 KOKKOS_LAMBDA(const size_t i, unsigned long long& overflowCount) {
145 if (rowPtrHost(i) > static_cast<unsigned long long>(std::numeric_limits<rowptr_value_type>::max())) {
146 ++overflowCount;
147 }
148 },
149 rowPtrOverflowCount);
150 TEUCHOS_TEST_FOR_EXCEPTION(rowPtrOverflowCount != 0,
151 std::overflow_error,
152 "Tpetra::BinaryIO: local sparse row offset does not fit in the local row pointer type.");
153 Kokkos::parallel_for(
154 "Tpetra::BinaryIO::copyRowPtrToDeviceHost",
155 Kokkos::RangePolicy<host_execution_space>(0, localNumRows + 1),
156 KOKKOS_LAMBDA(const size_t i) {
157 rowPtrDeviceHost(i) = static_cast<rowptr_value_type>(rowPtrHost(i));
158 });
159 Kokkos::deep_copy(rowPtrDevice, rowPtrDeviceHost);
160
161 Kokkos::View<GlobalOrdinal*, device_type> globalColumnsDevice("Tpetra::BinaryIO::globalColumns", localNnz);
162 Kokkos::deep_copy(globalColumnsDevice, globalColumnsHost);
163
164 colidx_type localColumnsDevice("Tpetra::BinaryIO::localColumns", localNnz);
165 const auto localColMap = colMap->getLocalMap();
166 const LocalOrdinal invalidLocalOrdinal = Tpetra::Details::OrdinalTraits<LocalOrdinal>::invalid();
167 unsigned long long invalidLocalColumnIdCount = 0;
168 Kokkos::parallel_reduce(
169 "Tpetra::BinaryIO::reindexColumns",
170 Kokkos::RangePolicy<execution_space>(0, localNnz),
171 KOKKOS_LAMBDA(const size_t i, unsigned long long& invalidCount) {
172 const LocalOrdinal lclCol = localColMap.getLocalElement(globalColumnsDevice(i));
173 localColumnsDevice(i) = lclCol;
174 if (lclCol == invalidLocalOrdinal) {
175 ++invalidCount;
176 }
177 },
178 invalidLocalColumnIdCount);
179 TEUCHOS_TEST_FOR_EXCEPTION(invalidLocalColumnIdCount != 0,
180 std::runtime_error,
181 "Tpetra::BinaryIO: Column map is missing " << invalidLocalColumnIdCount
182 << " column GID(s) stored in the sparse matrix file.");
183
184 values_type valuesDevice("Tpetra::BinaryIO::values", localNnz);
185 auto valuesDeviceHost = Kokkos::create_mirror_view(valuesDevice);
186 Kokkos::parallel_for(
187 "Tpetra::BinaryIO::copyValuesToDeviceHost",
188 Kokkos::RangePolicy<host_execution_space>(0, localNnz),
189 KOKKOS_LAMBDA(const size_t i) {
190 valuesDeviceHost(i) = static_cast<impl_scalar_type>(valuesHost(i));
191 });
192 Kokkos::deep_copy(valuesDevice, valuesDeviceHost);
193
194 Tpetra::Import_Util::sortCrsEntries(rowPtrDevice, localColumnsDevice, valuesDevice);
195 return Teuchos::rcp(new matrix_type(rowMap, colMap, rowPtrDevice, localColumnsDevice, valuesDevice));
196}

200template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
201typename BinaryIO<Scalar, LocalOrdinal, GlobalOrdinal, Node>::FileHeader
202BinaryIO<Scalar, LocalOrdinal, GlobalOrdinal, Node>::makeBaseHeader(const unsigned long long objectKind) {
203 FileHeader header;
204 std::memset(&header, 0, sizeof(FileHeader));
205 const char magic[8] = {'T', 'P', 'B', 'I', 'O', '0', '0', '1'};
206 std::memcpy(header.magic, magic, sizeof(magic));
207 header.version = fileVersion;
208 header.byteOrderMarker = byteOrderMarker;
209 header.objectKind = objectKind;
210 header.scalarSize = sizeof(Scalar);
211 header.scalarFlags = typeFlags<Scalar>();
212 header.localOrdinalSize = sizeof(LocalOrdinal);
213 header.localOrdinalFlags = typeFlags<LocalOrdinal>();
214 header.globalOrdinalSize = sizeof(GlobalOrdinal);
215 header.globalOrdinalFlags = typeFlags<GlobalOrdinal>();
216 return header;
217}

219template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
220void BinaryIO<Scalar, LocalOrdinal, GlobalOrdinal, Node>::validateHeader(const FileHeader& header,
221 const unsigned long long expectedObjectKind) {
222 const char expectedMagic[8] = {'T', 'P', 'B', 'I', 'O', '0', '0', '1'};
223 TEUCHOS_TEST_FOR_EXCEPTION(std::memcmp(header.magic, expectedMagic, sizeof(expectedMagic)) != 0,
224 std::runtime_error,
225 "Tpetra::BinaryIO: File does not have a recognized Tpetra binary I/O header. "
226 << "Legacy Xpetra binary matrix files must be converted with "
227 << "Xpetra::Experimental::ConvertLegacyBinaryToBinary before reading with Tpetra::Experimental::BinaryIO.");
228 TEUCHOS_TEST_FOR_EXCEPTION(header.version != fileVersion,
229 std::runtime_error,
230 "Tpetra::BinaryIO: Unsupported file version " << header.version << ".");
231 // Files are stored in the writer's native byte order; reading a file written
232 // with the opposite endianness is detected but not converted.
233 TEUCHOS_TEST_FOR_EXCEPTION(header.byteOrderMarker == Details::binaryIOByteSwap64(byteOrderMarker),
234 std::runtime_error,
235 "Tpetra::BinaryIO: File was written on a machine with the opposite byte order; "
236 "byte-order conversion is not supported.");
237 TEUCHOS_TEST_FOR_EXCEPTION(header.byteOrderMarker != byteOrderMarker,
238 std::runtime_error,
239 "Tpetra::BinaryIO: Unsupported byte order marker.");
240 TEUCHOS_TEST_FOR_EXCEPTION(header.objectKind != expectedObjectKind,
241 std::runtime_error,
242 "Tpetra::BinaryIO: File object kind " << header.objectKind
243 << " does not match expected kind " << expectedObjectKind << ".");
244 if (expectedObjectKind != MAP_OBJECT) {
245 TEUCHOS_TEST_FOR_EXCEPTION(header.scalarSize != sizeof(Scalar) || header.scalarFlags != typeFlags<Scalar>(),
246 std::runtime_error,
247 "Tpetra::BinaryIO: File scalar type does not match this BinaryIO instantiation.");
248 }
249 TEUCHOS_TEST_FOR_EXCEPTION(header.localOrdinalSize != sizeof(LocalOrdinal) || header.localOrdinalFlags != typeFlags<LocalOrdinal>(),
250 std::runtime_error,
251 "Tpetra::BinaryIO: File local ordinal type does not match this BinaryIO instantiation.");
252 TEUCHOS_TEST_FOR_EXCEPTION(header.globalOrdinalSize != sizeof(GlobalOrdinal) || header.globalOrdinalFlags != typeFlags<GlobalOrdinal>(),
253 std::runtime_error,
254 "Tpetra::BinaryIO: File global ordinal type does not match this BinaryIO instantiation.");
255}

257template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
258template <class T>
259unsigned long long BinaryIO<Scalar, LocalOrdinal, GlobalOrdinal, Node>::typeFlags() {
260 unsigned long long flags = 0;
261 if (Teuchos::ScalarTraits<T>::isOrdinal) {
262 flags |= TYPE_IS_ORDINAL;
263 }
264 if (Teuchos::ScalarTraits<T>::isComplex) {
265 flags |= TYPE_IS_COMPLEX;
266 }
267 if (std::is_floating_point<T>::value || Teuchos::ScalarTraits<T>::isComplex) {
268 flags |= TYPE_IS_FLOATING;
269 }
270 if (std::is_signed<T>::value || Teuchos::ScalarTraits<T>::isComplex || std::is_floating_point<T>::value) {
271 flags |= TYPE_IS_SIGNED;
272 }
273 return flags;
274}

276template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
277unsigned long long BinaryIO<Scalar, LocalOrdinal, GlobalOrdinal, Node>::mapFlags(const map_type& map) {
278 unsigned long long flags = 0;
279 if (map.isDistributed()) {
280 flags |= 1ull << 0;
281 }
282 if (map.isContiguous()) {
283 flags |= 1ull << 1;
284 }
285 if (map.isOneToOne()) {
286 flags |= 1ull << 2;
287 }
288 return flags;
289}

303template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
304void BinaryIO<Scalar, LocalOrdinal, GlobalOrdinal, Node>::writeHeaderToNewFile(const std::string& filename,
305 const FileHeader& header,
306 const trcp_tcomm_t& comm) {
307#ifdef HAVE_TPETRACORE_MPI
308 if (Tpetra::Details::teuchosCommIsAnMpiComm(*comm)) {
309 // Rank 0 alone creates, truncates, and writes the header. A collective
310 // MPI_File_set_size followed by rank 0's write conflicts under MPI's default
311 // consistency semantics: without sync-barrier-sync, another process's
312 // truncate may land after the header write and erase it. Closing on rank 0
313 // before the barrier makes the header visible to later collective opens.
314 int fileErr = MPI_SUCCESS;
315 if (comm->getRank() == 0) {
316 MPI_File file;
317 fileErr = MPI_File_open(MPI_COMM_SELF, const_cast<char*>(filename.c_str()),
318 MPI_MODE_CREATE | MPI_MODE_WRONLY,
319 MPI_INFO_NULL, &file);
320 if (fileErr == MPI_SUCCESS) {
321 fileErr = MPI_File_set_size(file, static_cast<MPI_Offset>(0));
322 if (fileErr == MPI_SUCCESS) {
323 MPI_Status status;
324 fileErr = MPI_File_write_at(file, 0, const_cast<FileHeader*>(&header),
325 sizeof(FileHeader), MPI_BYTE, &status);
326 }
327 const int closeErr = MPI_File_close(&file);
328 if (fileErr == MPI_SUCCESS) {
329 fileErr = closeErr;
330 }
331 }
332 }
333 Teuchos::broadcast(*comm, 0, 1, &fileErr);
334 TEUCHOS_TEST_FOR_EXCEPTION(fileErr != MPI_SUCCESS, std::runtime_error,
335 "Tpetra::BinaryIO: Failed to create file '" << filename << "' and write its header.");
336 comm->barrier();
337 return;
338 }
339#endif
340 bool success = true;
341 if (comm->getRank() == 0) {
342 std::ofstream out(filename.c_str(), std::ios::binary | std::ios::trunc);
343 success = static_cast<bool>(out);
344 if (success) {
345 out.write(reinterpret_cast<const char*>(&header), sizeof(FileHeader));
346 success = static_cast<bool>(out);
347 }
348 }
349 checkFileOpen(success, filename, comm, "writing");
350 comm->barrier();
351}

353template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
354typename BinaryIO<Scalar, LocalOrdinal, GlobalOrdinal, Node>::FileHeader
355BinaryIO<Scalar, LocalOrdinal, GlobalOrdinal, Node>::readHeaderFromFile(const std::string& filename,
356 const trcp_tcomm_t& comm) {
357 FileHeader header;
358 std::memset(&header, 0, sizeof(FileHeader));
359#ifdef HAVE_TPETRACORE_MPI
360 if (Tpetra::Details::teuchosCommIsAnMpiComm(*comm)) {
361 MPI_Comm rawComm = Tpetra::Details::extractMpiCommFromTeuchos(*comm);
362 MPI_File file;
363 const int err = MPI_File_open(rawComm, const_cast<char*>(filename.c_str()),
364 MPI_MODE_RDONLY, MPI_INFO_NULL, &file);
365 TEUCHOS_TEST_FOR_EXCEPTION(err != MPI_SUCCESS, std::runtime_error,
366 "Tpetra::BinaryIO: MPI_File_open failed for '" << filename << "'.");
367 MPI_Status status;
368 const int readErr = MPI_File_read_at_all(file, 0, &header, sizeof(FileHeader), MPI_BYTE, &status);
369 MPI_File_close(&file);
370 TEUCHOS_TEST_FOR_EXCEPTION(readErr != MPI_SUCCESS, std::runtime_error,
371 "Tpetra::BinaryIO: MPI_File_read_at_all failed while reading header.");
372 return header;
373 }
374#endif
375 bool success = true;
376 if (comm->getRank() == 0) {
377 std::ifstream in(filename.c_str(), std::ios::binary);
378 success = static_cast<bool>(in);
379 if (success) {
380 in.read(reinterpret_cast<char*>(&header), sizeof(FileHeader));
381 success = static_cast<bool>(in);
382 }
383 }
384 checkFileOpen(success, filename, comm, "reading");
385 Teuchos::broadcast(*comm, 0, static_cast<int>(sizeof(FileHeader)), reinterpret_cast<char*>(&header));
386 return header;
387}

417template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
418void BinaryIO<Scalar, LocalOrdinal, GlobalOrdinal, Node>::writeMapSectionHeader(const std::string& filename,
419 const unsigned long long mapSectionOffset,
420 const MapSectionHeader& header,
421 const trcp_tcomm_t& comm) {
422#ifdef HAVE_TPETRACORE_MPI
423 if (Tpetra::Details::teuchosCommIsAnMpiComm(*comm)) {
424 MPI_Comm rawComm = Tpetra::Details::extractMpiCommFromTeuchos(*comm);
425 MPI_File file;
426 const int err = MPI_File_open(rawComm, const_cast<char*>(filename.c_str()), MPI_MODE_WRONLY, MPI_INFO_NULL, &file);
427 TEUCHOS_TEST_FOR_EXCEPTION(err != MPI_SUCCESS, std::runtime_error,
428 "Tpetra::BinaryIO: MPI_File_open failed while writing map section header.");
429 if (comm->getRank() == 0) {
430 MPI_Status status;
431 const int writeErr = MPI_File_write_at(file, static_cast<MPI_Offset>(mapSectionOffset),
432 const_cast<MapSectionHeader*>(&header), sizeof(MapSectionHeader), MPI_BYTE, &status);
433 TEUCHOS_TEST_FOR_EXCEPTION(writeErr != MPI_SUCCESS, std::runtime_error,
434 "Tpetra::BinaryIO: MPI_File_write_at failed while writing map section header.");
435 }
436 MPI_File_close(&file);
437 comm->barrier();
438 return;
439 }
440#endif
441 if (comm->getRank() == 0) {
442 std::fstream out(filename.c_str(), std::ios::binary | std::ios::in | std::ios::out);
443 TEUCHOS_TEST_FOR_EXCEPTION(!out.good(), std::runtime_error,
444 "Tpetra::BinaryIO: Failed to reopen file '" << filename << "' while writing map section header.");
445 out.seekp(static_cast<std::streamoff>(mapSectionOffset), std::ios::beg);
446 out.write(reinterpret_cast<const char*>(&header), sizeof(MapSectionHeader));
447 TEUCHOS_TEST_FOR_EXCEPTION(!out.good(), std::runtime_error,
448 "Tpetra::BinaryIO: Failed to write map section header to file '" << filename << "'.");
449 }
450 comm->barrier();
451}

489template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
490unsigned long long BinaryIO<Scalar, LocalOrdinal, GlobalOrdinal, Node>::exclusiveScanUnsignedLongLong(const unsigned long long localValue,
491 const trcp_tcomm_t& comm) {
492#ifdef HAVE_TPETRACORE_MPI
493 if (Tpetra::Details::teuchosCommIsAnMpiComm(*comm)) {
494 MPI_Comm rawComm = Tpetra::Details::extractMpiCommFromTeuchos(*comm);
495 unsigned long long result = 0;
496 const int err = MPI_Exscan(const_cast<unsigned long long*>(&localValue), &result, 1,
497 MPI_UNSIGNED_LONG_LONG, MPI_SUM, rawComm);
498 TEUCHOS_TEST_FOR_EXCEPTION(err != MPI_SUCCESS, std::runtime_error,
499 "Tpetra::BinaryIO: MPI_Exscan failed.");
500 if (comm->getRank() == 0) {
501 result = 0;
502 }
503 return result;
504 }
505#endif
506 unsigned long long inclusive = 0;
507 Teuchos::scan<int, unsigned long long>(*comm, Teuchos::REDUCE_SUM, 1,
508 const_cast<unsigned long long*>(&localValue), &inclusive);
509 return inclusive - localValue;
510}

560 comm);

574 globalOffset,
575 comm);

579 Kokkos::deep_copy(gidsDevice, gids);

604template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
605void BinaryIO<Scalar, LocalOrdinal, GlobalOrdinal, Node>::writeDenseFile(const std::string& filename,
606 const multivector_type& X) {
607 const auto map = X.getMap();
608 FileHeader header = makeBaseHeader(MULTIVECTOR_OBJECT);
609 header.numGlobalRows = static_cast<unsigned long long>(X.getGlobalLength());
610 header.numGlobalCols = static_cast<unsigned long long>(X.getNumVectors());
611 header.numVectors = static_cast<unsigned long long>(X.getNumVectors());
612 header.rowMapOffset = static_cast<unsigned long long>(sizeof(FileHeader));
613 header.valuesOffset = header.rowMapOffset + mapSectionSize(*map) + header.numGlobalRows * sizeof(GlobalOrdinal);
614 writeHeaderToNewFile(filename, header, map->getComm());
615 writeMapSection(filename, header.rowMapOffset, *map, map->getComm());
616
617 const unsigned long long localCount = static_cast<unsigned long long>(map->getLocalNumElements());
618 const unsigned long long globalOffset = exclusiveScanUnsignedLongLong(localCount, map->getComm());
619 const unsigned long long globalLength = static_cast<unsigned long long>(X.getGlobalLength());
620
621 using host_execution_space = Kokkos::DefaultHostExecutionSpace;
622 auto localXDevice = X.getLocalViewDevice(Tpetra::Access::ReadOnly);
623 auto localXHost = Kokkos::create_mirror_view_and_copy(Kokkos::HostSpace(), localXDevice);
624 Kokkos::View<Scalar*, Kokkos::HostSpace> localValues("Tpetra::BinaryIO::denseLocalValues", Details::binaryIOCheckedSize(localCount, "dense local row count"));
625 for (size_t j = 0; j < X.getNumVectors(); ++j) {
626 const unsigned long long columnOffset = static_cast<unsigned long long>(j) * globalLength + globalOffset;
627 Kokkos::parallel_for(
628 "Tpetra::BinaryIO::copyDenseColumnToHost",
629 Kokkos::RangePolicy<host_execution_space>(0, localValues.extent(0)),
630 KOKKOS_LAMBDA(const size_t i) {
631 localValues(i) = static_cast<Scalar>(localXHost(i, j));
632 });
633 Details::binaryIOWriteArrayCollective(filename, header.valuesOffset, localValues.data(), localCount, columnOffset, map->getComm());
634 }

647template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
648Teuchos::RCP<typename BinaryIO<Scalar, LocalOrdinal, GlobalOrdinal, Node>::multivector_type>
649BinaryIO<Scalar, LocalOrdinal, GlobalOrdinal, Node>::readDenseFile(const std::string& filename,
650 const Teuchos::RCP<const map_type>& map) {
651 const auto comm = map->getComm();
652 const FileHeader header = readHeaderFromFile(filename, comm);
653 validateHeader(header, MULTIVECTOR_OBJECT);
654
655 TEUCHOS_TEST_FOR_EXCEPTION(static_cast<unsigned long long>(map->getGlobalNumElements()) != header.numGlobalRows,
656 std::invalid_argument,
657 "Tpetra::BinaryIO: Requested dense read map has "
658 << map->getGlobalNumElements() << " global elements, but file has "
659 << header.numGlobalRows << " rows.");
660
661 const unsigned long long globalLength = header.numGlobalRows;
662 auto fileMap = readMapSection(filename, header.rowMapOffset, comm);
663 TEUCHOS_TEST_FOR_EXCEPTION(static_cast<unsigned long long>(fileMap->getGlobalNumElements()) != globalLength,
664 std::runtime_error,
665 "Tpetra::BinaryIO: Dense file row-map length does not match header row count.");
666
667 auto fileX = Teuchos::rcp(new multivector_type(fileMap, static_cast<size_t>(header.numVectors)));
668 const unsigned long long fileLocalCount = static_cast<unsigned long long>(fileMap->getLocalNumElements());
669 const unsigned long long fileGlobalOffset = exclusiveScanUnsignedLongLong(fileLocalCount, comm);
670 using impl_scalar_type = typename multivector_type::impl_scalar_type;
671 using host_execution_space = Kokkos::DefaultHostExecutionSpace;
672 auto localFileXDevice = fileX->getLocalViewDevice(Tpetra::Access::OverwriteAll);
673 auto localFileXHost = Kokkos::create_mirror_view(localFileXDevice);
674 Kokkos::View<Scalar*, Kokkos::HostSpace> localValues("Tpetra::BinaryIO::denseLocalValues", Details::binaryIOCheckedSize(fileLocalCount, "dense local row count"));
675
676 for (size_t j = 0; j < fileX->getNumVectors(); ++j) {
677 const unsigned long long columnOffset = static_cast<unsigned long long>(j) * globalLength + fileGlobalOffset;
678 Details::binaryIOReadArrayCollective(filename, header.valuesOffset, localValues.data(), fileLocalCount, columnOffset, comm);
679 Kokkos::parallel_for(
680 "Tpetra::BinaryIO::copyDenseColumnToDeviceHost",
681 Kokkos::RangePolicy<host_execution_space>(0, localValues.extent(0)),
682 KOKKOS_LAMBDA(const size_t i) {
683 localFileXHost(i, j) = static_cast<impl_scalar_type>(localValues(i));
684 });
685 }
686 Kokkos::deep_copy(localFileXDevice, localFileXHost);
687
688 if (map->isSameAs(*fileMap)) {
689 return fileX;
690 }
691
692 using import_type = Tpetra::Import<LocalOrdinal, GlobalOrdinal, Node>;
693 auto X = Teuchos::rcp(new multivector_type(map, static_cast<size_t>(header.numVectors)));
694 import_type importer(fileMap, map);
695 X->doImport(*fileX, importer, Tpetra::INSERT);
696 return X;
697}

725template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
726void BinaryIO<Scalar, LocalOrdinal, GlobalOrdinal, Node>::writeSparseFile(const std::string& filename,
727 const sparse_matrix_type& A) {
728 TEUCHOS_TEST_FOR_EXCEPTION(!A.isFillComplete(), std::invalid_argument,
729 "Tpetra::BinaryIO::writeSparseFile requires a fill-complete matrix.");
730
731 const auto rowMap = A.getRowMap();
732 const auto domainMap = A.getDomainMap();
733 const auto rangeMap = A.getRangeMap();
734 const auto colMap = A.getColMap();
735 TEUCHOS_TEST_FOR_EXCEPTION(colMap.is_null(), std::invalid_argument,
736 "Tpetra::BinaryIO::writeSparseFile requires a nonnull column map.");
737
738 const unsigned long long globalNumRows = static_cast<unsigned long long>(A.getGlobalNumRows());
739 const unsigned long long globalNumEntries = static_cast<unsigned long long>(A.getGlobalNumEntries());
740 const unsigned long long rowMapBytes = mapSectionSize(*rowMap) + globalNumRows * sizeof(GlobalOrdinal);
741 const unsigned long long domainMapBytes = mapSectionSize(*domainMap) + static_cast<unsigned long long>(domainMap->getGlobalNumElements()) * sizeof(GlobalOrdinal);
742 const unsigned long long rangeMapBytes = mapSectionSize(*rangeMap) + static_cast<unsigned long long>(rangeMap->getGlobalNumElements()) * sizeof(GlobalOrdinal);
743
744 FileHeader header = makeBaseHeader(MATRIX_OBJECT);
745 header.numGlobalRows = globalNumRows;
746 header.numGlobalCols = static_cast<unsigned long long>(A.getGlobalNumCols());
747 header.numGlobalEntries = globalNumEntries;
748 header.rowMapOffset = static_cast<unsigned long long>(sizeof(FileHeader));
749 header.domainMapOffset = header.rowMapOffset + rowMapBytes;
750 header.rangeMapOffset = header.domainMapOffset + domainMapBytes;
751 header.rowPtrOffset = header.rangeMapOffset + rangeMapBytes;
752 header.columnIndicesOffset = header.rowPtrOffset + (globalNumRows + 1ull) * sizeof(unsigned long long);
753 header.valuesOffset = header.columnIndicesOffset + globalNumEntries * sizeof(GlobalOrdinal);
754
755 writeHeaderToNewFile(filename, header, rowMap->getComm());
756 writeMapSection(filename, header.rowMapOffset, *rowMap, rowMap->getComm());
757 writeMapSection(filename, header.domainMapOffset, *domainMap, domainMap->getComm());
758 writeMapSection(filename, header.rangeMapOffset, *rangeMap, rangeMap->getComm());
759
760 using device_type = typename sparse_matrix_type::device_type;
761 using execution_space = typename device_type::execution_space;
762 using host_execution_space = Kokkos::DefaultHostExecutionSpace;
763
764 const size_t localNumRows = rowMap->getLocalNumElements();
765 const auto localMatrix = A.getLocalMatrixDevice();
766 const auto rowPtrDevice = localMatrix.graph.row_map;
767 auto rowPtrDeviceHost = Kokkos::create_mirror_view_and_copy(Kokkos::HostSpace(), rowPtrDevice);
768
769 const unsigned long long localNnz = static_cast<unsigned long long>(rowPtrDeviceHost(localNumRows));
770 const size_t localNnzSize = Details::binaryIOCheckedSize(localNnz, "local sparse entry count");
771 Kokkos::View<unsigned long long*, Kokkos::HostSpace> localRowPtr("Tpetra::BinaryIO::localRowPtr", localNumRows + 1);
772 Kokkos::parallel_for(
773 "Tpetra::BinaryIO::copyLocalRowPtr",
774 Kokkos::RangePolicy<host_execution_space>(0, localNumRows + 1),
775 KOKKOS_LAMBDA(const size_t i) {
776 localRowPtr(i) = static_cast<unsigned long long>(rowPtrDeviceHost(i));
777 });
778
779 Kokkos::View<GlobalOrdinal*, device_type> globalColumnsDevice("Tpetra::BinaryIO::globalColumns", localNnzSize);
780 const auto localColumnsDevice = localMatrix.graph.entries;
781 const auto localColMap = colMap->getLocalMap();
782 Kokkos::parallel_for(
783 "Tpetra::BinaryIO::convertColumnsToGlobal",
784 Kokkos::RangePolicy<execution_space>(0, localNnzSize),
785 KOKKOS_LAMBDA(const size_t i) {
786 globalColumnsDevice(i) = localColMap.getGlobalElement(localColumnsDevice(i));
787 });
788 auto localColumns = Kokkos::create_mirror_view_and_copy(Kokkos::HostSpace(), globalColumnsDevice);
789
790 auto localValuesImpl = Kokkos::create_mirror_view_and_copy(Kokkos::HostSpace(), localMatrix.values);
791 Kokkos::View<Scalar*, Kokkos::HostSpace> localValues("Tpetra::BinaryIO::localValues", localNnzSize);
792 Kokkos::parallel_for(
793 "Tpetra::BinaryIO::copyLocalValues",
794 Kokkos::RangePolicy<host_execution_space>(0, localNnzSize),
795 KOKKOS_LAMBDA(const size_t i) {
796 localValues(i) = static_cast<Scalar>(localValuesImpl(i));
797 });
798
799 const unsigned long long globalRowOffset = exclusiveScanUnsignedLongLong(static_cast<unsigned long long>(localNumRows), rowMap->getComm());
800 const unsigned long long globalNnzOffset = exclusiveScanUnsignedLongLong(localNnz, rowMap->getComm());
801 Kokkos::parallel_for(
802 "Tpetra::BinaryIO::offsetLocalRowPtr",
803 Kokkos::RangePolicy<host_execution_space>(0, localNumRows + 1),
804 KOKKOS_LAMBDA(const size_t i) {
805 localRowPtr(i) += globalNnzOffset;
806 });
807
808 Details::binaryIOWriteArrayCollective(filename, header.rowPtrOffset, localRowPtr.data(),
809 static_cast<unsigned long long>(localRowPtr.extent(0)), globalRowOffset, rowMap->getComm());
810 Details::binaryIOWriteArrayCollective(filename, header.columnIndicesOffset, localColumns.data(), localNnz, globalNnzOffset, rowMap->getComm());
811 Details::binaryIOWriteArrayCollective(filename, header.valuesOffset, localValues.data(), localNnz, globalNnzOffset, rowMap->getComm());
812}

826template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
827Teuchos::RCP<typename BinaryIO<Scalar, LocalOrdinal, GlobalOrdinal, Node>::sparse_matrix_type>
828BinaryIO<Scalar, LocalOrdinal, GlobalOrdinal, Node>::readSparseFile(const std::string& filename,
829 const Teuchos::RCP<const map_type>& rowMap,
830 const Teuchos::RCP<const map_type>& colMap,
831 const Teuchos::RCP<const map_type>& domainMap,
832 const Teuchos::RCP<const map_type>& rangeMap,
833 const bool callFillComplete) {
834 const auto comm = rowMap->getComm();
835 const FileHeader header = readHeaderFromFile(filename, comm);
836 validateHeader(header, MATRIX_OBJECT);
837
838 TEUCHOS_TEST_FOR_EXCEPTION(static_cast<unsigned long long>(rowMap->getGlobalNumElements()) != header.numGlobalRows,
839 std::invalid_argument,
840 "Tpetra::BinaryIO: Requested sparse read row map has "
841 << rowMap->getGlobalNumElements() << " global elements, but file has "
842 << header.numGlobalRows << " rows.");
843
844 const auto fileRowMap = readMapSection(filename, header.rowMapOffset, comm);
845 const auto fileDomainMap = readMapSection(filename, header.domainMapOffset, comm);
846 const auto fileRangeMap = readMapSection(filename, header.rangeMapOffset, comm);
847
848 using host_execution_space = Kokkos::DefaultHostExecutionSpace;
849
850 const size_t localNumRows = fileRowMap->getLocalNumElements();
851 const unsigned long long localNumRows64 = static_cast<unsigned long long>(localNumRows);
852 const unsigned long long globalRowOffset = exclusiveScanUnsignedLongLong(localNumRows64, comm);
853
854 Kokkos::View<unsigned long long*, Kokkos::HostSpace> localRowPtr("Tpetra::BinaryIO::localRowPtr", localNumRows + 1);
855 Details::binaryIOReadArrayCollective(filename, header.rowPtrOffset, localRowPtr.data(),
856 localNumRows64 + 1, globalRowOffset, comm);
857 TEUCHOS_TEST_FOR_EXCEPTION(localRowPtr(localNumRows) < localRowPtr(0),
858 std::runtime_error,
859 "Tpetra::BinaryIO: Sparse row pointers in file are not monotonic on this rank.");
860 const unsigned long long nnzStart = localRowPtr(0);
861 unsigned long long nonmonotonicRowPtrCount = 0;
862 Kokkos::parallel_reduce(
863 "Tpetra::BinaryIO::checkReadRowPtr",
864 Kokkos::RangePolicy<host_execution_space>(0, localNumRows),
865 KOKKOS_LAMBDA(const size_t i, unsigned long long& count) {
866 if (localRowPtr(i + 1) < localRowPtr(i)) {
867 ++count;
868 }
869 },
870 nonmonotonicRowPtrCount);
871 TEUCHOS_TEST_FOR_EXCEPTION(nonmonotonicRowPtrCount != 0,
872 std::runtime_error,
873 "Tpetra::BinaryIO: Sparse row pointers in file are not monotonic on this rank.");
874 Kokkos::parallel_for(
875 "Tpetra::BinaryIO::normalizeReadRowPtr",
876 Kokkos::RangePolicy<host_execution_space>(0, localNumRows + 1),
877 KOKKOS_LAMBDA(const size_t i) {
878 localRowPtr(i) -= nnzStart;
879 });
880
881 const unsigned long long localNnz64 = localRowPtr(localNumRows);
882 const size_t localNnz = Details::binaryIOCheckedSize(localNnz64, "local sparse entry count");
883 Kokkos::View<GlobalOrdinal*, Kokkos::HostSpace> globalColumns("Tpetra::BinaryIO::globalColumns", localNnz);
884 Kokkos::View<Scalar*, Kokkos::HostSpace> values("Tpetra::BinaryIO::values", localNnz);
885 Details::binaryIOReadArrayCollective(filename, header.columnIndicesOffset, globalColumns.data(), localNnz64, nnzStart, comm);
886 Details::binaryIOReadArrayCollective(filename, header.valuesOffset, values.data(), localNnz64, nnzStart, comm);
887
888 const auto dom = domainMap.is_null() ? fileDomainMap : domainMap;
889 const auto ran = rangeMap.is_null() ? fileRangeMap : rangeMap;
890
891 int mapsAreSameLocal = rowMap->locallySameAs(*fileRowMap) ? 1 : 0;
892 int mapsAreSame = 0;
893 Teuchos::reduceAll(*comm, Teuchos::REDUCE_MIN, 1, &mapsAreSameLocal, &mapsAreSame);
894 if (mapsAreSame != 0) {
895 auto matrix = Details::buildSparseMatrixFromLocalCrsViews<Scalar, LocalOrdinal, GlobalOrdinal, Node>(rowMap, colMap, dom, localRowPtr, globalColumns, values);
896 if (callFillComplete) {
897 matrix->fillComplete(dom, ran);
898 }
899 return matrix;
900 }
901
902 auto fileMatrix = Details::buildSparseMatrixFromLocalCrsViews<Scalar, LocalOrdinal, GlobalOrdinal, Node>(fileRowMap, Teuchos::null, fileDomainMap, localRowPtr, globalColumns, values);
903 fileMatrix->fillComplete(fileDomainMap, fileRangeMap);
904
905 using import_type = Tpetra::Import<LocalOrdinal, GlobalOrdinal, Node>;
906 import_type importer(fileRowMap, rowMap);
907
908 if (callFillComplete && colMap.is_null()) {
909 return Tpetra::importAndFillCompleteCrsMatrix<sparse_matrix_type>(fileMatrix,
910 importer,
911 dom,
912 ran);
913 }
914
915 Teuchos::RCP<sparse_matrix_type> matrix;
916 if (colMap.is_null()) {
917 matrix = Teuchos::rcp(new sparse_matrix_type(rowMap,
918 static_cast<size_t>(fileMatrix->getGlobalMaxNumRowEntries())));
919 } else {
920 matrix = Teuchos::rcp(new sparse_matrix_type(rowMap,
921 colMap,
922 static_cast<size_t>(fileMatrix->getGlobalMaxNumRowEntries())));
923 }
924 matrix->doImport(*fileMatrix, importer, Tpetra::INSERT);
925 if (callFillComplete) {
926 matrix->fillComplete(dom, ran);
927 }
928 return matrix;
929}
