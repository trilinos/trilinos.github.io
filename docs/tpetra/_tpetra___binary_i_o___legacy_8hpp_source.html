113inline bool hasTpetraBinaryHeader(const std::string& fileName,
114 const Teuchos::RCP<const Teuchos::Comm<int>>& comm) {
115 int openOk = 1;
116 int hasHeader = 0;
117 if (comm->getRank() == 0) {
118 std::ifstream in(fileName.c_str(), std::ios::binary);
119 openOk = in.good() ? 1 : 0;
120 if (openOk) {
121 char magic[8] = {0, 0, 0, 0, 0, 0, 0, 0};
122 in.read(magic, sizeof(magic));
123 const char expectedMagic[8] = {'T', 'P', 'B', 'I', 'O', '0', '0', '1'};
124 hasHeader = in.good() && std::memcmp(magic, expectedMagic, sizeof(expectedMagic)) == 0 ? 1 : 0;
125 }
126 }
127 Teuchos::broadcast(*comm, 0, 1, &openOk);
128 Teuchos::broadcast(*comm, 0, 1, &hasHeader);
129 TEUCHOS_TEST_FOR_EXCEPTION(openOk == 0,
130 std::runtime_error,
131 "Tpetra::BinaryIO: Failed to open binary file '" << fileName << "'.");
132 return hasHeader != 0;
133}

142inline unsigned long long legacyBinaryFileSize(const std::string& fileName) {
143 std::ifstream in(fileName.c_str(), std::ios::binary | std::ios::ate);
144 TEUCHOS_TEST_FOR_EXCEPTION(!in.good(),
145 std::runtime_error,
146 "Tpetra::BinaryIO: Failed to open legacy binary file '" << fileName << "'.");
147 const std::streamoff size = in.tellg();
148 TEUCHOS_TEST_FOR_EXCEPTION(size < 0,
149 std::runtime_error,
150 "Tpetra::BinaryIO: Failed to determine the size of legacy binary file '" << fileName << "'.");
151 return static_cast<unsigned long long>(size);
152}

154inline LegacyBinaryHeader readAndValidateLegacyBinaryHeader(const std::string& fileName) {
155 std::ifstream in(fileName.c_str(), std::ios::binary);
156 TEUCHOS_TEST_FOR_EXCEPTION(!in.good(),
157 std::runtime_error,
158 "Tpetra::BinaryIO: Failed to open legacy binary file '" << fileName << "'.");
159
160 LegacyBinaryHeader header = {0, 0, 0};
161 in.read(reinterpret_cast<char*>(&header.numRows), sizeof(header.numRows));
162 in.read(reinterpret_cast<char*>(&header.numCols), sizeof(header.numCols));
163 in.read(reinterpret_cast<char*>(&header.numEntries), sizeof(header.numEntries));
164 TEUCHOS_TEST_FOR_EXCEPTION(!in.good(),
165 std::runtime_error,
166 "Tpetra::BinaryIO: File '" << fileName << "' is not a valid legacy Xpetra binary matrix file.");
167 TEUCHOS_TEST_FOR_EXCEPTION(header.numRows < 0 || header.numCols < 0 || header.numEntries < 0,
168 std::runtime_error,
169 "Tpetra::BinaryIO: Legacy binary file '" << fileName << "' has negative dimensions or entry count.");
170
171 const unsigned long long fileSize = legacyBinaryFileSize(fileName);
172 const unsigned long long minSize = static_cast<unsigned long long>(3 * sizeof(int)) +
173 static_cast<unsigned long long>(header.numRows) * static_cast<unsigned long long>(2 * sizeof(int)) +
174 static_cast<unsigned long long>(header.numEntries) * static_cast<unsigned long long>(sizeof(int) + sizeof(double));
175 // Legacy files may have trailing padding or metadata, but they must contain
176 // at least the bytes required by the dimensions and entry count in the header.
177 TEUCHOS_TEST_FOR_EXCEPTION(fileSize < minSize,
178 std::runtime_error,
179 "Tpetra::BinaryIO: File '" << fileName << "' is not a valid legacy Xpetra binary matrix file."
180 << " Expected at least " << minSize << " bytes from its header, but file has " << fileSize << " bytes.");
181 return header;
182}

208template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
209typename std::enable_if<binaryIOAvailableForScalar<Scalar>::value,
210 Teuchos::RCP<Tpetra::CrsMatrix<Scalar, LocalOrdinal, GlobalOrdinal, Node>>>::type
211readLegacyBinarySparseFile(const std::string& oldFileName,
212 const Teuchos::RCP<const Tpetra::Map<LocalOrdinal, GlobalOrdinal, Node>>& rowMapInput,
213 const Teuchos::RCP<const Tpetra::Map<LocalOrdinal, GlobalOrdinal, Node>>& colMapInput,
214 const Teuchos::RCP<const Tpetra::Map<LocalOrdinal, GlobalOrdinal, Node>>& domainMapInput,
215 const Teuchos::RCP<const Tpetra::Map<LocalOrdinal, GlobalOrdinal, Node>>& rangeMapInput,
216 const Teuchos::RCP<const Teuchos::Comm<int>>& comm,
217 const bool callFillComplete) {
218 using map_type = Tpetra::Map<LocalOrdinal, GlobalOrdinal, Node>;
219 using matrix_type = Tpetra::CrsMatrix<Scalar, LocalOrdinal, GlobalOrdinal, Node>;
220 using import_type = Tpetra::Import<LocalOrdinal, GlobalOrdinal, Node>;
221 using local_graph_type = typename matrix_type::local_graph_device_type;
222 using rowptr_type = typename local_graph_type::row_map_type::non_const_type;
223 using colidx_type = typename local_graph_type::entries_type::non_const_type;
224 using values_type = typename matrix_type::local_matrix_device_type::values_type::non_const_type;
225 using impl_scalar_type = typename matrix_type::impl_scalar_type;
226 using device_type = typename matrix_type::device_type;
227 using execution_space = typename device_type::execution_space;
228 using host_execution_space = Kokkos::DefaultHostExecutionSpace;
229
230 const LegacyBinaryHeader legacyHeader = readAndValidateLegacyBinaryHeader(oldFileName);
231 const int m = legacyHeader.numRows;
232 const int n = legacyHeader.numCols;
233 const int nnz = legacyHeader.numEntries;
234 const int myRank = comm->getRank();
235
236 const size_t fileLocalNumRows = (myRank == 0) ? static_cast<size_t>(m) : static_cast<size_t>(0);
237 const size_t fileLocalNumCols = (myRank == 0) ? static_cast<size_t>(n) : static_cast<size_t>(0);
238 const auto fileRowMap = Teuchos::rcp(new map_type(static_cast<Tpetra::global_size_t>(m), fileLocalNumRows, static_cast<GlobalOrdinal>(0), comm));
239 const auto fileColMap = Teuchos::rcp(new map_type(static_cast<Tpetra::global_size_t>(n), fileLocalNumCols, static_cast<GlobalOrdinal>(0), comm));
240 const auto fileDomainMap = fileColMap;
241 const auto fileRangeMap = fileRowMap;
242
243 Kokkos::View<unsigned long long*, Kokkos::HostSpace> rowLengths("Tpetra::BinaryIO::legacyRowLengths", fileLocalNumRows);
244 Kokkos::deep_copy(rowLengths, 0ull);
245 unsigned long long entriesRead = 0;
246
247 std::ifstream in;
248 if (myRank == 0) {
249 in.open(oldFileName.c_str(), std::ios::binary);
250 TEUCHOS_TEST_FOR_EXCEPTION(!in.good(),
251 std::runtime_error,
252 "Tpetra::BinaryIO: Failed to open legacy binary file '" << oldFileName << "'.");
253 in.seekg(static_cast<std::streamoff>(3 * sizeof(int)), std::ios::beg);
254 TEUCHOS_TEST_FOR_EXCEPTION(!in.good(),
255 std::runtime_error,
256 "Tpetra::BinaryIO: Failed to seek past the header of legacy binary file '" << oldFileName << "'.");
257
258 for (int record = 0; record < m; ++record) {
259 int row = 0;
260 int rownnz = 0;
261 readLegacyBinaryValue(in, row, oldFileName, "row index");
262 readLegacyBinaryValue(in, rownnz, oldFileName, "row entry count");
263 TEUCHOS_TEST_FOR_EXCEPTION(row < 0 || row >= m || rownnz < 0,
264 std::runtime_error,
265 "Tpetra::BinaryIO: Legacy binary file '" << oldFileName << "' has an invalid row record.");
266 entriesRead += static_cast<unsigned long long>(rownnz);
267 rowLengths(static_cast<size_t>(row)) += static_cast<unsigned long long>(rownnz);
268 for (int j = 0; j < rownnz; ++j) {
269 int col = 0;
270 readLegacyBinaryValue(in, col, oldFileName, "column index");
271 TEUCHOS_TEST_FOR_EXCEPTION(col < 0 || col >= n,
272 std::runtime_error,
273 "Tpetra::BinaryIO: Legacy binary file '" << oldFileName << "' has a column index outside [0, n).");
274 }
275 for (int j = 0; j < rownnz; ++j) {
276 double value = 0.0;
277 readLegacyBinaryValue(in, value, oldFileName, "matrix value");
278 }
279 }
280 TEUCHOS_TEST_FOR_EXCEPTION(entriesRead != static_cast<unsigned long long>(nnz),
281 std::runtime_error,
282 "Tpetra::BinaryIO: Legacy binary file '" << oldFileName << "' header entry count does not match row records.");
283 TEUCHOS_TEST_FOR_EXCEPTION(entriesRead > static_cast<unsigned long long>(std::numeric_limits<size_t>::max()),
284 std::runtime_error,
285 "Tpetra::BinaryIO: Legacy binary file '" << oldFileName << "' has too many entries for this platform.");
286 }
287
288 Kokkos::View<unsigned long long*, Kokkos::HostSpace> rowPtrHost("Tpetra::BinaryIO::legacyRowPtr", fileLocalNumRows + 1);
289 unsigned long long localNnz64 = 0;
290 Kokkos::parallel_scan(
291 "Tpetra::BinaryIO::legacyRowPtrScan",
292 Kokkos::RangePolicy<host_execution_space>(0, fileLocalNumRows),
293 KOKKOS_LAMBDA(const size_t row, unsigned long long& count, const bool finalPass) {
294 if (finalPass) {
295 rowPtrHost(row) = count;
296 }
297 count += rowLengths(row);
298 if (finalPass && row + 1 == fileLocalNumRows) {
299 rowPtrHost(row + 1) = count;
300 }
301 },
302 localNnz64);
303 if (fileLocalNumRows == 0) {
304 rowPtrHost(0) = 0;
305 }
306 const size_t localNnz = static_cast<size_t>(localNnz64);
307 Kokkos::View<unsigned long long*, Kokkos::HostSpace> nextPtr("Tpetra::BinaryIO::legacyNextPtr", fileLocalNumRows);
308 Kokkos::parallel_for(
309 "Tpetra::BinaryIO::legacyInitNextPtr",
310 Kokkos::RangePolicy<host_execution_space>(0, fileLocalNumRows),
311 KOKKOS_LAMBDA(const size_t row) {
312 nextPtr(row) = rowPtrHost(row);
313 });
314
315 Kokkos::View<GlobalOrdinal*, Kokkos::HostSpace> globalColumnsHost("Tpetra::BinaryIO::legacyGlobalColumns", localNnz);
316 Kokkos::View<Scalar*, Kokkos::HostSpace> valuesHost("Tpetra::BinaryIO::legacyValues", localNnz);
317
318 if (myRank == 0) {
319 in.clear();
320 in.seekg(static_cast<std::streamoff>(3 * sizeof(int)), std::ios::beg);
321 TEUCHOS_TEST_FOR_EXCEPTION(!in.good(),
322 std::runtime_error,
323 "Tpetra::BinaryIO: Failed to rewind legacy binary file '" << oldFileName << "'.");
324
325 for (int record = 0; record < m; ++record) {
326 int row = 0;
327 int rownnz = 0;
328 readLegacyBinaryValue(in, row, oldFileName, "row index");
329 readLegacyBinaryValue(in, rownnz, oldFileName, "row entry count");
330
331 Kokkos::View<int*, Kokkos::HostSpace> columns("Tpetra::BinaryIO::legacyRowColumns", static_cast<size_t>(rownnz));
332 for (int j = 0; j < rownnz; ++j) {
333 readLegacyBinaryValue(in, columns(static_cast<size_t>(j)), oldFileName, "column index");
334 }
335
336 size_t offset = static_cast<size_t>(nextPtr(static_cast<size_t>(row)));
337 for (int j = 0; j < rownnz; ++j) {
338 const int col = columns(static_cast<size_t>(j));
339 globalColumnsHost(offset + static_cast<size_t>(j)) = static_cast<GlobalOrdinal>(col);
340 }
341 for (int j = 0; j < rownnz; ++j) {
342 double value = 0.0;
343 readLegacyBinaryValue(in, value, oldFileName, "matrix value");
344 valuesHost(offset + static_cast<size_t>(j)) = legacyDoubleToScalar<Scalar>(value);
345 }
346 nextPtr(static_cast<size_t>(row)) += static_cast<unsigned long long>(rownnz);
347 }
348 }
349
350 rowptr_type rowPtrDevice("Tpetra::BinaryIO::legacyRowPtrDevice", fileLocalNumRows + 1);
351 auto rowPtrDeviceHost = Kokkos::create_mirror_view(rowPtrDevice);
352 Kokkos::parallel_for(
353 "Tpetra::BinaryIO::legacyCopyRowPtrToDeviceHost",
354 Kokkos::RangePolicy<host_execution_space>(0, fileLocalNumRows + 1),
355 KOKKOS_LAMBDA(const size_t i) {
356 rowPtrDeviceHost(i) = static_cast<typename rowptr_type::non_const_value_type>(rowPtrHost(i));
357 });
358 Kokkos::deep_copy(rowPtrDevice, rowPtrDeviceHost);
359
360 Kokkos::View<GlobalOrdinal*, device_type> globalColumnsDevice("Tpetra::BinaryIO::legacyGlobalColumnsDevice", localNnz);
361 Kokkos::deep_copy(globalColumnsDevice, globalColumnsHost);
362
363 colidx_type localColumnsDevice("Tpetra::BinaryIO::legacyLocalColumns", localNnz);
364 const auto localFileColMap = fileColMap->getLocalMap();
365 const LocalOrdinal invalidLocalOrdinal = Tpetra::Details::OrdinalTraits<LocalOrdinal>::invalid();
366 unsigned long long invalidCount = 0;
367 Kokkos::parallel_reduce(
368 "Tpetra::BinaryIO::legacyReindexColumns",
369 Kokkos::RangePolicy<execution_space>(0, localNnz),
370 KOKKOS_LAMBDA(const size_t i, unsigned long long& lclInvalidCount) {
371 const LocalOrdinal lclCol = localFileColMap.getLocalElement(globalColumnsDevice(i));
372 localColumnsDevice(i) = lclCol;
373 if (lclCol == invalidLocalOrdinal) {
374 ++lclInvalidCount;
375 }
376 },
377 invalidCount);
378 TEUCHOS_TEST_FOR_EXCEPTION(invalidCount != 0,
379 std::runtime_error,
380 "Tpetra::BinaryIO: Column map construction missed " << invalidCount
381 << " column GID(s) while reading legacy binary file '"
382 << oldFileName << "'.");
383
384 values_type valuesDevice("Tpetra::BinaryIO::legacyValuesDevice", localNnz);
385 auto valuesDeviceHost = Kokkos::create_mirror_view(valuesDevice);
386 Kokkos::parallel_for(
387 "Tpetra::BinaryIO::legacyCopyValuesToDeviceHost",
388 Kokkos::RangePolicy<host_execution_space>(0, localNnz),
389 KOKKOS_LAMBDA(const size_t i) {
390 valuesDeviceHost(i) = static_cast<impl_scalar_type>(valuesHost(i));
391 });
392 Kokkos::deep_copy(valuesDevice, valuesDeviceHost);
393
394 Tpetra::Import_Util::sortCrsEntries(rowPtrDevice, localColumnsDevice, valuesDevice);
395 auto fileMatrix = Teuchos::rcp(new matrix_type(fileRowMap, fileColMap, rowPtrDevice, localColumnsDevice, valuesDevice));
396 fileMatrix->fillComplete(fileDomainMap, fileRangeMap);
397
398 if (rowMapInput.is_null()) {
399 return fileMatrix;
400 }
401
402 Teuchos::RCP<const map_type> requestedDomainMap = domainMapInput.is_null() ? rowMapInput : domainMapInput;
403 Teuchos::RCP<const map_type> requestedRangeMap = rangeMapInput.is_null() ? rowMapInput : rangeMapInput;
404 import_type importer(fileRowMap, rowMapInput);
405 Teuchos::RCP<matrix_type> matrix;
406 if (colMapInput.is_null()) {
407 matrix = Teuchos::rcp(new matrix_type(rowMapInput,
408 static_cast<size_t>(fileMatrix->getGlobalMaxNumRowEntries())));
409 } else {
410 matrix = Teuchos::rcp(new matrix_type(rowMapInput,
411 colMapInput,
412 static_cast<size_t>(fileMatrix->getGlobalMaxNumRowEntries())));
413 }
414 matrix->doImport(*fileMatrix, importer, Tpetra::INSERT);
415 if (callFillComplete) {
416 matrix->fillComplete(requestedDomainMap, requestedRangeMap);
417 }
418 return matrix;
419}
