79inline unsigned long long binaryIOChunkCount(const unsigned long long byteCount,
80 const unsigned long long maxChunk) {
81 TEUCHOS_TEST_FOR_EXCEPTION(maxChunk == 0,
82 std::logic_error,
83 "Tpetra::BinaryIO: Chunk size must be nonzero.");
84 return byteCount == 0 ? 0 : 1ull + (byteCount - 1ull) / maxChunk;
85}

88inline unsigned long long binaryIOChunkSize(const unsigned long long byteCount,
89 const unsigned long long maxChunk,
90 const unsigned long long chunkIndex) {
91 const unsigned long long numChunks = binaryIOChunkCount(byteCount, maxChunk);
92 TEUCHOS_TEST_FOR_EXCEPTION(chunkIndex >= numChunks,
93 std::logic_error,
94 "Tpetra::BinaryIO: Chunk index is outside the chunked transfer.");
95 const unsigned long long chunkOffset = chunkIndex * maxChunk;
96 return std::min(byteCount - chunkOffset, maxChunk);
97}

109inline unsigned long long binaryIOByteSwap64(const unsigned long long value) {
110 unsigned long long swapped = 0;
111 for (int i = 0; i < 8; ++i) {
112 swapped = (swapped << 8) | ((value >> (8 * i)) & 0xffull);
113 }
114 return swapped;
115}

118inline unsigned long long binaryIOCheckedByteCount(const unsigned long long count,
119 const size_t elementSize) {

139inline void binaryIOBroadcastBytesFromRoot(char data[],
140 const unsigned long long byteCount,
141 const Teuchos::RCP<const Teuchos::Comm<int>>& comm) {
142 unsigned long long remaining = byteCount;
143 char* current = data;
144 const unsigned long long maxChunk = binaryIOIntCountMaxChunk();
145 while (remaining > 0) {
146 const unsigned long long chunk = std::min(remaining, maxChunk);
147 Teuchos::broadcast(*comm, 0, static_cast<int>(chunk), current);
148 current += static_cast<std::ptrdiff_t>(chunk);
149 remaining -= chunk;
150 }

181inline bool binaryIOByteRangeInFile(MPI_File file,
182 const unsigned long long offset,
183 const unsigned long long byteCount) {
184 if (byteCount == 0) {
185 return true;
186 }
187 MPI_Offset fileSize = 0;
188 if (MPI_File_get_size(file, &fileSize) != MPI_SUCCESS || fileSize < 0) {
189 return false;
190 }
191 return offset <= static_cast<unsigned long long>(fileSize) &&
192 byteCount <= static_cast<unsigned long long>(fileSize) - offset;
193}

197template <class T>
198void binaryIOWriteArrayFromRoot(const std::string& filename,
199 const unsigned long long dataOffset,
200 const T* data,
201 const unsigned long long count,
202 const Teuchos::RCP<const Teuchos::Comm<int>>& comm) {
203 const unsigned long long byteCount = binaryIOCheckedByteCount(count, sizeof(T));
204 if (byteCount > 0) {
205 binaryIOCheckedAddByteOffset(dataOffset, byteCount - 1ull);
206 }
207#ifdef HAVE_TPETRACORE_MPI
208 if (Tpetra::Details::teuchosCommIsAnMpiComm(*comm)) {
209 MPI_Comm rawComm = Tpetra::Details::extractMpiCommFromTeuchos(*comm);
210 MPI_File file;
211 const int openErr = MPI_File_open(rawComm, const_cast<char*>(filename.c_str()), MPI_MODE_WRONLY, MPI_INFO_NULL, &file);
212 TEUCHOS_TEST_FOR_EXCEPTION(openErr != MPI_SUCCESS, std::runtime_error,
213 "Tpetra::BinaryIO: MPI_File_open failed while writing root-owned array data.");
214 int writeErr = MPI_SUCCESS;
215 if (comm->getRank() == 0) {
216 const char* current = reinterpret_cast<const char*>(data);
217 unsigned long long currentOffset = dataOffset;
218 const unsigned long long maxChunk = binaryIOIntCountMaxChunk();
219 const unsigned long long numChunks = binaryIOChunkCount(byteCount, maxChunk);
220 for (unsigned long long chunkIndex = 0; chunkIndex < numChunks && writeErr == MPI_SUCCESS; ++chunkIndex) {
221 const unsigned long long chunk = binaryIOChunkSize(byteCount, maxChunk, chunkIndex);
222 MPI_Status status;
223 writeErr = MPI_File_write_at(file,
224 static_cast<MPI_Offset>(currentOffset),
225 const_cast<char*>(current),
226 static_cast<int>(chunk),
227 MPI_BYTE,
228 &status);
229 current += static_cast<std::ptrdiff_t>(chunk);
230 currentOffset += chunk;
231 }
232 }
233 MPI_File_close(&file);
234 Teuchos::broadcast(*comm, 0, 1, &writeErr);
235 TEUCHOS_TEST_FOR_EXCEPTION(writeErr != MPI_SUCCESS, std::runtime_error,
236 "Tpetra::BinaryIO: MPI_File_write_at failed while writing root-owned array data.");
237 comm->barrier();
238 return;
239 }
240#endif
241 if (comm->getRank() == 0) {
242 std::fstream out(filename.c_str(), std::ios::binary | std::ios::in | std::ios::out);
243 TEUCHOS_TEST_FOR_EXCEPTION(!out.good(), std::runtime_error,
244 "Tpetra::BinaryIO: Failed to open file '" << filename << "' while writing root-owned array data.");
245 out.seekp(static_cast<std::streamoff>(dataOffset), std::ios::beg);
246
247 const char* current = reinterpret_cast<const char*>(data);
248 unsigned long long remaining = byteCount;
249 const unsigned long long maxChunk = static_cast<unsigned long long>(std::numeric_limits<std::streamsize>::max());
250 while (remaining > 0) {
251 const unsigned long long chunk = std::min(remaining, maxChunk);
252 out.write(current, static_cast<std::streamsize>(chunk));
253 TEUCHOS_TEST_FOR_EXCEPTION(!out.good(), std::runtime_error,
254 "Tpetra::BinaryIO: Failed to write root-owned array data to file '" << filename << "'.");
255 current += static_cast<std::ptrdiff_t>(chunk);
256 remaining -= chunk;
257 }
258 }
259 comm->barrier();
260}

263template <class T>
264void binaryIOReadArrayFromRoot(const std::string& filename,
265 const unsigned long long dataOffset,
266 T* data,
267 const unsigned long long count,
268 const Teuchos::RCP<const Teuchos::Comm<int>>& comm) {
269 const unsigned long long byteCount = binaryIOCheckedByteCount(count, sizeof(T));
270 if (byteCount > 0) {
271 binaryIOCheckedAddByteOffset(dataOffset, byteCount - 1ull);
272 }
273#ifdef HAVE_TPETRACORE_MPI
274 if (Tpetra::Details::teuchosCommIsAnMpiComm(*comm)) {

276 MPI_File file;
277 const int openErr = MPI_File_open(rawComm, const_cast<char*>(filename.c_str()), MPI_MODE_RDONLY, MPI_INFO_NULL, &file);
278 TEUCHOS_TEST_FOR_EXCEPTION(openErr != MPI_SUCCESS, std::runtime_error,
279 "Tpetra::BinaryIO: MPI_File_open failed while reading root-owned array data.");
280 int readErr = MPI_SUCCESS;
281 if (comm->getRank() == 0) {
282 if (!binaryIOByteRangeInFile(file, dataOffset, byteCount)) {
283 readErr = MPI_ERR_IO;
284 }
285 char* current = reinterpret_cast<char*>(data);
286 unsigned long long currentOffset = dataOffset;
287 const unsigned long long maxChunk = binaryIOIntCountMaxChunk();
288 const unsigned long long numChunks = binaryIOChunkCount(byteCount, maxChunk);
289 for (unsigned long long chunkIndex = 0; chunkIndex < numChunks && readErr == MPI_SUCCESS; ++chunkIndex) {
290 const unsigned long long chunk = binaryIOChunkSize(byteCount, maxChunk, chunkIndex);
291 MPI_Status status;
292 readErr = MPI_File_read_at(file,
293 static_cast<MPI_Offset>(currentOffset),
294 current,
295 static_cast<int>(chunk),
296 MPI_BYTE,
297 &status);
298 current += static_cast<std::ptrdiff_t>(chunk);
299 currentOffset += chunk;
300 }
301 }
302 MPI_File_close(&file);
303 Teuchos::broadcast(*comm, 0, 1, &readErr);
304 TEUCHOS_TEST_FOR_EXCEPTION(readErr != MPI_SUCCESS, std::runtime_error,
305 "Tpetra::BinaryIO: Failed to read root-owned array data from file '"
306 << filename << "' (file truncated or MPI_File_read_at failed).");
307 if (comm->getSize() > 1) {
308 binaryIOBroadcastBytesFromRoot(reinterpret_cast<char*>(data), byteCount, comm);
309 }
310 return;
311 }
312#endif
313 if (comm->getRank() == 0) {
314 std::ifstream in(filename.c_str(), std::ios::binary);
315 TEUCHOS_TEST_FOR_EXCEPTION(!in.good(), std::runtime_error,
316 "Tpetra::BinaryIO: Failed to open file '" << filename << "' while reading root-owned array data.");
317 in.seekg(static_cast<std::streamoff>(dataOffset), std::ios::beg);
318
319 char* current = reinterpret_cast<char*>(data);
320 unsigned long long remaining = byteCount;
321 const unsigned long long maxChunk = static_cast<unsigned long long>(std::numeric_limits<std::streamsize>::max());
322 while (remaining > 0) {
323 const unsigned long long chunk = std::min(remaining, maxChunk);
324 in.read(current, static_cast<std::streamsize>(chunk));
325 TEUCHOS_TEST_FOR_EXCEPTION(!in.good(), std::runtime_error,
326 "Tpetra::BinaryIO: Failed to read root-owned array data from file '" << filename << "'.");
327 current += static_cast<std::ptrdiff_t>(chunk);
328 remaining -= chunk;
329 }
330 }
331 if (comm->getSize() > 1) {
332 binaryIOBroadcastBytesFromRoot(reinterpret_cast<char*>(data), byteCount, comm);
333 }
334}

339template <class T>
340void binaryIOWriteArrayCollective(const std::string& filename,
341 const unsigned long long dataOffset,
342 const T* data,
343 const unsigned long long count,
344 const unsigned long long globalOffset,
345 const Teuchos::RCP<const Teuchos::Comm<int>>& comm) {
346 const unsigned long long byteCount = binaryIOCheckedByteCount(count, sizeof(T));
347 const unsigned long long byteOffset = binaryIOCheckedByteOffset(dataOffset, globalOffset, sizeof(T));
348 binaryIOCheckedAddByteOffset(byteOffset, byteCount);
349#ifdef HAVE_TPETRACORE_MPI
350 if (Tpetra::Details::teuchosCommIsAnMpiComm(*comm)) {
351 MPI_Comm rawComm = Tpetra::Details::extractMpiCommFromTeuchos(*comm);
352 MPI_File file;
353 const int openErr = MPI_File_open(rawComm, const_cast<char*>(filename.c_str()), MPI_MODE_WRONLY, MPI_INFO_NULL, &file);
354 TEUCHOS_TEST_FOR_EXCEPTION(openErr != MPI_SUCCESS, std::runtime_error,
355 "Tpetra::BinaryIO: MPI_File_open failed while writing array data.");
356
357 const char* current = reinterpret_cast<const char*>(data);
358 unsigned long long remaining = byteCount;
359 unsigned long long currentOffset = byteOffset;
360 const unsigned long long maxChunk = binaryIOIntCountMaxChunk();
361 const unsigned long long localNumChunks = binaryIOChunkCount(byteCount, maxChunk);
362 unsigned long long numChunks = 0;
363 Teuchos::reduceAll(*comm, Teuchos::REDUCE_MAX, 1, &localNumChunks, &numChunks);
364 // Every process makes every collective call, even after a local error or
365 // once its own data is exhausted; errors are combined after the loop.
366 int writeErr = MPI_SUCCESS;
367 for (unsigned long long chunkIndex = 0; chunkIndex < numChunks; ++chunkIndex) {
368 const unsigned long long chunk = std::min(remaining, maxChunk);
369 MPI_Status status;
370 const int err = MPI_File_write_at_all(file,
371 static_cast<MPI_Offset>(currentOffset),
372 const_cast<char*>(current),
373 static_cast<int>(chunk),
374 MPI_BYTE,
375 &status);
376 if (writeErr == MPI_SUCCESS) {
377 writeErr = err;
378 }
379 current += static_cast<std::ptrdiff_t>(chunk);
380 currentOffset += chunk;
381 remaining -= chunk;
382 }
383 MPI_File_close(&file);
384 TEUCHOS_TEST_FOR_EXCEPTION(binaryIOAnyProcessFailed(writeErr != MPI_SUCCESS, comm), std::runtime_error,
385 "Tpetra::BinaryIO: MPI_File_write_at_all failed on at least one process while writing array data to file '"
386 << filename << "'.");
387 return;
388 }
389#endif
390 binaryIORequireSingleProcessComm(comm);
391 std::fstream out(filename.c_str(), std::ios::binary | std::ios::in | std::ios::out);
392 TEUCHOS_TEST_FOR_EXCEPTION(!out.good(), std::runtime_error,
393 "Tpetra::BinaryIO: Failed to open file '" << filename << "' while writing array data.");
394 out.seekp(static_cast<std::streamoff>(byteOffset), std::ios::beg);
395
396 const char* current = reinterpret_cast<const char*>(data);
397 unsigned long long remaining = byteCount;
398 const unsigned long long maxChunk = static_cast<unsigned long long>(std::numeric_limits<std::streamsize>::max());
399 while (remaining > 0) {
400 const unsigned long long chunk = std::min(remaining, maxChunk);
401 out.write(current, static_cast<std::streamsize>(chunk));
402 TEUCHOS_TEST_FOR_EXCEPTION(!out.good(), std::runtime_error,
403 "Tpetra::BinaryIO: Failed to write array data to file '" << filename << "'.");
404 current += static_cast<std::ptrdiff_t>(chunk);
405 remaining -= chunk;
406 }
407}
