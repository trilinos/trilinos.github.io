75template <class T>
76unsigned long long binaryIOTypeFlags() {
77 unsigned long long flags = 0;
78 if (Teuchos::ScalarTraits<T>::isOrdinal) {
79 flags |= 1u << 0;
80 }
81 if (Teuchos::ScalarTraits<T>::isComplex) {
82 flags |= 1u << 2;
83 }
84 if (std::is_floating_point<T>::value || Teuchos::ScalarTraits<T>::isComplex) {
85 flags |= 1u << 1;
86 }
87 if (std::is_signed<T>::value || Teuchos::ScalarTraits<T>::isComplex || std::is_floating_point<T>::value) {
88 flags |= 1u << 3;
89 }
90 return flags;
91}

104inline BinaryIOFileHeader binaryIOReadHeaderFromFile(const std::string& filename,
105 const Teuchos::RCP<const Teuchos::Comm<int>>& comm) {
106 BinaryIOFileHeader header;
107 std::memset(&header, 0, sizeof(BinaryIOFileHeader));
108#ifdef HAVE_TPETRACORE_MPI
109 if (Tpetra::Details::teuchosCommIsAnMpiComm(*comm)) {
110 MPI_Comm rawComm = Tpetra::Details::extractMpiCommFromTeuchos(*comm);
111 MPI_File file;
112 const int err = MPI_File_open(rawComm, const_cast<char*>(filename.c_str()), MPI_MODE_RDONLY, MPI_INFO_NULL, &file);
113 TEUCHOS_TEST_FOR_EXCEPTION(err != MPI_SUCCESS, std::runtime_error,
114 "Tpetra::BinaryIO: MPI_File_open failed for '" << filename << "'.");
115 MPI_Status status;
116 const int readErr = MPI_File_read_at_all(file, 0, &header, sizeof(BinaryIOFileHeader), MPI_BYTE, &status);
117 MPI_File_close(&file);
118 TEUCHOS_TEST_FOR_EXCEPTION(readErr != MPI_SUCCESS, std::runtime_error,
119 "Tpetra::BinaryIO: MPI_File_read_at_all failed while reading header.");
120 return header;
121 }
122#endif
123 bool success = true;
124 if (comm->getRank() == 0) {
125 std::ifstream in(filename.c_str(), std::ios::binary);
126 success = static_cast<bool>(in);
127 if (success) {
128 in.read(reinterpret_cast<char*>(&header), sizeof(BinaryIOFileHeader));
129 success = static_cast<bool>(in);
130 }
131 }
132 binaryIOCheckFileOpen(success, filename, comm, "reading");
133 Teuchos::broadcast(*comm, 0, static_cast<int>(sizeof(BinaryIOFileHeader)), reinterpret_cast<char*>(&header));
134 return header;
135}

137template <class LocalOrdinal, class GlobalOrdinal>
138void binaryIOValidateMapHeader(const BinaryIOFileHeader& header) {
139 const char expectedMagic[8] = {'T', 'P', 'B', 'I', 'O', '0', '0', '1'};
140 TEUCHOS_TEST_FOR_EXCEPTION(std::memcmp(header.magic, expectedMagic, sizeof(expectedMagic)) != 0,
141 std::runtime_error,
142 "Tpetra::BinaryIO: File does not have a recognized Tpetra binary I/O header. "
143 << "Legacy Xpetra binary matrix files must be converted with "
144 << "Xpetra::Experimental::ConvertLegacyBinaryToBinary before reading with Tpetra::Experimental::BinaryIO.");
145 TEUCHOS_TEST_FOR_EXCEPTION(header.version != binaryIOFileVersion,
146 std::runtime_error,
147 "Tpetra::BinaryIO: Unsupported file version " << header.version << ".");
148 // Files are stored in the writer's native byte order; reading a file written
149 // with the opposite endianness is detected but not converted.
150 TEUCHOS_TEST_FOR_EXCEPTION(header.byteOrderMarker == binaryIOByteSwap64(binaryIOByteOrderMarker),
151 std::runtime_error,
152 "Tpetra::BinaryIO: File was written on a machine with the opposite byte order; "
153 "byte-order conversion is not supported.");
154 TEUCHOS_TEST_FOR_EXCEPTION(header.byteOrderMarker != binaryIOByteOrderMarker,
155 std::runtime_error,
156 "Tpetra::BinaryIO: Unsupported byte order marker.");
157 TEUCHOS_TEST_FOR_EXCEPTION(header.objectKind != binaryIOMapObject,
158 std::runtime_error,
159 "Tpetra::BinaryIO: File object kind " << header.objectKind
160 << " does not match expected kind " << binaryIOMapObject << ".");
161 TEUCHOS_TEST_FOR_EXCEPTION(header.localOrdinalSize != sizeof(LocalOrdinal) || header.localOrdinalFlags != binaryIOTypeFlags<LocalOrdinal>(),
162 std::runtime_error,
163 "Tpetra::BinaryIO: File local ordinal type does not match this map reader instantiation.");
164 TEUCHOS_TEST_FOR_EXCEPTION(header.globalOrdinalSize != sizeof(GlobalOrdinal) || header.globalOrdinalFlags != binaryIOTypeFlags<GlobalOrdinal>(),
165 std::runtime_error,
166 "Tpetra::BinaryIO: File global ordinal type does not match this map reader instantiation.");
167}

169inline BinaryIOMapSectionHeader binaryIOReadMapSectionHeader(const std::string& filename,
170 const unsigned long long mapSectionOffset,
171 const Teuchos::RCP<const Teuchos::Comm<int>>& comm) {
172 BinaryIOMapSectionHeader header;
173 std::memset(&header, 0, sizeof(BinaryIOMapSectionHeader));
174#ifdef HAVE_TPETRACORE_MPI
175 if (Tpetra::Details::teuchosCommIsAnMpiComm(*comm)) {
176 MPI_Comm rawComm = Tpetra::Details::extractMpiCommFromTeuchos(*comm);
177 MPI_File file;
178 const int err = MPI_File_open(rawComm, const_cast<char*>(filename.c_str()), MPI_MODE_RDONLY, MPI_INFO_NULL, &file);
179 TEUCHOS_TEST_FOR_EXCEPTION(err != MPI_SUCCESS, std::runtime_error,
180 "Tpetra::BinaryIO: MPI_File_open failed while reading map section header.");
181 MPI_Status status;
182 const int readErr = MPI_File_read_at_all(file, static_cast<MPI_Offset>(mapSectionOffset),
183 &header, sizeof(BinaryIOMapSectionHeader), MPI_BYTE, &status);
184 MPI_File_close(&file);
185 TEUCHOS_TEST_FOR_EXCEPTION(readErr != MPI_SUCCESS, std::runtime_error,
186 "Tpetra::BinaryIO: MPI_File_read_at_all failed while reading map section header.");
187 return header;
188 }
189#endif
190 if (comm->getRank() == 0) {
191 std::ifstream in(filename.c_str(), std::ios::binary);
192 TEUCHOS_TEST_FOR_EXCEPTION(!in.good(), std::runtime_error,
193 "Tpetra::BinaryIO: Failed to open file '" << filename << "' while reading map section header.");
194 in.seekg(static_cast<std::streamoff>(mapSectionOffset), std::ios::beg);
195 in.read(reinterpret_cast<char*>(&header), sizeof(BinaryIOMapSectionHeader));
196 TEUCHOS_TEST_FOR_EXCEPTION(!in.good(), std::runtime_error,
197 "Tpetra::BinaryIO: Failed to read map section header from file '" << filename << "'.");
198 }
199 Teuchos::broadcast(*comm, 0, static_cast<int>(sizeof(BinaryIOMapSectionHeader)), reinterpret_cast<char*>(&header));
200 return header;
201}

203template <class LocalOrdinal, class GlobalOrdinal, class Node>
204Teuchos::RCP<const Tpetra::Map<LocalOrdinal, GlobalOrdinal, Node>>
205binaryIOReadMapSection(const std::string& filename,
206 const unsigned long long mapSectionOffset,
207 const Teuchos::RCP<const Teuchos::Comm<int>>& comm) {
208 using map_type = Tpetra::Map<LocalOrdinal, GlobalOrdinal, Node>;
209 const BinaryIOMapSectionHeader sectionHeader = binaryIOReadMapSectionHeader(filename, mapSectionOffset, comm);
210 const unsigned long long globalCount = sectionHeader.numGlobalElements;
211
212 TEUCHOS_TEST_FOR_EXCEPTION(sectionHeader.numRanks != static_cast<unsigned long long>(comm->getSize()),
213 std::runtime_error,
214 "Tpetra::BinaryIO: Map section was written for " << sectionHeader.numRanks
215 << " ranks, but the read communicator has "
216 << comm->getSize() << " ranks.");
217
218 Kokkos::View<unsigned long long*, Kokkos::HostSpace> localCounts("Tpetra::BinaryIO::mapLocalCounts",
219 binaryIOCheckedSize(sectionHeader.numRanks, "map section rank count"));
220 binaryIOReadArrayFromRoot(filename,
221 mapSectionOffset + static_cast<unsigned long long>(sizeof(BinaryIOMapSectionHeader)),
222 localCounts.data(),
223 static_cast<unsigned long long>(localCounts.extent(0)),
224 comm);
225
226 const unsigned long long localCount = localCounts(comm->getRank());
227 unsigned long long globalOffset = 0;
228 for (int rank = 0; rank < comm->getRank(); ++rank) {
229 globalOffset += localCounts(rank);
230 }
231
232 Kokkos::View<GlobalOrdinal*, Kokkos::HostSpace> gids("Tpetra::BinaryIO::mapGids",
233 binaryIOCheckedSize(localCount, "map section local element count"));
234 binaryIOReadArrayCollective(filename,
235 mapSectionOffset + static_cast<unsigned long long>(sizeof(BinaryIOMapSectionHeader)) + sectionHeader.numRanks * sizeof(unsigned long long),
236 gids.data(),
237 localCount,
238 globalOffset,
239 comm);
240
241 using device_type = typename map_type::device_type;
242 Kokkos::View<GlobalOrdinal*, device_type> gidsDevice("Tpetra::BinaryIO::mapGidsDevice", gids.extent(0));
243 Kokkos::deep_copy(gidsDevice, gids);
244
245 return Teuchos::rcp(new map_type(static_cast<Tpetra::global_size_t>(globalCount),
246 gidsDevice,
247 static_cast<GlobalOrdinal>(sectionHeader.indexBase),
248 comm));
249}

253template <class LocalOrdinal, class GlobalOrdinal, class Node>
254Teuchos::RCP<const Tpetra::Map<LocalOrdinal, GlobalOrdinal, Node>>
255readBinaryMapFile(const std::string& filename,
256 const Teuchos::RCP<const Teuchos::Comm<int>>& comm) {
257 const Details::BinaryIOFileHeader header = Details::binaryIOReadHeaderFromFile(filename, comm);
258 Details::binaryIOValidateMapHeader<LocalOrdinal, GlobalOrdinal>(header);
259 return Details::binaryIOReadMapSection<LocalOrdinal, GlobalOrdinal, Node>(filename, header.rowMapOffset, comm);
260}
