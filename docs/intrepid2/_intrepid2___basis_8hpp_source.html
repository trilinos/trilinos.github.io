242 template<typename OrdinalTypeView3D,
243 typename OrdinalTypeView2D,
244 typename OrdinalTypeView1D>
245 void setOrdinalTagData( OrdinalTypeView3D &tagToOrdinal,
246 OrdinalTypeView2D &ordinalToTag,
247 const OrdinalTypeView1D tags,
248 const ordinal_type basisCard,
249 const ordinal_type tagSize,
250 const ordinal_type posScDim,
251 const ordinal_type posScOrd,
252 const ordinal_type posDfOrd ) {
253 // The tables are host views, filled here with plain loops rather than with Kokkos::deep_copy(): Basis_TensorBasis
254 // calls this lazily from const queries, which may come from inside an OpenMP parallel region, where dispatching a
255 // kernel on the OpenMP execution space (as deep_copy() of a nonzero value does) aborts.
256
257 // Create ordinalToTag; every entry is set below
258 ordinalToTag = OrdinalTypeView2D(Kokkos::view_alloc("ordinalToTag", Kokkos::WithoutInitializing), basisCard, tagSize);
259
260 // Copy tags
261 for (ordinal_type i=0;i<basisCard;++i)
262 for (ordinal_type j=0;j<tagSize;++j)
263 ordinalToTag(i, j) = tags(i*tagSize + j);
264
265 // Find out dimension of tagToOrdinal
266 auto maxScDim = 0; // first dimension of tagToOrdinal
267 for (ordinal_type i=0;i<basisCard;++i)
268 if (maxScDim < tags(i*tagSize + posScDim))
269 maxScDim = tags(i*tagSize + posScDim);
270 ++maxScDim;
271
272 auto maxScOrd = 0; // second dimension of tagToOrdinal
273 for (ordinal_type i=0;i<basisCard;++i)
274 if (maxScOrd < tags(i*tagSize + posScOrd))
275 maxScOrd = tags(i*tagSize + posScOrd);
276 ++maxScOrd;
277
278 auto maxDfOrd = 0; // third dimension of tagToOrdinal
279 for (ordinal_type i=0;i<basisCard;++i)
280 if (maxDfOrd < tags(i*tagSize + posDfOrd))
281 maxDfOrd = tags(i*tagSize + posDfOrd);
282 ++maxDfOrd;
283
284 // Create tagToOrdinal, and initialize with -1
285 tagToOrdinal = OrdinalTypeView3D(Kokkos::view_alloc("tagToOrdinal", Kokkos::WithoutInitializing), maxScDim, maxScOrd, maxDfOrd);
286 for (ordinal_type i=0;i<maxScDim;++i)
287 for (ordinal_type j=0;j<maxScOrd;++j)
288 for (ordinal_type k=0;k<maxDfOrd;++k)
289 tagToOrdinal(i, j, k) = -1;
290
291 // Overwrite elements of the array corresponding to tags with local DoF Id's, leave all other = -1
292 for (ordinal_type i=0;i<basisCard;++i)
293 tagToOrdinal(tags(i*tagSize), tags(i*tagSize+1), tags(i*tagSize+2)) = i;
294 }

369 virtual BasisValues<OutputValueType,DeviceType> allocateBasisValues( TensorPoints<PointValueType,DeviceType> points, const EOperator operatorType = OPERATOR_VALUE) const
370 {
371 const bool operatorIsDk = (operatorType >= OPERATOR_D1) && (operatorType <= OPERATOR_D10);
372 const bool operatorSupported = (operatorType == OPERATOR_VALUE) || (operatorType == OPERATOR_GRAD) || (operatorType == OPERATOR_CURL) || (operatorType == OPERATOR_DIV) || operatorIsDk;
373 INTREPID2_TEST_FOR_EXCEPTION(!operatorSupported, std::invalid_argument, "operator is not supported by allocateBasisValues");
374
375 const int numPoints = points.extent_int(0);
376
377 using Scalar = OutputValueType;
378
379 auto dataView = allocateOutputView(numPoints, operatorType);
380 Data<Scalar,DeviceType> data(dataView);
381
382 bool useVectorData = (rank(dataView) == 3);
383
384 if (useVectorData)
385 {
386 VectorData<Scalar,DeviceType> vectorData(data);
387 return BasisValues<Scalar,DeviceType>(vectorData);
388 }
389 else
390 {
391 TensorData<Scalar,DeviceType> tensorData(data);
392 return BasisValues<Scalar,DeviceType>(tensorData);
393 }
394 }

503 const TensorPoints<PointValueType,DeviceType> inputPoints,
504 const EOperator operatorType = OPERATOR_VALUE ) const {
505 // note the extra allocation/copy here (this is one reason, among several, to override this method):

510 if (outputValues.numTensorDataFamilies() > 0)

514 }
515 else if (outputValues.vectorData().isValid())
516 {
517 INTREPID2_TEST_FOR_EXCEPTION(outputValues.vectorData().numComponents() != 1, std::invalid_argument, "default implementation of getValues() only supports outputValues with trivial tensor-product structure");
518 INTREPID2_TEST_FOR_EXCEPTION(outputValues.vectorData().getComponent(0).numTensorComponents() != 1, std::invalid_argument, "default implementation of getValues() only supports outputValues with trivial tensor-product structure");
519 outputData = outputValues.vectorData().getComponent(0).getTensorComponent(0);
520 }
521
522 this->getValues(outputData.getUnderlyingView(), rawExpandedPoints, operatorType);
523 }

591 int degreeEntryLength = fieldOrdinalPolynomialDegree_.extent_int(1);
592 int requestedDegreeLength = degrees.extent_int(0);

601 }
602 if (matches) fieldOrdinalsVector.push_back(basisOrdinal);
603 }
604 OrdinalTypeArray1DHost fieldOrdinals("fieldOrdinalsForDegree",fieldOrdinalsVector.size());
605 for (unsigned i=0; i<fieldOrdinalsVector.size(); i++)
606 {
607 fieldOrdinals(i) = fieldOrdinalsVector[i];
608 }
609 return fieldOrdinals;

879 virtual ordinal_type
880 getDofCount( const ordinal_type subcDim,
881 const ordinal_type subcOrd ) const {
882 this->requireTags();
883 if ( subcDim >= 0 && subcDim < static_cast<ordinal_type>(tagToOrdinal_.extent(0)) &&
884 subcOrd >= 0 && subcOrd < static_cast<ordinal_type>(tagToOrdinal_.extent(1)) )
885 {
886 int firstDofOrdinal = tagToOrdinal_(subcDim, subcOrd, 0); // will be -1 if no dofs for subcell
887 if (firstDofOrdinal == -1) return static_cast<ordinal_type>(0);
888 // otherwise, lookup its tag and return the dof count stored there
889 return static_cast<ordinal_type>(this->getDofTag(firstDofOrdinal)[3]);
890 }
891 else
892 {
893 // subcDim and/or subcOrd out of bounds -> no dofs associated with subcell
894 return static_cast<ordinal_type>(0);
895 }
896 }

906 virtual ordinal_type
907 getDofOrdinal( const ordinal_type subcDim,
908 const ordinal_type subcOrd,
909 const ordinal_type subcDofOrd ) const {
910 this->requireTags();
911 // this should be abort and able to be called as a device function
912#ifdef HAVE_INTREPID2_DEBUG
913 INTREPID2_TEST_FOR_EXCEPTION( subcDim < 0 || subcDim >= static_cast<ordinal_type>(tagToOrdinal_.extent(0)), std::out_of_range,
914 ">>> ERROR (Basis::getDofOrdinal): subcDim is out of range");
915 INTREPID2_TEST_FOR_EXCEPTION( subcOrd < 0 || subcOrd >= static_cast<ordinal_type>(tagToOrdinal_.extent(1)), std::out_of_range,
916 ">>> ERROR (Basis::getDofOrdinal): subcOrd is out of range");
917 INTREPID2_TEST_FOR_EXCEPTION( subcDofOrd < 0 || subcDofOrd >= static_cast<ordinal_type>(tagToOrdinal_.extent(2)), std::out_of_range,
918 ">>> ERROR (Basis::getDofOrdinal): subcDofOrd is out of range");
919#endif
920 ordinal_type r_val = -1;
921 if ( subcDim < static_cast<ordinal_type>(tagToOrdinal_.extent(0)) &&
922 subcOrd < static_cast<ordinal_type>(tagToOrdinal_.extent(1)) &&
923 subcDofOrd < static_cast<ordinal_type>(tagToOrdinal_.extent(2)) )
924 r_val = tagToOrdinal_(subcDim, subcOrd, subcDofOrd);
925#ifdef HAVE_INTREPID2_DEBUG
926 INTREPID2_TEST_FOR_EXCEPTION( r_val == -1, std::runtime_error,
927 ">>> ERROR (Basis::getDofOrdinal): Invalid DoF tag is found.");
928#endif
929 return r_val;
930 }
