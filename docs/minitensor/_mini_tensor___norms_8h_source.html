131//
132// R^N tensor Frobenius norm
133// \return \f$ \sqrt{A:A} \f$
134//
135template<typename T, Index N>
136KOKKOS_INLINE_FUNCTION
137T
138norm(Tensor<T, N> const & A)
139{
140 Index const
141 dimension = A.get_dimension();
142
143 T
144 s = 0.0;
145
146 switch (dimension) {
147
148 default:
149 s = norm_f_square(A);
150 break;
151
152 case 3:
153 s+= A(0,0)*A(0,0) + A(0,1)*A(0,1) + A(0,2)*A(0,2);
154 s+= A(1,0)*A(1,0) + A(1,1)*A(1,1) + A(1,2)*A(1,2);
155 s+= A(2,0)*A(2,0) + A(2,1)*A(2,1) + A(2,2)*A(2,2);
156 break;
157
158 case 2:
159 s+= A(0,0)*A(0,0) + A(0,1)*A(0,1);
160 s+= A(1,0)*A(1,0) + A(1,1)*A(1,1);
161 break;
162
163 case 1:
164 s+= A(0,0)*A(0,0);
165 break;
166
167 }
168
169 if (s > 0.0) return std::sqrt(s);
170 return 0.0;
171
172}

174//
175// R^N tensor 1-norm
176// \return \f$ \max_{j \in {0,\cdots,N}}\Sigma_{i=0}^N |A_{ij}| \f$
177//
178template<typename T, Index N>
179KOKKOS_INLINE_FUNCTION
180T
181norm_1(Tensor<T, N> const & A)
182{
183 Index const
184 dimension = A.get_dimension();
185
186 Vector<T, N>
187 v(dimension);
188
189 T
190 s = 0.0;
191
192 switch (dimension) {
193
194 default:
195
196 for (Index i = 0; i < dimension; ++i) {
197 T t = 0.0;
198 for (Index j = 0; j < dimension; ++j) {
199 t += minitensor::abs(A(j, i));
200 }
201 v(i) = t;
202 }
203
204 for (Index i = 0; i < dimension; ++i) {
205 s = max(s, v(i));
206 }
207 break;
208
209 case 3:
210 v(0) = minitensor::abs(A(0, 0)) + minitensor::abs(A(1, 0)) + minitensor::abs(A(2, 0));
211 v(1) = minitensor::abs(A(0, 1)) + minitensor::abs(A(1, 1)) + minitensor::abs(A(2, 1));
212 v(2) = minitensor::abs(A(0, 2)) + minitensor::abs(A(1, 2)) + minitensor::abs(A(2, 2));
213
214 s = max(max(v(0), v(1)), v(2));
215 break;
216
217 case 2:
218 v(0) = minitensor::abs(A(0, 0)) + minitensor::abs(A(1, 0));
219 v(1) = minitensor::abs(A(0, 1)) + minitensor::abs(A(1, 1));
220
221 s = max(v(0), v(1));
222 break;
223
224 case 1:
225 s = minitensor::abs(A(0, 0));
226 break;
227 }
228
229 return s;
230}

289//
290// R^N determinant
291// Laplace expansion. Warning: no pivoting.
292// Casual use only. Use Teuchos LAPACK interface for
293// more efficient and robust techniques.
294// \param A tensor
295// \return \f$ \det A \f$
296//
297template<typename T, Index N>
298KOKKOS_INLINE_FUNCTION
299T
300det(Tensor<T, N> const & A)
301{
302 Index const
303 dimension = A.get_dimension();
304
305 T
306 s = 0.0;
307
308 switch (dimension) {
309
310 default:
311 {
312 int sign = 1;
313 for (Index i = 0; i < dimension; ++i) {
314 const T d = det(subtensor(A, i, 1));
315 s += sign * d * A(i, 1);
316 sign *= -1;
317 }
318 }
319 break;
320
321 case 3:
322 s = -A(0,2)*A(1,1)*A(2,0) + A(0,1)*A(1,2)*A(2,0) +
323 A(0,2)*A(1,0)*A(2,1) - A(0,0)*A(1,2)*A(2,1) -
324 A(0,1)*A(1,0)*A(2,2) + A(0,0)*A(1,1)*A(2,2);
325 break;
326
327 case 2:
328 s = A(0,0) * A(1,1) - A(1,0) * A(0,1);
329 break;
330
331 case 1:
332 s = A(0,0);
333 break;
334
335 }
336
337 return s;
338}

348trace(Tensor<T, N> const & A)
349{

354

356

395// \param A tensor
396// \return \f$ II_A \f$
397//
398template<typename T, Index N>
399KOKKOS_INLINE_FUNCTION
400T
401I2(Tensor<T, N> const & A)
402{
403 Index const
404 dimension = A.get_dimension();
405
406 T
407 s = 0.0;
408
409 T const
410 trA = trace(A);
411
412 switch (dimension) {
413
414 default:
415#ifdef KOKKOS_ENABLE_CUDA
416 Kokkos::abort("I2 for N > 3 not implemented.");
417 return T();
418#else
419 std::cerr << "I2 for N > 3 not implemented." << std::endl;
420 exit(1);
421#endif
422 break;
423
424 case 3:
425 s = 0.5 * (trA*trA - A(0,0)*A(0,0) - A(1,1)*A(1,1) - A(2,2)*A(2,2)) -
426 A(0,1)*A(1,0) - A(0,2)*A(2,0) - A(1,2)*A(2,1);
427 break;
428
429 case 2:
430 s = - det(A);
431 break;
432
433 case 1:
434 s = 0.0;
435 break;
436
437 }
438

487//
488// R^N Subtensor
489// \param A tensor
490// \param i index
491// \param j index
492// \return Subtensor with i-row and j-col deleted.
493//
494template<typename T, Index N>
495KOKKOS_INLINE_FUNCTION
496Tensor<T, N>
497subtensor(Tensor<T, N> const & A, Index const i, Index const j)
498{
499 Index const
500 dimension = A.get_dimension();
501
502 assert(i < dimension);
503 assert(j < dimension);
504
505 Tensor<T, N>
506 B(dimension - 1);
507
508 Index p = 0;
509 for (Index m = 0; m < dimension; ++m) {
510 if (m == i) continue;
511 Index q = 0;
512 for (Index n = 0; n < dimension; ++n) {
513 if (n == j) continue;
514 B(p, q) = A(m, n);
515 ++q;
516 }
517 ++p;
518 }
519
520 return B;
521}

523//
524// R^N off-diagonal norm. Useful for SVD and other algorithms
525// that rely on Jacobi-type procedures.
526// \param A
527// \return \f$ \sqrt(\sum_i \sum_{j, j\neq i} a_{ij}^2) \f$
528//
529template<typename T, Index N>
530KOKKOS_INLINE_FUNCTION
531T
532norm_off_diagonal(Tensor<T, N> const & A)
533{
534 Index const
535 dimension = A.get_dimension();
536
537 T
538 s = 0.0;
539
540 switch (dimension) {
541
542 default:
543 for (Index i = 0; i < dimension; ++i) {
544 for (Index j = 0; j < dimension; ++j) {
545 if (i != j) s += A(i,j)*A(i,j);
546 }
547 }
548 break;
549
550 case 3:
551 s = A(0,1)*A(0,1) + A(0,2)*A(0,2) + A(1,2)*A(1,2) +
552 A(1,0)*A(1,0) + A(2,0)*A(2,0) + A(2,1)*A(2,1);
553 break;
554
555 case 2:
556 s = A(0,1)*A(0,1) + A(1,0)*A(1,0);
557 break;
558
559 case 1:
560 s = 0.0;
561 break;
562
563 }
564 return std::sqrt(s);
565}

567//
568// R^N arg max abs. Useful for inverse and other algorithms
569// that rely on Jacobi-type procedures.
570// \param A
571// \return \f$ (p,q) = arg max_{i,j} |a_{ij}| \f$
572//
573template <typename T, Index N>
574std::pair<Index, Index> arg_max_abs(Tensor<T, N> const &A) {
575
576 Index p = 0;
577 Index q = 0;
578
579 T
580 s = std::abs(A(p,q));
581
582 Index const
583 dimension = A.get_dimension();
584
585 for (Index i = 0; i < dimension; ++i) {
586 for (Index j = 0; j < dimension; ++j) {
587 if (std::abs(A(i,j)) > s) {
588 p = i;
589 q = j;
590 s = std::abs(A(i,j));
591 }
592 }
593 }
594
595 return std::make_pair(p,q);
596}
