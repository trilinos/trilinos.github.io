93 KOKKOS_INLINE_FUNCTION
94 void operator()(const typename Kokkos::TeamPolicy<execution_space>::member_type& thread, size_t& nnz) const {
95 auto agg = thread.league_rank();
96
97 const auto aggOffset = aggRows(agg);
98 // size of aggregate: number of DOFs in aggregate
99 const auto aggSize = aggRows(agg + 1) - aggOffset;
100
101 const impl_SC one = impl_ATS::one();
102 const impl_SC zero = impl_ATS::zero();
103
104 const int m = aggSize;
105 const int n = fineNS.extent(1);
106
107 // calculate row offset for coarse nullspace
108 Xpetra::global_size_t offset = agg * n;
109
110 if (doQRStep) {
111 // A is m x n
112 // Q is m x m
113 // R is m x n
114
115 // A (initially) gets overwritten with R in QR
116 shared_matrix r(thread.team_scratch(scratchLevel), m, n);
117 // Q
118 shared_matrix q(thread.team_scratch(scratchLevel), m, m);
119
120 // Extract the piece of the nullspace corresponding to the aggregate
121 for (int j = 0; j < n; j++)
122 for (int k = 0; k < m; k++)
123 r(k, j) = fineNS(agg2RowMapLO(aggOffset + k), j);
124
125 if (m >= n) {
126 // tau has size n
127 shared_vector tau(thread.team_scratch(scratchLevel), n);
128
129 // work has size m
130 shared_vector work(thread.team_scratch(scratchLevel), m);
131
132 // Calculate QR. After this call R will be in the upper triangular part of r.
133 // Q is in encoded in the strictly lower triangular part of r and tau. We will
134 // need to explicitly form Q as a matrix below.
135 KokkosBatched::SerialQR<KokkosBlas::Algo::QR::Unblocked>::invoke(r, tau, work);
136
137 // Initialize Q to an identity matrix
138 KokkosBatched::SerialSetIdentity::invoke(q);
139
140 // Form Q as a matrix but multiplying against the identity
141 KokkosBatched::SerialApplyQ<KokkosBatched::Side::Left, KokkosBlas::Trans::NoTranspose, KokkosBlas::Algo::ApplyQ::Unblocked>::invoke(r, tau, q, work);
142
143 // Build coarse nullspace using the upper triangular part of R
144 for (int j = 0; j < n; j++) {
145 for (int k = 0; k < n; k++)
146 coarseNS(offset + k, j) = (k <= j) ? r(k, j) : zero;
147 }
148
149 // Process each row in the local Q factor and fill helper arrays to assemble P
150 for (int j = 0; j < m; j++) {
151 LO localRow = agg2RowMapLO(aggRows(agg) + j);
152 size_t rowStart = rowsAux(localRow);
153 size_t lnnz = 0;
154 for (int k = 0; k < n; k++) {
155 // skip zeros
156 if (q(j, k) != zero) {
157 colsAux(rowStart + lnnz) = offset + k;
158 valsAux(rowStart + lnnz) = q(j, k);
159 lnnz++;
160 }
161 }
162 rows(localRow + 1) = lnnz;
163 nnz += lnnz;
164 }
165
166 } else {
167 // Special handling for m < n (i.e. single node aggregates in structural mechanics)
168
169 // The local QR decomposition is not possible in the "overconstrained"
170 // case (i.e. number of columns in qr > number of rowsAux), which
171 // corresponds to #DOFs in Aggregate < n. For usual problems this
172 // is only possible for single node aggregates in structural mechanics.
173 // (Similar problems may arise in discontinuous Galerkin problems...)
174 // We bypass the QR decomposition and use an identity block in the
175 // tentative prolongator for the single node aggregate and transfer the
176 // corresponding fine level null space information 1-to-1 to the coarse
177 // level null space part.
178
179 // NOTE: The resulting tentative prolongation operator has
180 // (m*DofsPerNode-n) zero columns leading to a singular
181 // coarse level operator A. To deal with that one has the following
182 // options:
183 // - Use the "RepairMainDiagonal" flag in the RAPFactory (default:
184 // false) to add some identity block to the diagonal of the zero rowsAux
185 // in the coarse level operator A, such that standard level smoothers
186 // can be used again.
187 // - Use special (projection-based) level smoothers, which can deal
188 // with singular matrices (very application specific)
189 // - Adapt the code below to avoid zero columns. However, we do not
190 // support a variable number of DOFs per node in MueLu/Xpetra which
191 // makes the implementation really hard.
192 //
193 // FIXME: do we need to check for singularity here somehow? Zero
194 // columns would be easy but linear dependency would require proper QR.
195
196 // R = extended (by adding identity rowsAux) qr
197 for (int j = 0; j < n; j++)
198 for (int k = 0; k < n; k++)
199 if (k < m)
200 coarseNS(offset + k, j) = r(k, j);
201 else
202 coarseNS(offset + k, j) = (k == j ? one : zero);
203
204 // Q = I (rectangular)
205 // no need to actually fill q
206
207 // Process each row in the local Q factor and fill helper arrays to assemble P
208 for (int j = 0; j < m; j++) {
209 LO localRow = agg2RowMapLO(aggRows(agg) + j);
210 size_t rowStart = rowsAux(localRow);
211 colsAux(rowStart) = offset + j;
212 valsAux(rowStart) = one;
213 rows(localRow + 1) = 1;
214 nnz += 1;
215 }
216 }
217 } else {
219 // "no-QR" option //
221 // Local Q factor is just the fine nullspace support over the current aggregate.
222 // Local R factor is the identity.
223 // TODO I have not implemented any special handling for aggregates that are too
224 // TODO small to locally support the nullspace, as is done in the standard QR
225 // TODO case above.
226
227 for (int j = 0; j < m; j++) {
228 LO localRow = agg2RowMapLO(aggRows(agg) + j);
229 size_t rowStart = rowsAux(localRow);
230 size_t lnnz = 0;
231 for (int k = 0; k < n; k++) {
232 const impl_SC qr_jk = fineNS(localRow, k);
233 // skip zeros
234 if (qr_jk != zero) {
235 colsAux(rowStart + lnnz) = offset + k;
236 valsAux(rowStart + lnnz) = qr_jk;
237 lnnz++;
238 }
239 }
240 rows(localRow + 1) = lnnz;
241 nnz += lnnz;
242 }
243
244 for (int j = 0; j < n; j++)
245 coarseNS(offset + j, j) = one;
246 }
247 }
