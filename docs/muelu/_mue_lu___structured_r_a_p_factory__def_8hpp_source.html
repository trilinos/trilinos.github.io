44KOKKOS_INLINE_FUNCTION void getLocalNodeIndices(const GO localNode,
45 const LocalNodes& localNodes,
46 GO& x, GO& y, GO& z) {
47 x = localNode % localNodes[0];
48 y = (localNode / localNodes[0]) % localNodes[1];
49 z = localNode / (localNodes[0] * localNodes[1]);

52template <class GO, class Offset, class LocalNodes, class ProcGrid, class RankData>
53KOKKOS_INLINE_FUNCTION bool resolveNeighbor(const GO x, const GO y, const GO z,
54 const Offset& offset,
55 const LocalNodes& localNodes,
56 const ProcGrid& procGrid,
57 const RankData& rankData,
58 const int myProcX,
59 const int myProcY,
60 const int myProcZ,
61 int& neighborRank,
62 GO& neighborNode) {
63 int neighborProcX = myProcX;
64 int neighborProcY = myProcY;
65 int neighborProcZ = myProcZ;
66 GO neighborX = x + static_cast<GO>(offset.x);
67 GO neighborY = y + static_cast<GO>(offset.y);
68 GO neighborZ = z + static_cast<GO>(offset.z);
69
70 if (neighborX < 0)
71 --neighborProcX;
72 else if (neighborX >= localNodes[0])
73 ++neighborProcX;
74 if (neighborY < 0)
75 --neighborProcY;
76 else if (neighborY >= localNodes[1])
77 ++neighborProcY;
78 if (neighborZ < 0)
79 --neighborProcZ;
80 else if (neighborZ >= localNodes[2])
81 ++neighborProcZ;
82
83 if (neighborProcX < 0 || neighborProcX >= procGrid[0] ||
84 neighborProcY < 0 || neighborProcY >= procGrid[1] ||
85 neighborProcZ < 0 || neighborProcZ >= procGrid[2])
86 return false;
87
88 neighborRank = neighborProcZ * procGrid[0] * procGrid[1] + neighborProcY * procGrid[0] + neighborProcX;
89 const GO neighborLocalNx = rankData(4 * neighborRank + 1);
90 const GO neighborLocalNy = rankData(4 * neighborRank + 2);
91 const GO neighborLocalNz = rankData(4 * neighborRank + 3);
92 if (neighborX < 0)
93 neighborX = neighborLocalNx - 1;
94 else if (neighborX >= localNodes[0])
95 neighborX = 0;
96 if (neighborY < 0)
97 neighborY = neighborLocalNy - 1;
98 else if (neighborY >= localNodes[1])
99 neighborY = 0;
100 if (neighborZ < 0)
101 neighborZ = neighborLocalNz - 1;
102 else if (neighborZ >= localNodes[2])
103 neighborZ = 0;
104
105 neighborNode = rankData(4 * neighborRank) +
106 neighborZ * neighborLocalNx * neighborLocalNy +
107 neighborY * neighborLocalNx + neighborX;
108 return true;
109}

191template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
192void StructuredRAPFactory<Scalar, LocalOrdinal, GlobalOrdinal, Node>::DeclareInput(Level& fineLevel, Level& coarseLevel) const {
193 const Teuchos::ParameterList& pL = GetParameterList();
194 TEUCHOS_TEST_FOR_EXCEPTION(
195 !pL.get<bool>("transpose: use implicit"), Exceptions::RuntimeError,
196 "StructuredRAPFactory requires \"transpose: use implicit\" = true because "
197 "the prebuilt coarse graph assumes the Galerkin product P^T A P.");
198
199 const bool useRAPDelegate =
200 !pL.get<bool>("rap: prebuild coarse graph");
201
202 if (useRAPDelegate) {
203 ConfigureRAPFactoryDelegate();
204 coarseLevel.DeclareInput("A", rapFactoryDelegate_.get(), this);
205 coarseLevel.DeclareInput("RAP reuse data", rapFactoryDelegate_.get(), this);
206 hasDeclaredInput_ = true;
207 return;
208 }
209
210 Input(fineLevel, "A");
211 Input(coarseLevel, "P");
212
213 const RCP<const FactoryBase> coarseDimensionsFactory =
214 GetStructuredMetadataFactory("lCoarseNodesPerDim");
215 const RCP<const FactoryBase> interpolationOrderFactory =
216 GetStructuredMetadataFactory("structuredInterpolationOrder");
217 fineLevel.DeclareInput("lCoarseNodesPerDim", coarseDimensionsFactory.get(), this);
218 fineLevel.DeclareInput("structuredInterpolationOrder", interpolationOrderFactory.get(), this);
219 if (fineLevel.GetLevelID() == 0) {
220 TEUCHOS_TEST_FOR_EXCEPTION(!fineLevel.IsAvailable("numDimensions", NoFactory::get()),
221 Exceptions::RuntimeError,
222 "StructuredRAPFactory: numDimensions was not provided on level 0.");
223 TEUCHOS_TEST_FOR_EXCEPTION(!fineLevel.IsAvailable("lNodesPerDim", NoFactory::get()),
224 Exceptions::RuntimeError,
225 "StructuredRAPFactory: lNodesPerDim was not provided on level 0.");
226 fineLevel.DeclareInput("numDimensions", NoFactory::get(), this);
227 fineLevel.DeclareInput("lNodesPerDim", NoFactory::get(), this);
228 } else {
229 Input(fineLevel, "numDimensions");
230 Input(fineLevel, "lNodesPerDim");
231 }
232
233 // call DeclareInput of all user-given transfer factories
234 for (std::vector<RCP<const FactoryBase>>::const_iterator it = transferFacts_.begin(); it != transferFacts_.end(); ++it)
235 (*it)->CallDeclareInput(coarseLevel);
236
237 hasDeclaredInput_ = true;
238}

240// Infer the fine-grid stencil of A from one representative interior node
241template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
242typename StructuredRAPFactory<Scalar, LocalOrdinal, GlobalOrdinal, Node>::FineStencilSpec
243StructuredRAPFactory<Scalar, LocalOrdinal, GlobalOrdinal, Node>::DetectFineStencil(
244 const Matrix& A, const int numDimensions,
245 const Teuchos::Array<LocalOrdinal>& lFineNodesPerDim) const {
246 const std::string prefix = "StructuredRAPFactory::DetectFineStencil: ";
247
248 TEUCHOS_TEST_FOR_EXCEPTION(numDimensions < 1 || numDimensions > 3, Exceptions::RuntimeError,
249 prefix << "the number of dimensions must be between one and three.");
250 TEUCHOS_TEST_FOR_EXCEPTION(lFineNodesPerDim.size() < numDimensions, Exceptions::RuntimeError,
251 prefix << "insufficient fine-grid dimensions were supplied.");
252
253 const auto rowMap = A.getRowMap();
254 const auto colMap = A.getColMap();
255 const auto comm = rowMap->getComm();
256 const auto graph = A.getCrsGraph();
257 TEUCHOS_TEST_FOR_EXCEPTION(graph.is_null(), Exceptions::RuntimeError,
258 prefix << "A does not expose a CRS graph.");
259
260 const LO dofsPerNode = Teuchos::as<LO>(A.GetFixedBlockSize());
261 TEUCHOS_TEST_FOR_EXCEPTION(dofsPerNode <= 0, Exceptions::RuntimeError,
262 prefix << "A must have a positive fixed block size.");
263
264 Kokkos::Array<LO, 3> fineNodes{{LO(1), LO(1), LO(1)}};
265 Kokkos::Array<LO, 3> interior{{LO(0), LO(0), LO(0)}};
266 size_t numFineNodes = 1;
267 bool hasLocalInterior = true;
268 // Select interior node in the local subdomain as a representative node for detecting the fine-grid stencil
269 for (int dim = 0; dim < numDimensions; ++dim) {
270 fineNodes[dim] = lFineNodesPerDim[dim];
271 hasLocalInterior = hasLocalInterior && fineNodes[dim] >= LO(3);
272 if (fineNodes[dim] > LO(0)) {
273 interior[dim] = fineNodes[dim] / LO(2);
274 numFineNodes *= Teuchos::as<size_t>(fineNodes[dim]);
275 } else {
276 numFineNodes = 0;
277 }
278 }
279
280 const bool localDimensionsMatch =
281 rowMap->getLocalNumElements() ==
282 numFineNodes * Teuchos::as<size_t>(dofsPerNode);
283 int invalidLocalDimensions = localDimensionsMatch ? 0 : 1;
284 int invalidDimensions = 0;
285 Teuchos::reduceAll(*comm, Teuchos::REDUCE_MAX, 1,
286 &invalidLocalDimensions, &invalidDimensions);
287 TEUCHOS_TEST_FOR_EXCEPTION(
288 invalidDimensions != 0, Exceptions::RuntimeError,
289 prefix << "the structured local dimensions and fixed block size do not match A on every rank.");
290
291 const LO interiorNode = interior[2] * fineNodes[0] * fineNodes[1] +
292 interior[1] * fineNodes[0] + interior[0];
293
294 const auto localA = A.getLocalMatrixDevice();
295 const LO invalidLocalOrdinal = Teuchos::OrdinalTraits<LO>::invalid();
296 std::vector<std::vector<LO>> ownedColumns(
297 Teuchos::as<size_t>(dofsPerNode));
298 bool hasCompleteLocalStencil = hasLocalInterior;
299 if (hasLocalInterior) {
300 const size_t firstRow = Teuchos::as<size_t>(interiorNode * dofsPerNode);
301 const auto detectedRowMap = Kokkos::subview(
302 localA.graph.row_map,
303 std::make_pair(firstRow,
304 firstRow + Teuchos::as<size_t>(dofsPerNode) + 1));
305 const auto detectedRowMapHost = Kokkos::create_mirror_view_and_copy(
306 Kokkos::HostSpace(), detectedRowMap);
307 const size_t firstEntry = detectedRowMapHost(0);
308 const size_t lastEntry =
309 detectedRowMapHost(Teuchos::as<size_t>(dofsPerNode));
310 const auto detectedEntries = Kokkos::subview(
311 localA.graph.entries, std::make_pair(firstEntry, lastEntry));
312 using device_type = typename Node::device_type;
313 using execution_space = typename device_type::execution_space;
314 Kokkos::View<LO*, device_type> detectedOwnedColumns(
315 Kokkos::ViewAllocateWithoutInitializing(
316 "StructuredRAP: detected owned columns"),
317 lastEntry - firstEntry);
318 const auto rowLocalMap = rowMap->getLocalMap();
319 const auto colLocalMap = colMap->getLocalMap();
320 Kokkos::parallel_for(
321 "StructuredRAP: map detected stencil columns",
322 Kokkos::RangePolicy<execution_space>(0, lastEntry - firstEntry),
323 KOKKOS_LAMBDA(const size_t entry) {
324 const GO globalColumn =
325 colLocalMap.getGlobalElement(detectedEntries(entry));
326 detectedOwnedColumns(entry) =
327 rowLocalMap.getLocalElement(globalColumn);
328 });
329 const auto detectedOwnedColumnsHost =
330 Kokkos::create_mirror_view_and_copy(
331 Kokkos::HostSpace(), detectedOwnedColumns);
332
333 for (LO rowDof = 0; rowDof < dofsPerNode; ++rowDof) {
334 const size_t rowBegin = detectedRowMapHost(Teuchos::as<size_t>(rowDof));
335 const size_t rowEnd =
336 detectedRowMapHost(Teuchos::as<size_t>(rowDof) + 1);
337 if (rowBegin == rowEnd)
338 hasCompleteLocalStencil = false;
339 auto& rowColumns = ownedColumns[Teuchos::as<size_t>(rowDof)];
340 rowColumns.reserve(rowEnd - rowBegin);
341 for (size_t entry = rowBegin; entry < rowEnd; ++entry) {
342 const LO ownedColumn =
343 detectedOwnedColumnsHost(entry - firstEntry);
344 rowColumns.push_back(ownedColumn);
345 if (ownedColumn == invalidLocalOrdinal)
346 hasCompleteLocalStencil = false;
347 }
348 }
349 }
350
351 const int myRank = comm->getRank();
352 const int numRanks = comm->getSize();
353 const int candidateRank = hasCompleteLocalStencil ? myRank : numRanks;
354 int detectorRank = numRanks;
355 Teuchos::reduceAll(*comm, Teuchos::REDUCE_MIN, 1, &candidateRank, &detectorRank);
356 TEUCHOS_TEST_FOR_EXCEPTION(
357 detectorRank == numRanks, Exceptions::RuntimeError,
358 prefix << "no MPI rank owns a local interior node whose complete stencil is locally owned; "
359 << "increase the local structured subdomain size.");
360
361 FineStencilSpec stencil;
362 stencil.numDimensions = numDimensions;
363 stencil.dofsPerNode = dofsPerNode;
364
365 // Only the elected rank performs detection
366 if (myRank == detectorRank) {
367 for (LO rowDof = 0; rowDof < dofsPerNode; ++rowDof) {
368 const auto& rowColumns = ownedColumns[Teuchos::as<size_t>(rowDof)];
369 const size_t rowLength = rowColumns.size();
370 for (size_t entry = 0; entry < rowLength; ++entry) {
371 const LO ownedColumn = rowColumns[entry];
372 const LO columnNode = ownedColumn / dofsPerNode;
373 const LO columnDof = ownedColumn % dofsPerNode;
374 LO columnX, columnY, columnZ;
375 StructuredRAPFactoryDetails::getLocalNodeIndices(
376 columnNode, fineNodes, columnX, columnY, columnZ);
377 const StencilOffset offset{
378 Teuchos::as<int>(columnX - interior[0]),
379 Teuchos::as<int>(columnY - interior[1]),
380 Teuchos::as<int>(columnZ - interior[2])};
381
382 bool knownOffset = false;
383 for (const StencilOffset& known : stencil.stencilOffsets) {
384 if (known.x == offset.x && known.y == offset.y && known.z == offset.z) {
385 knownOffset = true;
386 break;
387 }
388 }
389 if (!knownOffset)
390 stencil.stencilOffsets.push_back(offset);
391
392 stencil.entries.push_back(FineStencilEntry{
393 offset, rowDof, columnDof});
394 }
395 }
396 }
397
398 int counts[2] = {Teuchos::as<int>(stencil.stencilOffsets.size()),
399 Teuchos::as<int>(stencil.entries.size())};
400 Teuchos::broadcast<int, int>(*comm, detectorRank, 2, counts);
401
402 Teuchos::Array<int> offsetData(3 * counts[0]);
403 Teuchos::Array<int> entryOffsetData(3 * counts[1]);
404 Teuchos::Array<LO> entryData(2 * counts[1]);
405 if (myRank == detectorRank) {
406 for (int offset = 0; offset < counts[0]; ++offset) {
407 offsetData[3 * offset] = stencil.stencilOffsets[offset].x;
408 offsetData[3 * offset + 1] = stencil.stencilOffsets[offset].y;
409 offsetData[3 * offset + 2] = stencil.stencilOffsets[offset].z;
410 }
411 for (int entry = 0; entry < counts[1]; ++entry) {
412 entryOffsetData[3 * entry] = stencil.entries[entry].offset.x;
413 entryOffsetData[3 * entry + 1] = stencil.entries[entry].offset.y;
414 entryOffsetData[3 * entry + 2] = stencil.entries[entry].offset.z;
415 entryData[2 * entry] = stencil.entries[entry].rowDof;
416 entryData[2 * entry + 1] = stencil.entries[entry].columnDof;
417 }
418 }
419
420 if (counts[0] > 0)
421 Teuchos::broadcast<int, int>(*comm, detectorRank, 3 * counts[0],
422 offsetData.getRawPtr());
423 if (counts[1] > 0) {
424 Teuchos::broadcast<int, int>(*comm, detectorRank, 3 * counts[1],
425 entryOffsetData.getRawPtr());
426 Teuchos::broadcast<int, LO>(*comm, detectorRank, 2 * counts[1],
427 entryData.getRawPtr());
428 }
429
430 if (myRank != detectorRank) {
431 stencil.stencilOffsets.reserve(counts[0]);
432 for (int offset = 0; offset < counts[0]; ++offset)
433 stencil.stencilOffsets.push_back(StencilOffset{
434 offsetData[3 * offset], offsetData[3 * offset + 1],
435 offsetData[3 * offset + 2]});
436
437 stencil.entries.reserve(counts[1]);
438 for (int entry = 0; entry < counts[1]; ++entry)
439 stencil.entries.push_back(FineStencilEntry{
440 StencilOffset{entryOffsetData[3 * entry],
441 entryOffsetData[3 * entry + 1],
442 entryOffsetData[3 * entry + 2]},
443 entryData[2 * entry], entryData[2 * entry + 1]});
444 }
445
446 TEUCHOS_TEST_FOR_EXCEPTION(stencil.stencilOffsets.empty(), Exceptions::RuntimeError,
447 prefix << "the representative interior rows are empty.");
448 return stencil;
449}

451template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
452typename StructuredRAPFactory<Scalar, LocalOrdinal, GlobalOrdinal, Node>::StructuredGraphSpec
453StructuredRAPFactory<Scalar, LocalOrdinal, GlobalOrdinal, Node>::DeriveCoarseRAPStencil(
454 const FineStencilSpec& fineStencil, const int interpolationOrder) const {
455 const std::string prefix = "StructuredRAPFactory::DeriveCoarseRAPStencil: ";
456
457 TEUCHOS_TEST_FOR_EXCEPTION(fineStencil.numDimensions < 1 || fineStencil.numDimensions > 3,
458 Exceptions::RuntimeError,
459 prefix << "the number of dimensions must be between one and three.");
460 TEUCHOS_TEST_FOR_EXCEPTION(fineStencil.dofsPerNode <= 0, Exceptions::RuntimeError,
461 prefix << "dofsPerNode must be positive.");
462 TEUCHOS_TEST_FOR_EXCEPTION(fineStencil.stencilOffsets.empty(), Exceptions::RuntimeError,
463 prefix << "the fine-grid stencil is empty.");
464 TEUCHOS_TEST_FOR_EXCEPTION(interpolationOrder < 0 || interpolationOrder > 1,
465 Exceptions::RuntimeError,
466 prefix << "interpolation order " << interpolationOrder
467 << " is not supported; expected zero or one.");
468 for (const StencilOffset& offset : fineStencil.stencilOffsets) {
469 TEUCHOS_TEST_FOR_EXCEPTION(
470 offset.x < -1 || offset.x > 1 ||
471 offset.y < -1 || offset.y > 1 ||
472 offset.z < -1 || offset.z > 1,
473 Exceptions::RuntimeError,
474 prefix << "fine-grid stencil offset (" << offset.x << ", "
475 << offset.y << ", " << offset.z
476 << ") exceeds the supported radius one; disable "
477 "\"rap: prebuild coarse graph\" for wider stencils.");
478 }
479
480 StructuredGraphSpec coarseStencil;
481 coarseStencil.numDimensions = fineStencil.numDimensions;
482 coarseStencil.dofsPerNode = fineStencil.dofsPerNode;
483
484 if (interpolationOrder == 0) {
485 coarseStencil.stencilOffsets = fineStencil.stencilOffsets;
486 bool hasCenter = false;
487 for (const StencilOffset& offset : coarseStencil.stencilOffsets)
488 hasCenter = hasCenter || (offset.x == 0 && offset.y == 0 && offset.z == 0);
489 if (!hasCenter)
490 coarseStencil.stencilOffsets.push_back(StencilOffset{0, 0, 0});
491 } else {
492 // The supports of neighboring piecewise-linear basis functions overlap.
493 // For a radius-one fine stencil this produces the full radius-one coarse
494 // stencil in every active dimension.
495 const int minY = fineStencil.numDimensions > 1 ? -1 : 0;
496 const int maxY = fineStencil.numDimensions > 1 ? 1 : 0;
497 const int minZ = fineStencil.numDimensions > 2 ? -1 : 0;
498 const int maxZ = fineStencil.numDimensions > 2 ? 1 : 0;
499 for (int z = minZ; z <= maxZ; ++z)
500 for (int y = minY; y <= maxY; ++y)
501 for (int x = -1; x <= 1; ++x)
502 coarseStencil.stencilOffsets.push_back(StencilOffset{x, y, z});
503 }
504
505 int minimum[3] = {coarseStencil.stencilOffsets[0].x,
506 coarseStencil.stencilOffsets[0].y,
507 coarseStencil.stencilOffsets[0].z};
508 int maximum[3] = {minimum[0], minimum[1], minimum[2]};
509 for (const StencilOffset& offset : coarseStencil.stencilOffsets) {
510 const int components[3] = {offset.x, offset.y, offset.z};
511 for (int dim = 0; dim < fineStencil.numDimensions; ++dim) {
512 if (components[dim] < minimum[dim])
513 minimum[dim] = components[dim];
514 if (components[dim] > maximum[dim])
515 maximum[dim] = components[dim];
516 }
517 }
518
519 std::ostringstream description;
520 for (int dim = 0; dim < fineStencil.numDimensions; ++dim) {
521 if (dim != 0)
522 description << "x";
523 description << maximum[dim] - minimum[dim] + 1;
524 }
525 coarseStencil.description = description.str();
526 return coarseStencil;
527}

529// Prebuild sparsity structure of coarse matrix
530template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
531void StructuredRAPFactory<Scalar, LocalOrdinal, GlobalOrdinal, Node>::GetStructuredGraph(
532 RCP<Matrix>& Ac, const RCP<Matrix>& P,
533 const Teuchos::Array<LocalOrdinal>& lCoarseNodesPerDim,
534 const StructuredGraphSpec& graphSpec) const {
535 using local_graph_type = typename CrsGraph::local_graph_type;
536 using row_map_type = typename local_graph_type::row_map_type::non_const_type;
537 using entries_type = typename local_graph_type::entries_type::non_const_type;
538 using device_type = typename Node::device_type;
539 using execution_space = typename device_type::execution_space;

866 // For each low/interior/high boundary combination, precompute the affine
867 // number of stencil interactions preceding (x,y,z) in lexicographic order.
868 // This makes row offsets constant-time in the graph-fill kernel and avoids a
869 // device-wide scan without traversing the stencil for every interior node.
870 constexpr size_t numBoundaryClasses = 64;
871 Kokkos::Array<std::int64_t, numBoundaryClasses> prefixConstant{};
872 Kokkos::Array<std::int64_t, numBoundaryClasses> prefixX{};
873 Kokkos::Array<std::int64_t, numBoundaryClasses> prefixY{};
874 Kokkos::Array<std::int64_t, numBoundaryClasses> prefixZ{};
875 size_t localNnz = 0;
876 const int procCoordinates[3] = {myProcX, myProcY, myProcZ};
877 for (size_t stencil = 0; stencil < stencilSize; ++stencil) {
878 const StencilOffset& offset = stencilOffsets[stencil];
879 const int offsets[3] = {offset.x, offset.y, offset.z};
880 GO lower[3] = {0, 0, 0};
881 GO upper[3] = {localNodes[0], localNodes[1], localNodes[2]};
882 for (int dim = 0; dim < 3; ++dim) {
883 if (offsets[dim] < 0 && procCoordinates[dim] == 0)
884 lower[dim] = 1;
885 if (offsets[dim] > 0 && procCoordinates[dim] + 1 == procGrid[dim])
886 --upper[dim];
887 }
888 const std::int64_t lengths[3] = {
889 static_cast<std::int64_t>(upper[0] - lower[0]),
890 static_cast<std::int64_t>(upper[1] - lower[1]),
891 static_cast<std::int64_t>(upper[2] - lower[2])};
892 localNnz += static_cast<size_t>(lengths[0] * lengths[1] * lengths[2]) *
893 rowsPerNode * rowsPerNode;
894
895 for (size_t boundaryClass = 0; boundaryClass < numBoundaryClasses; ++boundaryClass) {
896 std::int64_t coordinates[3];
897 for (int dim = 0; dim < 3; ++dim) {
898 const bool low = boundaryClass & (size_t(1) << (2 * dim));
899 const bool high = boundaryClass & (size_t(2) << (2 * dim));
900 coordinates[dim] = low ? 0 : high ? static_cast<std::int64_t>(localNodes[dim] - 1)
901 : 1;
902 }
903 std::int64_t* coefficients[3] = {
904 &prefixX[boundaryClass], &prefixY[boundaryClass], &prefixZ[boundaryClass]};
905 auto addClamped = [&](const int dim, const std::int64_t multiplier) {
906 const std::int64_t begin = static_cast<std::int64_t>(lower[dim]);
907 const std::int64_t end = static_cast<std::int64_t>(upper[dim]);
908 if (coordinates[dim] >= end)
909 prefixConstant[boundaryClass] += (end - begin) * multiplier;
910 else if (coordinates[dim] >= begin) {
911 prefixConstant[boundaryClass] -= begin * multiplier;
912 *coefficients[dim] += multiplier;
913 }
914 };
915 const bool insideZ = coordinates[2] >= lower[2] && coordinates[2] < upper[2];
916 const bool insideY = coordinates[1] >= lower[1] && coordinates[1] < upper[1];
917 addClamped(2, lengths[0] * lengths[1]);
918 if (insideZ)
919 addClamped(1, lengths[0]);
920 if (insideZ && insideY)
921 addClamped(0, 1);
922 }
923 }

951 (z == 0 ? size_t(16) : 0) | (z + 1 == localNodes[2] ? size_t(32) : 0);

953 prefixConstant[boundaryClass] + prefixX[boundaryClass] * static_cast<std::int64_t>(x) +

986 GO neighborNode = 0;

988 x, y, z, stencilOffsets[stencil], localNodes, procGrid, rankDataDevice,
989 myProcX, myProcY, myProcZ, neighborRank, neighborNode))

992 for (size_t colDof = 0; colDof < rowsPerNode; ++colDof) {

999 while (region < numRemoteRegions && remoteRegionRanks[region] != neighborRank)

1002 const GO neighborNx = remoteRegionNx[region];
1003 const GO neighborNy = remoteRegionNy[region];
1004 const GO localNeighborNode = neighborNode - remoteRegionFirstNodes[region];
1005 const GO neighborX = localNeighborNode % neighborNx;
1006 const GO neighborY = (localNeighborNode / neighborNx) % neighborNy;
1007 const GO neighborZ = localNeighborNode / (neighborNx * neighborNy);
1008 const StencilOffset direction = remoteRegionDirections[region];

1010 size_t stride = 1;
1011 if (direction.x == 0) {
1012 regionNode += static_cast<size_t>(neighborX) * stride;
1013 stride *= static_cast<size_t>(neighborNx);
1014 }
1015 if (direction.y == 0) {
1016 regionNode += static_cast<size_t>(neighborY) * stride;

1018 }

1023 }

1030 }
1031 }
1032
1033 for (size_t entry = 1; entry < columnOffset; ++entry) {
1034 const LO value = nodeColumns[entry];
1035 size_t insertion = entry;
1036 while (insertion > 0 && value < nodeColumns[insertion - 1]) {
1037 nodeColumns[insertion] = nodeColumns[insertion - 1];
1038 --insertion;
1039 }
1040 nodeColumns[insertion] = value;
1041 }
1042
1043 const size_t firstRow = localNodeIndex * rowsPerNode;
1044 const size_t rowLength = columnOffset;
1045 for (size_t rowDof = 0; rowDof < rowsPerNode; ++rowDof) {
1046 const size_t rowStart = nodeRowStart + rowDof * rowLength;
1047 rowptr(firstRow + rowDof) = rowStart;
1048 for (size_t entry = 0; entry < columnOffset; ++entry)
1049 colind(rowStart + entry) = nodeColumns[entry];
1050 }
1051 if (localNodeIndex + 1 == localNumNodes)
1052 rowptr(localNumRows) = localNnz;
1053 });
1054 executionSpace.fence("StructuredRAP: graph column indices ready");
1055 }
1056
1057 if (debug) {
1058 int invalidColumnHost = 0;
1059 Kokkos::deep_copy(executionSpace, invalidColumnHost, invalidColumn);
1060 TEUCHOS_TEST_FOR_EXCEPTION(invalidColumnHost != 0, Exceptions::RuntimeError,
1061 "StructuredRAPFactory::GetStructuredGraph(" << graphSpec.description
1062 << "): a generated column GID was not found in the coarse column map.");
1063
1064 size_t unsortedRows = 0;
1065 Kokkos::parallel_reduce(
1066 "StructuredRAP: validate sorted graph rows", range_policy(executionSpace, 0, localNumRows),
1067 KOKKOS_LAMBDA(const size_t row, size_t& invalid) {
1068 for (size_t entry = rowptr(row) + 1; entry < rowptr(row + 1); ++entry) {
1069 if (colind(entry) < colind(entry - 1)) {
1070 ++invalid;
1071 break;
1072 }
1073 }
1074 },
1075 unsortedRows);
1076 TEUCHOS_TEST_FOR_EXCEPTION(unsortedRows != 0, Exceptions::RuntimeError,
1077 "StructuredRAPFactory::GetStructuredGraph(" << graphSpec.description
1078 << "): generated graph rows are not sorted.");
1079 }
1080
1081 local_graph_type localGraph(colind, rowptr);
1082 RCP<CrsGraph> graph = CrsGraphFactory::Build(
1083 localGraph, rowMap, colMap, rowMap, rowMap, paramList);
1084 using values_type = typename Matrix::local_matrix_type::values_type;
1085 values_type values(
1086 Kokkos::ViewAllocateWithoutInitializing(
1087 "StructuredRAP: matrix values"),
1088 localNnz);
1089 Ac = MatrixFactory::Build(graph, values, paramList);
1090}
