42KOKKOS_INLINE_FUNCTION void getLocalNodeIndices(const GO localNode,
43 const LocalNodes& localNodes,
44 GO& x, GO& y, GO& z) {
45 x = localNode % localNodes[0];
46 y = (localNode / localNodes[0]) % localNodes[1];
47 z = localNode / (localNodes[0] * localNodes[1]);

50template <class GO, class Offset, class LocalNodes, class ProcGrid, class RankData>
51KOKKOS_INLINE_FUNCTION bool resolveNeighbor(const GO x, const GO y, const GO z,
52 const Offset& offset,
53 const LocalNodes& localNodes,
54 const ProcGrid& procGrid,
55 const RankData& rankData,
56 const int myProcX,
57 const int myProcY,
58 const int myProcZ,
59 int& neighborRank,
60 GO& neighborNode) {
61 int neighborProcX = myProcX;
62 int neighborProcY = myProcY;
63 int neighborProcZ = myProcZ;
64 GO neighborX = x + static_cast<GO>(offset.x);
65 GO neighborY = y + static_cast<GO>(offset.y);
66 GO neighborZ = z + static_cast<GO>(offset.z);
67
68 if (neighborX < 0)
69 --neighborProcX;
70 else if (neighborX >= localNodes[0])
71 ++neighborProcX;
72 if (neighborY < 0)
73 --neighborProcY;
74 else if (neighborY >= localNodes[1])
75 ++neighborProcY;
76 if (neighborZ < 0)
77 --neighborProcZ;
78 else if (neighborZ >= localNodes[2])
79 ++neighborProcZ;
80
81 if (neighborProcX < 0 || neighborProcX >= procGrid[0] ||
82 neighborProcY < 0 || neighborProcY >= procGrid[1] ||
83 neighborProcZ < 0 || neighborProcZ >= procGrid[2])
84 return false;
85
86 neighborRank = neighborProcZ * procGrid[0] * procGrid[1] + neighborProcY * procGrid[0] + neighborProcX;
87 const GO neighborLocalNx = rankData(4 * neighborRank + 1);
88 const GO neighborLocalNy = rankData(4 * neighborRank + 2);
89 const GO neighborLocalNz = rankData(4 * neighborRank + 3);
90 if (neighborX < 0)
91 neighborX = neighborLocalNx - 1;
92 else if (neighborX >= localNodes[0])
93 neighborX = 0;
94 if (neighborY < 0)
95 neighborY = neighborLocalNy - 1;
96 else if (neighborY >= localNodes[1])
97 neighborY = 0;
98 if (neighborZ < 0)
99 neighborZ = neighborLocalNz - 1;
100 else if (neighborZ >= localNodes[2])
101 neighborZ = 0;
102
103 neighborNode = rankData(4 * neighborRank) +
104 neighborZ * neighborLocalNx * neighborLocalNy +
105 neighborY * neighborLocalNx + neighborX;
106 return true;
107}

184template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
185void StructuredRAPFactory<Scalar, LocalOrdinal, GlobalOrdinal, Node>::DeclareInput(Level& fineLevel, Level& coarseLevel) const {
186 const Teuchos::ParameterList& pL = GetParameterList();
187 TEUCHOS_TEST_FOR_EXCEPTION(
188 !pL.get<bool>("transpose: use implicit"), Exceptions::RuntimeError,
189 "StructuredRAPFactory requires \"transpose: use implicit\" = true because "
190 "the prebuilt coarse graph assumes the Galerkin product P^T A P.");
191
192 const bool prebuildCoarseGraph = pL.get<bool>("rap: prebuild coarse graph");
193 const bool useRAPDelegate = !prebuildCoarseGraph;
194
195 if (useRAPDelegate) {
196 ConfigureRAPFactoryDelegate();
197 coarseLevel.DeclareInput("A", rapFactoryDelegate_.get(), this);
198 coarseLevel.DeclareInput("RAP reuse data", rapFactoryDelegate_.get(), this);
199 hasDeclaredInput_ = true;
200 return;
201 }
202
203 Input(fineLevel, "A");
204 Input(coarseLevel, "P");
205
206 if (prebuildCoarseGraph) {
207 const RCP<const FactoryBase> coarseDimensionsFactory =
208 GetStructuredMetadataFactory("lCoarseNodesPerDim");
209 const RCP<const FactoryBase> interpolationOrderFactory =
210 GetStructuredMetadataFactory("structuredInterpolationOrder");
211 fineLevel.DeclareInput("lCoarseNodesPerDim", coarseDimensionsFactory.get(), this);
212 fineLevel.DeclareInput("structuredInterpolationOrder", interpolationOrderFactory.get(), this);
213 if (fineLevel.GetLevelID() == 0) {
214 TEUCHOS_TEST_FOR_EXCEPTION(!fineLevel.IsAvailable("numDimensions", NoFactory::get()),
215 Exceptions::RuntimeError,
216 "StructuredRAPFactory: numDimensions was not provided on level 0.");
217 TEUCHOS_TEST_FOR_EXCEPTION(!fineLevel.IsAvailable("lNodesPerDim", NoFactory::get()),
218 Exceptions::RuntimeError,
219 "StructuredRAPFactory: lNodesPerDim was not provided on level 0.");
220 fineLevel.DeclareInput("numDimensions", NoFactory::get(), this);
221 fineLevel.DeclareInput("lNodesPerDim", NoFactory::get(), this);
222 } else {
223 Input(fineLevel, "numDimensions");
224 Input(fineLevel, "lNodesPerDim");
225 }
226 }
227
228 // call DeclareInput of all user-given transfer factories
229 for (std::vector<RCP<const FactoryBase>>::const_iterator it = transferFacts_.begin(); it != transferFacts_.end(); ++it)
230 (*it)->CallDeclareInput(coarseLevel);
231
232 hasDeclaredInput_ = true;
233}

235// Infer the fine-grid stencil of A from one representative interior node
236template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
237typename StructuredRAPFactory<Scalar, LocalOrdinal, GlobalOrdinal, Node>::FineStencilSpec
238StructuredRAPFactory<Scalar, LocalOrdinal, GlobalOrdinal, Node>::DetectFineStencil(
239 const Matrix& A, const int numDimensions,
240 const Teuchos::Array<LocalOrdinal>& lFineNodesPerDim) const {
241 const std::string prefix = "StructuredRAPFactory::DetectFineStencil: ";
242
243 TEUCHOS_TEST_FOR_EXCEPTION(numDimensions < 1 || numDimensions > 3, Exceptions::RuntimeError,
244 prefix << "the number of dimensions must be between one and three.");
245 TEUCHOS_TEST_FOR_EXCEPTION(lFineNodesPerDim.size() < numDimensions, Exceptions::RuntimeError,
246 prefix << "insufficient fine-grid dimensions were supplied.");
247
248 const auto rowMap = A.getRowMap();
249 const auto colMap = A.getColMap();
250 const auto comm = rowMap->getComm();
251 const auto graph = A.getCrsGraph();
252 TEUCHOS_TEST_FOR_EXCEPTION(graph.is_null(), Exceptions::RuntimeError,
253 prefix << "A does not expose a CRS graph.");
254
255 const LO dofsPerNode = Teuchos::as<LO>(A.GetFixedBlockSize());
256 TEUCHOS_TEST_FOR_EXCEPTION(dofsPerNode <= 0, Exceptions::RuntimeError,
257 prefix << "A must have a positive fixed block size.");
258
259 Kokkos::Array<LO, 3> fineNodes{{LO(1), LO(1), LO(1)}};
260 Kokkos::Array<LO, 3> interior{{LO(0), LO(0), LO(0)}};
261 size_t numFineNodes = 1;
262 bool hasLocalInterior = true;
263 // Select interior node in the local subdomain as a representative node for detecting the fine-grid stencil
264 for (int dim = 0; dim < numDimensions; ++dim) {
265 fineNodes[dim] = lFineNodesPerDim[dim];
266 hasLocalInterior = hasLocalInterior && fineNodes[dim] >= LO(3);
267 if (fineNodes[dim] > LO(0)) {
268 interior[dim] = fineNodes[dim] / LO(2);
269 numFineNodes *= Teuchos::as<size_t>(fineNodes[dim]);
270 } else {
271 numFineNodes = 0;
272 }
273 }
274
275 const bool localDimensionsMatch =
276 rowMap->getLocalNumElements() ==
277 numFineNodes * Teuchos::as<size_t>(dofsPerNode);
278 int invalidLocalDimensions = localDimensionsMatch ? 0 : 1;
279 int invalidDimensions = 0;
280 Teuchos::reduceAll(*comm, Teuchos::REDUCE_MAX, 1,
281 &invalidLocalDimensions, &invalidDimensions);
282 TEUCHOS_TEST_FOR_EXCEPTION(
283 invalidDimensions != 0, Exceptions::RuntimeError,
284 prefix << "the structured local dimensions and fixed block size do not match A on every rank.");
285
286 const LO interiorNode = interior[2] * fineNodes[0] * fineNodes[1] +
287 interior[1] * fineNodes[0] + interior[0];
288
289 bool hasCompleteLocalStencil = hasLocalInterior;
290 // Every rank checks its candidate interior node which can be used to detect the fine-grid stencil
291 if (hasCompleteLocalStencil) {
292 for (LO rowDof = 0; rowDof < dofsPerNode && hasCompleteLocalStencil; ++rowDof) {
293 const LO row = interiorNode * dofsPerNode + rowDof;
294 Teuchos::ArrayView<const LO> columns;
295 graph->getLocalRowView(row, columns);
296 if (columns.size() == 0) {
297 hasCompleteLocalStencil = false;
298 break;
299 }
300 for (int entry = 0; entry < columns.size(); ++entry) {
301 const GO globalColumn = colMap->getGlobalElement(columns[entry]);
302 if (!rowMap->isNodeGlobalElement(globalColumn)) {
303 hasCompleteLocalStencil = false;
304 break;
305 }
306 }
307 }
308 }
309
310 const int myRank = comm->getRank();
311 const int numRanks = comm->getSize();
312 const int candidateRank = hasCompleteLocalStencil ? myRank : numRanks;
313 int detectorRank = numRanks;
314 Teuchos::reduceAll(*comm, Teuchos::REDUCE_MIN, 1, &candidateRank, &detectorRank);
315 TEUCHOS_TEST_FOR_EXCEPTION(
316 detectorRank == numRanks, Exceptions::RuntimeError,
317 prefix << "no MPI rank owns a local interior node whose complete stencil is locally owned; "
318 << "increase the local structured subdomain size.");
319
320 FineStencilSpec stencil;
321 stencil.numDimensions = numDimensions;
322 stencil.dofsPerNode = dofsPerNode;
323
324 // Only the elected rank performs detection
325 if (myRank == detectorRank) {
326 for (LO rowDof = 0; rowDof < dofsPerNode; ++rowDof) {
327 const LO row = interiorNode * dofsPerNode + rowDof;
328 Teuchos::ArrayView<const LO> columns;
329 graph->getLocalRowView(row, columns);
330
331 for (int entry = 0; entry < columns.size(); ++entry) {
332 const GO globalColumn = colMap->getGlobalElement(columns[entry]);
333 const LO ownedColumn = rowMap->getLocalElement(globalColumn);
334 TEUCHOS_TEST_FOR_EXCEPTION(
335 ownedColumn == Teuchos::OrdinalTraits<LO>::invalid(), Exceptions::RuntimeError,
336 prefix << "the selected detector row contains a nonowned column.");
337
338 const LO columnNode = ownedColumn / dofsPerNode;
339 const LO columnDof = ownedColumn % dofsPerNode;
340 LO columnX, columnY, columnZ;
341 StructuredRAPFactoryDetails::getLocalNodeIndices(
342 columnNode, fineNodes, columnX, columnY, columnZ);
343 const StencilOffset offset{
344 Teuchos::as<int>(columnX - interior[0]),
345 Teuchos::as<int>(columnY - interior[1]),
346 Teuchos::as<int>(columnZ - interior[2])};
347
348 bool knownOffset = false;
349 for (const StencilOffset& known : stencil.stencilOffsets) {
350 if (known.x == offset.x && known.y == offset.y && known.z == offset.z) {
351 knownOffset = true;
352 break;
353 }
354 }
355 if (!knownOffset)
356 stencil.stencilOffsets.push_back(offset);
357
358 stencil.entries.push_back(FineStencilEntry{
359 offset, rowDof, columnDof, Teuchos::as<LO>(entry)});
360 }
361 }
362 }
363
364 int counts[2] = {Teuchos::as<int>(stencil.stencilOffsets.size()),
365 Teuchos::as<int>(stencil.entries.size())};
366 Teuchos::broadcast<int, int>(*comm, detectorRank, 2, counts);
367
368 Teuchos::Array<int> offsetData(3 * counts[0]);
369 Teuchos::Array<int> entryOffsetData(3 * counts[1]);
370 Teuchos::Array<LO> entryData(3 * counts[1]);
371 if (myRank == detectorRank) {
372 for (int offset = 0; offset < counts[0]; ++offset) {
373 offsetData[3 * offset] = stencil.stencilOffsets[offset].x;
374 offsetData[3 * offset + 1] = stencil.stencilOffsets[offset].y;
375 offsetData[3 * offset + 2] = stencil.stencilOffsets[offset].z;
376 }
377 for (int entry = 0; entry < counts[1]; ++entry) {
378 entryOffsetData[3 * entry] = stencil.entries[entry].offset.x;
379 entryOffsetData[3 * entry + 1] = stencil.entries[entry].offset.y;
380 entryOffsetData[3 * entry + 2] = stencil.entries[entry].offset.z;
381 entryData[3 * entry] = stencil.entries[entry].rowDof;
382 entryData[3 * entry + 1] = stencil.entries[entry].columnDof;
383 entryData[3 * entry + 2] = stencil.entries[entry].entryOrdinal;
384 }
385 }
386
387 if (counts[0] > 0)
388 Teuchos::broadcast<int, int>(*comm, detectorRank, 3 * counts[0],
389 offsetData.getRawPtr());
390 if (counts[1] > 0) {
391 Teuchos::broadcast<int, int>(*comm, detectorRank, 3 * counts[1],
392 entryOffsetData.getRawPtr());
393 Teuchos::broadcast<int, LO>(*comm, detectorRank, 3 * counts[1],
394 entryData.getRawPtr());
395 }
396
397 if (myRank != detectorRank) {
398 stencil.stencilOffsets.reserve(counts[0]);
399 for (int offset = 0; offset < counts[0]; ++offset)
400 stencil.stencilOffsets.push_back(StencilOffset{
401 offsetData[3 * offset], offsetData[3 * offset + 1],
402 offsetData[3 * offset + 2]});
403
404 stencil.entries.reserve(counts[1]);
405 for (int entry = 0; entry < counts[1]; ++entry)
406 stencil.entries.push_back(FineStencilEntry{
407 StencilOffset{entryOffsetData[3 * entry],
408 entryOffsetData[3 * entry + 1],
409 entryOffsetData[3 * entry + 2]},
410 entryData[3 * entry], entryData[3 * entry + 1],
411 entryData[3 * entry + 2]});
412 }
413
414 TEUCHOS_TEST_FOR_EXCEPTION(stencil.stencilOffsets.empty(), Exceptions::RuntimeError,
415 prefix << "the representative interior rows are empty.");
416 return stencil;
417}

419template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
420typename StructuredRAPFactory<Scalar, LocalOrdinal, GlobalOrdinal, Node>::StructuredGraphSpec
421StructuredRAPFactory<Scalar, LocalOrdinal, GlobalOrdinal, Node>::DeriveCoarseRAPStencil(
422 const FineStencilSpec& fineStencil, const int interpolationOrder) const {
423 const std::string prefix = "StructuredRAPFactory::DeriveCoarseRAPStencil: ";
424
425 TEUCHOS_TEST_FOR_EXCEPTION(fineStencil.numDimensions < 1 || fineStencil.numDimensions > 3,
426 Exceptions::RuntimeError,
427 prefix << "the number of dimensions must be between one and three.");
428 TEUCHOS_TEST_FOR_EXCEPTION(fineStencil.dofsPerNode <= 0, Exceptions::RuntimeError,
429 prefix << "dofsPerNode must be positive.");
430 TEUCHOS_TEST_FOR_EXCEPTION(fineStencil.stencilOffsets.empty(), Exceptions::RuntimeError,
431 prefix << "the fine-grid stencil is empty.");
432 TEUCHOS_TEST_FOR_EXCEPTION(interpolationOrder < 0 || interpolationOrder > 1,
433 Exceptions::RuntimeError,
434 prefix << "interpolation order " << interpolationOrder
435 << " is not supported; expected zero or one.");
436 for (const StencilOffset& offset : fineStencil.stencilOffsets) {
437 TEUCHOS_TEST_FOR_EXCEPTION(
438 offset.x < -1 || offset.x > 1 ||
439 offset.y < -1 || offset.y > 1 ||
440 offset.z < -1 || offset.z > 1,
441 Exceptions::RuntimeError,
442 prefix << "fine-grid stencil offset (" << offset.x << ", "
443 << offset.y << ", " << offset.z
444 << ") exceeds the supported radius one; disable "
445 "\"rap: prebuild coarse graph\" for wider stencils.");
446 }
447
448 StructuredGraphSpec coarseStencil;
449 coarseStencil.numDimensions = fineStencil.numDimensions;
450 coarseStencil.dofsPerNode = fineStencil.dofsPerNode;
451
452 if (interpolationOrder == 0) {
453 coarseStencil.stencilOffsets = fineStencil.stencilOffsets;
454 bool hasCenter = false;
455 for (const StencilOffset& offset : coarseStencil.stencilOffsets)
456 hasCenter = hasCenter || (offset.x == 0 && offset.y == 0 && offset.z == 0);
457 if (!hasCenter)
458 coarseStencil.stencilOffsets.push_back(StencilOffset{0, 0, 0});
459 } else {
460 // The supports of neighboring piecewise-linear basis functions overlap.
461 // For a radius-one fine stencil this produces the full radius-one coarse
462 // stencil in every active dimension.
463 const int minY = fineStencil.numDimensions > 1 ? -1 : 0;
464 const int maxY = fineStencil.numDimensions > 1 ? 1 : 0;
465 const int minZ = fineStencil.numDimensions > 2 ? -1 : 0;
466 const int maxZ = fineStencil.numDimensions > 2 ? 1 : 0;
467 for (int z = minZ; z <= maxZ; ++z)
468 for (int y = minY; y <= maxY; ++y)
469 for (int x = -1; x <= 1; ++x)
470 coarseStencil.stencilOffsets.push_back(StencilOffset{x, y, z});
471 }
472
473 int minimum[3] = {coarseStencil.stencilOffsets[0].x,
474 coarseStencil.stencilOffsets[0].y,
475 coarseStencil.stencilOffsets[0].z};
476 int maximum[3] = {minimum[0], minimum[1], minimum[2]};
477 for (const StencilOffset& offset : coarseStencil.stencilOffsets) {
478 const int components[3] = {offset.x, offset.y, offset.z};
479 for (int dim = 0; dim < fineStencil.numDimensions; ++dim) {
480 if (components[dim] < minimum[dim])
481 minimum[dim] = components[dim];
482 if (components[dim] > maximum[dim])
483 maximum[dim] = components[dim];
484 }
485 }
486
487 std::ostringstream description;
488 for (int dim = 0; dim < fineStencil.numDimensions; ++dim) {
489 if (dim != 0)
490 description << "x";
491 description << maximum[dim] - minimum[dim] + 1;
492 }
493 coarseStencil.description = description.str();
494 return coarseStencil;
495}

498template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
499typename StructuredRAPFactory<Scalar, LocalOrdinal, GlobalOrdinal, Node>::StructuredGraphSpec
500StructuredRAPFactory<Scalar, LocalOrdinal, GlobalOrdinal, Node>::GetStructuredGraphSpec(
501 const Matrix& A, const Matrix& P, const int numDimensions,

505 DetectFineStencil(A, numDimensions, lFineNodesPerDim);
506

517 << " coarse stencil with " << coarseStencil.stencilOffsets.size()
518 << " nodal offset(s)";
519 out << std::endl;
520
521 return coarseStencil;
522}

524// Prebuild sparsity structure of coarse matrix
525template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
526void StructuredRAPFactory<Scalar, LocalOrdinal, GlobalOrdinal, Node>::GetStructuredGraph(
527 RCP<Matrix>& Ac, RCP<Matrix> P,
528 const Teuchos::Array<LocalOrdinal>& lCoarseNodesPerDim,
529 const StructuredGraphSpec& graphSpec) const {
530 using local_graph_type = typename CrsGraph::local_graph_type;
531 using row_map_type = typename local_graph_type::row_map_type::non_const_type;
532 using entries_type = typename local_graph_type::entries_type::non_const_type;
533 using device_type = typename Node::device_type;
534 using execution_space = typename device_type::execution_space;
535 using range_policy = Kokkos::RangePolicy<execution_space, Kokkos::IndexType<size_t>>;
536
537 TEUCHOS_TEST_FOR_EXCEPTION(graphSpec.numDimensions < 1 || graphSpec.numDimensions > 3, Exceptions::RuntimeError,
538 "StructuredRAPFactory::GetStructuredGraph(" << graphSpec.description
539 << "): number of dimensions must be between one and three.");
540 TEUCHOS_TEST_FOR_EXCEPTION(lCoarseNodesPerDim.size() < graphSpec.numDimensions, Exceptions::RuntimeError,

668 // Keep this factorization limit synchronized with Galeri_XpetraMaps_def.hpp.
669 const int galeriFactorSearchLimit = 50;

671 for (int factor = 2; factor < galeriFactorSearchLimit; ++factor) {
672 while (procTemp % factor == 0) {
673 ++factors[factor];
674 procTemp /= factor;
675 }
676 }
677 procGrid[0] = procTemp;
678 for (int factor = galeriFactorSearchLimit - 1; factor > 0; --factor) {
679 while (factors[factor] != 0) {
680 if (procGrid[0] <= procGrid[1] && procGrid[0] <= procGrid[2])
681 procGrid[0] *= factor;
682 else if (procGrid[1] <= procGrid[0] && procGrid[1] <= procGrid[2])
683 procGrid[1] *= factor;
684 else
685 procGrid[2] *= factor;
686 --factors[factor];
687 }
688 }
689 }

797 (procDy == 0 ? neighborNy : 1) *
798 (procDz == 0 ? neighborNz : 1));

803 remoteRegionNx[numRemoteRegions] = neighborNx;
804 remoteRegionNy[numRemoteRegions] = neighborNy;
805 remoteRegionNz[numRemoteRegions] = neighborNz;

807 remoteRegionOffsets[numRemoteRegions] + regionNodes * rowsPerNode;

809 }

851 neighborZ = static_cast<GO>(regionNode);
