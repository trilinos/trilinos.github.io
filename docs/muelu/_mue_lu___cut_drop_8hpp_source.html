220 }

222 if (results(offset + y) != UNDECIDED) {
223 // sort undecided to the right
224 return false;
225 } else {
226 return get_value(x) > get_value(y);
227 }
228 }
229 }

231 private:
232 KOKKOS_INLINE_FUNCTION
233 magnitudeType get_value_impl(size_t x) const {
234 if constexpr (measure == Misc::SmoothedAggregationMeasure) {
235 auto x_aij = ATS::magnitude(A.values(offset + x) * A.values(offset + x));
236 auto x_aiiajj = ATS::magnitude(diag(rlid) * diag(A.graph.entries(offset + x)));
237 return (x_aij / x_aiiajj);
238 } else if constexpr (measure == Misc::SignedRugeStuebenMeasure) {
239 auto val = A.values(offset + x);
240 auto neg_aij = -ATS::real(val);
241 auto max_neg_aik = ATS::real(diag(rlid));
242 auto v = neg_aij / max_neg_aik;
243 if (ATS::real(v) <= mATS::zero()) {
244 return -ATS::magnitude(v * v);
245 } else {
246 return ATS::magnitude(v * v);
247 }
248 } else if constexpr (measure == Misc::SignedSmoothedAggregationMeasure) {
249 auto val = A.values(offset + x);
250 auto x_aiiajj = ATS::magnitude(diag(rlid) * diag(A.graph.entries(offset + x)));
251 const bool is_nonpositive = ATS::real(val) <= mATS::zero();
252 magnitudeType aij2 = ATS::magnitude(val) * ATS::magnitude(val);
253 if (!is_nonpositive)
254 aij2 = -aij2;
255 return (aij2 / x_aiiajj);
256 }
257 }
258 };

260 using comparator_type = Comparator<local_matrix_type, diag_view_type>;
261
262 KOKKOS_INLINE_FUNCTION
263 comparator_type getComparator(local_ordinal_type rlid) const {
264 return comparator_type(A, diag, rlid, results, values);

936 auto row = A.rowConst(rlid);
937 size_t nnz = row.length;
938
939 auto drop_view = Kokkos::subview(results, Kokkos::make_pair(A.graph.row_map(rlid), A.graph.row_map(rlid + 1)));
940 auto row_permutation = Kokkos::subview(index, Kokkos::make_pair(A.graph.row_map(rlid), A.graph.row_map(rlid + 1)));
941
942 auto comparator = comparison.getComparator(rlid);
943
944 for (size_t i = 0; i < nnz; ++i) {
945 row_permutation(i) = i;
946 }
947 Misc::serialHeapSort(row_permutation, comparator);
948
949#ifdef MUELU_COALESCE_DROP_DEBUG
950 {
951 Kokkos::printf("SoC: ");
952 for (local_ordinal_type k = 0; k < row.length; ++k) {
953 Kokkos::printf("%5f ", comparator.get_value(k));
954 }
955 Kokkos::printf("\n");

959 Kokkos::printf("%5f ", comparator.get_value(x));
960 }
961 Kokkos::printf("\n");
962 }
963#endif
964
965 size_t keepStart = 0;
966 size_t dropStart = nnz;
967 // find index where dropping starts
968 for (size_t i = 2 * blockSize; i < nnz; ++i) {
969 auto const& x = row_permutation(i - 1);
970 auto const& y = row_permutation(i);
971 if ((drop_view(x) != UNDECIDED) && (drop_view(y) == UNDECIDED))
972 keepStart = i;
973 if ((drop_view(x) != UNDECIDED) || (drop_view(y) != UNDECIDED))
974 continue;
975 magnitudeType x_aij = comparator.get_value(x);
976 magnitudeType y_aij = comparator.get_value(y);
977 if (eps * eps * x_aij > y_aij) {
978 if (i < dropStart) {
979#ifdef MUELU_COALESCE_DROP_DEBUG
980 {
981 Kokkos::printf("Changing dropStart: %d %5f %5f %5f\n", x_aij, y_aij, eps, i);
982 }
983#endif
984 dropStart = i;
985 }
986 }
987 }
988
989#ifdef MUELU_COALESCE_DROP_DEBUG
990 {
991 Kokkos::printf("cut-drop: %d %d %d\n", keepStart, dropStart, nnz);
992 }
993#endif
994
995 // drop everything to the right of where values stop passing threshold
996 for (size_t i = keepStart; i < nnz; ++i) {
997 drop_view(row_permutation(i)) = Kokkos::max(dropStart <= i ? DROP : KEEP, drop_view(row_permutation(i)));
998 }
999 }
