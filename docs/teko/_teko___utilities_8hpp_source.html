222inline MultiVector datacopy(const MultiVector &src, MultiVector &dst) {
223 if (dst == Teuchos::null) return deepcopy(src);
224
225 bool rangeCompat = src->range()->isCompatible(*dst->range());
226 bool domainCompat = src->domain()->isCompatible(*dst->domain());
227
228 if (not(rangeCompat && domainCompat)) return deepcopy(src);
229
230 // perform data copy
231 Thyra::assign<double>(dst.ptr(), *src);
232 return dst;
233}
