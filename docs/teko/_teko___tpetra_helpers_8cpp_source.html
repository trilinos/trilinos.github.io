129void identityRowIndices(const Tpetra::Map<LO, GO, NT>& rowMap,
130 const Tpetra::CrsMatrix<ST, LO, GO, NT>& mat, std::vector<GO>& outIndices) {
131 // loop over elements owned by this processor
132 for (size_t i = 0; i < rowMap.getLocalNumElements(); i++) {
133 bool rowIsIdentity = true;
134 GO rowGID = rowMap.getGlobalElement(i);
135
136 size_t numEntries = mat.getNumEntriesInGlobalRow(i);
137 auto indices = typename Tpetra::CrsMatrix<ST, LO, GO, NT>::nonconst_global_inds_host_view_type(
138 Kokkos::ViewAllocateWithoutInitializing("rowIndices"), numEntries);
139 auto values = typename Tpetra::CrsMatrix<ST, LO, GO, NT>::nonconst_values_host_view_type(
140 Kokkos::ViewAllocateWithoutInitializing("rowIndices"), numEntries);
141
142 mat.getGlobalRowCopy(rowGID, indices, values, numEntries);
143
144 // loop over the columns of this row
145 for (size_t j = 0; j < numEntries; j++) {
146 GO colGID = indices(j);
147
148 // look at row entries
149 if (colGID == rowGID)
150 rowIsIdentity &= values(j) == 1.0;
151 else
152 rowIsIdentity &= values(j) == 0.0;
153
154 // not a dirchlet row...quit
155 if (not rowIsIdentity) break;
156 }
157
158 // save a row that is dirchlet
159 if (rowIsIdentity) outIndices.push_back(rowGID);
160 }
161}

173void zeroMultiVectorRowIndices(Tpetra::MultiVector<ST, LO, GO, NT>& mv,
174 const std::vector<GO>& zeroIndices) {
175 LO colCnt = mv.getNumVectors();
176 std::vector<GO>::const_iterator itr;
177
178 // loop over the indices to zero
179 for (itr = zeroIndices.begin(); itr != zeroIndices.end(); ++itr) {
180 // loop over columns
181 for (int j = 0; j < colCnt; j++) mv.replaceGlobalValue(*itr, j, 0.0);
182 }
183}

214bool isTpetraLinearOp(const LinearOp& op) {
215 // See if the operator is a TpetraLinearOp
216 RCP<const Thyra::TpetraLinearOp<ST, LO, GO, NT> > tOp =
217 rcp_dynamic_cast<const Thyra::TpetraLinearOp<ST, LO, GO, NT> >(op);
218 if (!tOp.is_null()) return true;
219
220 // See if the operator is a wrapped TpetraLinearOp
221 ST scalar = 0.0;
222 Thyra::EOpTransp transp = Thyra::NOTRANS;
223 RCP<const Thyra::LinearOpBase<ST> > wrapped_op;
224 Thyra::unwrap(op, &scalar, &transp, &wrapped_op);
225 tOp = rcp_dynamic_cast<const Thyra::TpetraLinearOp<ST, LO, GO, NT> >(wrapped_op);
226 if (!tOp.is_null()) return true;
227
228 return false;
229}

231RCP<const Tpetra::CrsMatrix<ST, LO, GO, NT> > getTpetraCrsMatrix(const LinearOp& op, ST* scalar,
232 bool* transp) {
233 // If the operator is a TpetraLinearOp
234 RCP<const Thyra::TpetraLinearOp<ST, LO, GO, NT> > tOp =
235 rcp_dynamic_cast<const Thyra::TpetraLinearOp<ST, LO, GO, NT> >(op);
236 if (!tOp.is_null()) {
237 RCP<const Tpetra::CrsMatrix<ST, LO, GO, NT> > matrix =
238 rcp_dynamic_cast<const Tpetra::CrsMatrix<ST, LO, GO, NT> >(tOp->getConstTpetraOperator(),
239 true);
240 *scalar = 1.0;
241 *transp = false;
242 return matrix;
243 }
244
245 // If the operator is a wrapped TpetraLinearOp
246 RCP<const Thyra::LinearOpBase<ST> > wrapped_op;
247 Thyra::EOpTransp eTransp = Thyra::NOTRANS;
248 Thyra::unwrap(op, scalar, &eTransp, &wrapped_op);
249 tOp = rcp_dynamic_cast<const Thyra::TpetraLinearOp<ST, LO, GO, NT> >(wrapped_op, true);
250 if (!tOp.is_null()) {
251 RCP<const Tpetra::CrsMatrix<ST, LO, GO, NT> > matrix =
252 rcp_dynamic_cast<const Tpetra::CrsMatrix<ST, LO, GO, NT> >(tOp->getConstTpetraOperator(),
253 true);
254 *transp = true;
255 if (eTransp == Thyra::NOTRANS) *transp = false;
256 return matrix;
257 }
258
259 return Teuchos::null;
260}
