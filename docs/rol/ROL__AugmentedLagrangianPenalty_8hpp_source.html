112 virtual Real value( const Vector<Real> &x, Real &tol ) {
113 // Compute penalty function value
114 Real val = getDualVec(x,tol)->norm();
115 val = val*val/(Real(2)*penaltyParameter_);
116 val *= cscale_;
117 return val;
118 }
119
120 virtual void gradient( Vector<Real> &g, const Vector<Real> &x, Real &tol ) {
121 // Compute penalty function gradient
122 con_->applyAdjointJacobian(g,*getDualVec(x,tol),x,tol);
123 g.scale(cscale_);
124 }
125
126 virtual void hessVec( Vector<Real> &hv, const Vector<Real> &v, const Vector<Real> &x, Real &tol ) {
127 // Apply penalty Hessian to a vector
128 if (hessianApprox_ >= 3) {
129 hv.zero();
130 return;
131 }
132 con_->applyJacobian(*primConVector1_,v,x,tol);
133 dualConVector_->set(primConVector1_->dual());
134 primConVector2_->set(*getConstraintVec(x,tol));
135 primConVector2_->axpy(Real(1)/penaltyParameter_,multiplier_->dual());
136 proj_->applyJacobian(*primConVector1_,*primConVector2_);
137 dualConVector_->axpy(-1,primConVector1_->dual());
138 dualConVector_->scale(penaltyParameter_);
139 con_->applyAdjointJacobian(hv,*dualConVector_,x,tol);
140 if (hessianApprox_ == 0) {
141 con_->applyAdjointHessian(*dualOptVector_,*getDualVec(x,tol),v,x,tol);
142 hv.plus(*dualOptVector_);
143 }
144 hv.scale(cscale_);
145 }
146

183 // Return constraint value
184 const Ptr<const Vector<Real>> getConstraintVec( const Vector<Real> &x, Real &tol ) {
185 int key(0);
186 if (!conValue_->isComputed(key)) {
187 if (conValue_->isNull(key)) conValue_->allocate(*primConVector1_,key);
188 con_->value(*conValue_->set(key),x,tol); ncval_++;
189 }
190 return conValue_->get(key);
191 }
192
193 // Return dual value (proximal point step)
194 const Ptr<const Vector<Real>> getDualVec( const Vector<Real> &x, Real &tol ) {
195 int key(0);
196 if (!dualValue_->isComputed(key)) {
197 if (dualValue_->isNull(key)) dualValue_->allocate(*dualConVector_,key);
198 primConVector1_->set(*getConstraintVec(x,tol));
199 primConVector1_->axpy(Real(1)/penaltyParameter_,multiplier_->dual());
200 dualConVector_->set(primConVector1_->dual());
201 proj_->project(*primConVector1_,bhs_);
202 dualConVector_->axpy(Real(-1),primConVector1_->dual());
203 dualConVector_->scale(penaltyParameter_);
204 dualValue_->set(key)->set(*dualConVector_);
205 }
206 return dualValue_->get(key);
207 }

209 // Return total number of constraint evaluations
210 int getNumberConstraintEvaluations(void) const {
211 return ncval_;
212 }

216 ncval_ = 0;
217 }
