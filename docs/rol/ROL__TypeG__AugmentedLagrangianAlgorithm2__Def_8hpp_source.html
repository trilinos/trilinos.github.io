95template<typename Real>
96void AugmentedLagrangianAlgorithm2<Real>::initialize( Vector<Real> &x,
97 const Vector<Real> &g,
98 AugmentedLagrangianObjective2<Real> &alobj,
99 std::ostream &outStream ) {
100
101 if (useRelTol_) {
102 outStream << "Warning: \"Use Relative Tolerances\" parameter is unsupported!" << std::endl;
103 useRelTol_ = false;
104 }
105 if (subproblemTolFactor_ >= Real(1)) {
106 outStream << "Warning: \"Subproblem Tolerance Lower Bound Factor\" is greater than or equal to 1! "
107 << "The lower bounds will not be tighter than the corresponding outer tolerances."
108 << std::endl;
109 }
110
111 const Real one(1);
112 Real tol = std::sqrt(ROL_EPSILON<Real>());
113 // > TypeG::Algorithm<Real>::initialize(x,g,l,c);
114 if (state_->iterateVec == nullPtr) {
115 state_->iterateVec = x.clone();
116 }
117 state_->iterateVec->set(x);
118 // > if (state_->lagmultVec == nullPtr) {
119 // > state_->lagmultVec = mul.clone();
120 // > }
121 // > state_->lagmultVec->set(mul);
122 if (state_->stepVec == nullPtr) {
123 state_->stepVec = x.clone();
124 }
125 state_->stepVec->zero();
126 if (state_->gradientVec == nullPtr) {
127 state_->gradientVec = g.clone();
128 }
129 state_->gradientVec->set(g);
130 // > if (state_->constraintVec == nullPtr) {
131 // > state_->constraintVec = c.clone();
132 // > }
133 // > state_->constraintVec->zero();
134 if (state_->minIterVec == nullPtr) {
135 state_->minIterVec = x.clone();
136 }
137 state_->minIterVec->set(x);
138 state_->minIter = state_->iter;
139 state_->minValue = state_->value;
140
141 // Initialize the algorithm state
142 state_->nfval = 0;
143 state_->ncval = 0;
144 state_->ngrad = 0;
145
146 // Compute objective value
147 alobj.update(x,UpdateType::Initial,state_->iter);
148 state_->value = alobj.getObjectiveValue(x,tol);
149 // > alobj.gradient(*state_->gradientVec,x,tol);
150
151 // ========================================================================
152 // lagmultVec, constraintVec, and gradientVec are unused state_ elements.
153 // ========================================================================
154
155 if (useDefaultScaling_) {
156 fscale_ = one/std::max(one,alobj.getObjectiveGradient(x,tol)->norm());
157 alobj.setScaling(fscale_);
158 }
159
160 unsigned numberPenalties = alobj.getNumberConstraints();
161
162 feasibilities_.resize(numberPenalties);
163
164 // Compute constraint violation
165 for (unsigned i = 0; i < numberPenalties; ++i)
166 feasibilities_[i] = alobj.feasibility(x,tol,i);
167 state_->cnorm = 0;
168 if (!feasibilities_.empty())
169 state_->cnorm = *std::max_element(feasibilities_.begin(),feasibilities_.end());
170
171 // Update evaluation counters
172 state_->ncval += alobj.getNumberConstraintEvaluations();
173 state_->nfval += alobj.getNumberFunctionEvaluations();
174 state_->ngrad += alobj.getNumberGradientEvaluations();
175
176 // Compute problem scaling
177 std::vector<Real> constraintScalings(numberPenalties);
178 Ptr<Vector<Real>> constraintVec;
179 // > TODO
180 // >if (useDefaultScaling_) {
181 // > Ptr<Vector<Real>> ji = x.clone();
182 // > // Helper function
183 // > auto getConstraintScale = [&] (const Vector<Real> &c) -> Real {
184 // > Real cscale(1);
185 // > try {
186 // > Real maxji(0), normji(0);
187 // > for (int i = 0; i < c.dimension(); ++i) {
188 // > c.applyAdjointJacobian(*ji,*c.basis(i),x,tol);
189 // > normji = ji->norm();
190 // > maxji = std::max(normji,maxji);
191 // > }
192 // > cscale = one/std::max(one,maxji);
193 // > }
194 // > catch (std::exception &e) {}
195 // > return cscale;
196 // > };
197 // > for (unsigned k = 0; k < numberPenalties; ++k) {
198 // > constraintVec = alobj.getConstraintVec(x,tol,k);
199 // > constraintScalings[k] = getConstraintScale(*constraintVec);
200 // > alobj.setScaling(constraintScalings[k],k);
201 // > }
202 // > if (!constraintScalings.empty())
203 // > cscale_ = *std::max_element(constraintScalings.begin(),constraintScalings.end());
204 // >}
205 // >else {
206 if (!useDefaultScaling_) {
207 for (unsigned i = 0; i < numberPenalties; ++i) {
208 constraintScalings[i] = cscale_;
209 alobj.setScaling(constraintScalings[i],i);
210 }
211 }
212
213 // > // Compute gradient of the lagrangian
214 // > x.axpy(-one,state_->gradientVec->dual());
215 // > proj_->project(x,outStream);
216 // > x.axpy(-one/std::min(fscale_,cscale_),*state_->iterateVec);
217 // > state_->gnorm = x.norm();
218 // > x.set(*state_->iterateVec);
219 // > TODO: problem.getSubproblemStationarityMeasure(*state_->iterateVec,x); which is used below to set tolerances
220
221 Real temp;
222
223 // Initial penalty parameters
224 if (useDefaultInitPen_) {
225 const Real oem8(1e-8), oem2(1e-2);
226 state_->searchSize = 0;
227 for (unsigned i = 0; i < numberPenalties; ++i) {
228 temp = alobj.dualNorm(x,tol,i);
229 if (temp <= oem8) temp = 1;
230 temp = std::max(oem8, oem2*std::abs(fscale_*state_->value)/temp);
231 alobj.setPenaltyParameter(temp,i);
232 state_->searchSize = std::max(state_->searchSize,temp);
233 }
234 }
235 else {
236 for (unsigned i = 0; i < numberPenalties; ++i) {
237 Real max_penalty = 0;
238 temp = list_.sublist("Step").sublist("Augmented Lagrangian").sublist(group_names_[i]).get("Initial Penalty Parameter",state_->searchSize);
239 alobj.setPenaltyParameter(temp,i);
240 max_penalty = std::max(max_penalty,temp);
241 state_->searchSize = max_penalty;
242 }
243 }
244
245 // Penalty updates
246 for (unsigned i = 0; i < numberPenalties; ++i) {
247 temp = list_.sublist("Step").sublist("Augmented Lagrangian").sublist(group_names_[i]).get("Penalty Parameter Growth Factor",penaltyUpdate_);
248 penalty_growthf_.push_back(temp);
249 }
250
251 // Subproblem stopping
252 if (useDefaultInitTol_) {
253 temp = 0;
254 for (unsigned i = 0; i < numberPenalties; ++i)
255 temp += alobj.getPenaltyParameter(i);
256 temp = std::min(minPenaltyReciprocal_,1./temp);
257 epsilon_ = std::max(subproblemTolFactor_*outerOptTolerance_, epsilon_*std::pow(temp,optDecreaseExponent_));
258 delta_ = std::max(subproblemTolFactor_*outerFeasTolerance_, delta_*std::pow(temp,optDecreaseExponent_));
259 }
260
261 alobj.reset();
262
263 if (verbosity_ > 1) {
264 outStream << std::endl;
265 outStream << "Augmented Lagrangian Initialize" << std::endl;
266 outStream << "Objective Scaling: " << fscale_ << std::endl;
267 outStream << "Maximum Constraint Scaling: " << cscale_ << std::endl;
268 outStream << "Maximum Penalty Parameter: " << state_->searchSize << std::endl;
269 outStream << std::endl;
270 }

273template<typename Real>
274void AugmentedLagrangianAlgorithm2<Real>::run( Problem<Real> &problem,
275 std::ostream &outStream ) {
276
277 // ========================================================================
278 // STEP 1: Get augmented Lagrangian objective and subproblem
279 // ========================================================================
280
281 problem.finalize(false,verbosity_>1,outStream,false);
282 Vector<Real> &x = *problem.getPrimalOptimizationVector();
283
284 auto makePenalty = [&] (Ptr<ConstraintData<Real>>& constraint_data)
285 -> Ptr<AugmentedLagrangianPenalty<Real>> {
286 Ptr<AugmentedLagrangianPenalty<Real>> penalty = makePtr<AugmentedLagrangianPenalty<Real>>(
287 constraint_data->constraint,
288 constraint_data->projection,
289 Real(1.0),
290 x.dual(),
291 *constraint_data->residual,
292 *constraint_data->multiplier,
293 hessianApprox_);
294 penalty->setMultiplier(*constraint_data->multiplier);
295 return penalty;
296 };
297
298 group_names_ = problem.getGroupNames();
299
300 Ptr<BoundConstraint<Real>> bound_constraint = problem.getBoundConstraint();
301 bool has_bound_constraint = (bound_constraint != nullPtr);
302 std::vector<std::string> ungrouped_linear_equality_constraints = problem.getUngroupedLinearEqualityConstraintNames();
303 std::vector<std::string> ungrouped_equality_constraints = problem.getUngroupedEqualityConstraintNames();
304
305 Ptr<ConstraintData<Real>> constraint_data;
306
307 // Construct augmented Lagrangian objective
308
309 std::vector<Ptr<AugmentedLagrangianPenalty<Real>>> penalties;
310 for (const std::string& name : group_names_) {
311 constraint_data = problem.getGroupConstraintData(name);
312 penalties.push_back(makePenalty(constraint_data));
313 }
314 // Penalizes the bound constraint
315 if (ungrouped_equality_constraints.size() > 0 && has_bound_constraint) {
316 Ptr<Projection<Real>> projection = makePtr<PolyhedralProjection<Real>>(bound_constraint);
317 Ptr<Vector<Real>> b = x.clone();
318 b->zero();
319 Ptr<LinearOperator<Real>> A = makePtr<IdentityOperator<Real>>();
320 Ptr<Constraint<Real>> constraint = makePtr<LinearConstraint<Real>>(A,b);
321 constraint_data = makePtr<ConstraintData<Real>>(constraint,x.dual().clone(),makePtrFromRef(x),nullPtr,projection);
322 penalties.push_back(makePenalty(constraint_data));
323 group_names_.push_back("Bounds");
324 }
325 Ptr<Objective<Real>> objective = problem.getObjective();
326 Ptr<AugmentedLagrangianObjective2<Real>> alobj = makePtr<AugmentedLagrangianObjective2<Real>>(objective,penalties,x.dual(),false);
327
328 Ptr<Problem<Real>> subproblem = makePtr<Problem<Real>>(alobj,makePtrFromRef(x));
329
330 if (ungrouped_equality_constraints.size() == 0 && has_bound_constraint)
331 subproblem->addBoundConstraint(bound_constraint);
332
333 for (const auto& name : ungrouped_linear_equality_constraints) {
334 constraint_data = problem.getConstraintData(name);
335 subproblem->addLinearConstraint(name,constraint_data->constraint,constraint_data->multiplier,constraint_data->residual);
336 }
337
338 for (const auto& name : ungrouped_equality_constraints) {
339 constraint_data = problem.getConstraintData(name);
340 subproblem->addConstraint(name,constraint_data->constraint,constraint_data->multiplier,constraint_data->residual);
341 }
342
343 if (verbosity_ > 1) {
344 outStream << std::endl << "Subproblem Finalize:" << std::endl;
345 subproblem->finalize(false,verbosity_>0,outStream,true);
346 }
347
348 // ========================================================================
349 // STEP 2: Configure
350 // ========================================================================
351
352 initialize(x,x.dual(),*alobj,outStream);
353
354 // Constants
355 const Real one(1);
356 Real tol(std::sqrt(ROL_EPSILON<Real>()));
357
358 // Additional parameters
359 unsigned maxSubproblemFails = 2;
360 int k0 = 100;
361 Real nu = 1.e6;
362 Real gamma = 0.49;
363
364 state_->gnorm = ROL_INF<Real>();
365
366 // Output
367 if (verbosity_ > 0) writeOutput(outStream,true);
368
369 // Data
370
371 unsigned numberPenalties = alobj->getNumberConstraints();
372
373 std::vector<Real> dualResiduals(numberPenalties);
374
375 std::vector<Real> dualTolerances(numberPenalties); // \tau
376 for (unsigned i = 0; i < numberPenalties; ++i)
377 dualTolerances[i] = tau0_*std::pow(theta_,alphat_);
378
379 EExitStatus statusFlag;
380 bool isSubproblemConverged = false;
381 bool isForcedUpdate = false;
382 bool tolerancesUpdated = false;
383 Real penaltyParameter, reduction, theta, oldEpsilon, oldDelta;
384
385 // ========================================================================
386 // STEP 3: Run algorithm
387 // ========================================================================
388
389 while (true) {
390 bool continueAlgorithm = status_->check(*state_);
391
392 // A zero primal-dual step does not indicate stagnation when the preceding
393 // iteration updated the penalty model. Solve the new subproblem before
394 // applying the step-tolerance stopping criterion.
395 if (!continueAlgorithm
396 && state_->statusFlag == EXITSTATUS_STEPTOL
397 && (isUpdated_ || tolerancesUpdated)
398 && state_->iter < outerIterationLimit_) {
399 continueAlgorithm = true;
400 state_->statusFlag = EXITSTATUS_LAST;
401 }
402
403 if (!continueAlgorithm) {
404 break;
405 }
406
407 // Solve augmented Lagrangian subproblem
408 list_.sublist("Status Test").set("Gradient Tolerance",epsilon_);
409 list_.sublist("Status Test").set("Constraint Tolerance",delta_);
410 //list_.sublist("Status Test").set("Step Tolerance",1.e-6*optTolerance_);
411 list_.sublist("Status Test").set("Step Tolerance",1.e-14);
412 Solver<Real> solver(subproblem,list_);
413 isSubproblemConverged = false;
414 for (unsigned i = 0; i < maxSubproblemFails; ++i) {
415 solver.reset();
416 alobj->update(x,UpdateType::Initial,state_->iter);
417 solver.solve(outStream);
418 statusFlag = solver.getAlgorithmState()->statusFlag;
419 isSubproblemConverged = (statusFlag == EXITSTATUS_CONVERGED) || (statusFlag == EXITSTATUS_USERDEFINED);
420 if (isSubproblemConverged)
421 break;
422 }
423
424 if (!isSubproblemConverged) {
425 if (verbosity_ > 0)
426 outStream << "Warning: Augmented Lagrangian subproblem failed to converge!" << std::endl;
427 }
428
429 subproblemIter_ = solver.getAlgorithmState()->iter;
430
431 // Compute step
432 state_->stepVec->set(x);
433 state_->stepVec->axpy(-one,*state_->iterateVec);
434 state_->snorm = state_->stepVec->norm();
435
436 // Update iteration information
437 state_->iter++;
438 state_->iterateVec->set(x);
439 state_->value = alobj->getObjectiveValue(x,tol);
440 for (unsigned i = 0; i < numberPenalties; ++i)
441 feasibilities_[i] = alobj->feasibility(x,tol,i);
442 if (!feasibilities_.empty())
443 state_->cnorm = *std::max_element(feasibilities_.begin(),feasibilities_.end());
444 state_->gnorm = solver.getAlgorithmState()->gnorm;
445 //alobj.update(x,UpdateType::Accept,state_->iter);
446
447 // Update evaluation counters
448 state_->nfval += alobj->getNumberFunctionEvaluations();
449 state_->ngrad += alobj->getNumberGradientEvaluations();
450 state_->ncval += alobj->getNumberConstraintEvaluations();
451
452 // ========================================================================
453 // ALESQP Updates (Algorithm 4.1)
454 // ========================================================================
455
456 for (unsigned i = 0; i < numberPenalties; ++i)
457 dualResiduals[i] = alobj->dualResidual(x,tol,i);
458
459 // Forced updates for large iterations (line 10)
460 isForcedUpdate = false;
461 if (state_->iter > k0) {
462 for (unsigned i = 0; i < numberPenalties; ++i) {
463 penaltyParameter = alobj->getPenaltyParameter(i);
464 if (alobj->getScaling(i)*dualResiduals[i] > penaltyParameter*dualTolerances[i])
465 isForcedUpdate = true;
466 }
467 }
468
469 // Update (line 13)
470 isUpdated_ = false;
471 for (unsigned i = 0; i < numberPenalties; ++i) {
472 penaltyParameter = alobj->getPenaltyParameter(i);
473 if (isForcedUpdate || alobj->getScaling(i)*dualResiduals[i] > penaltyParameter*dualTolerances[i]) {
474 penaltyParameter *= penalty_growthf_[i];
475 penaltyParameter = std::min(penaltyParameter,maxPenaltyParam_);
476 state_->searchSize = std::max(state_->searchSize,penaltyParameter);
477 theta = std::min(one/penaltyParameter,theta_);
478 dualTolerances[i] = tau0_*std::pow(theta,alphat_);
479 alobj->setPenaltyParameter(penaltyParameter,i);
480 isUpdated_ = true;
481 }
482 else {
483 theta = std::min(one/penaltyParameter,theta_);
484 dualTolerances[i] *= std::pow(theta,betat_);
485 }
486 if (alobj->getScaling(i)*dualResiduals[i] <= nu*std::pow(penaltyParameter,gamma)) {
487 state_->snorm += alobj->updateMultiplier(x,tol,i);
488 }
489 }
490
491 reduction = isUpdated_ ? Real(0.9) : Real(0.25);
492 oldEpsilon = epsilon_;
493 oldDelta = delta_;
494 epsilon_ = std::max(subproblemTolFactor_*outerOptTolerance_, reduction*epsilon_);
495 delta_ = std::max(subproblemTolFactor_*outerFeasTolerance_,reduction*delta_);
496 tolerancesUpdated = epsilon_ < oldEpsilon || delta_ < oldDelta;
497
498 alobj->reset();
499
500 // Update Output
501 if (verbosity_ > 0) writeOutput(outStream,false);
502 }
503 if (verbosity_ > 0) {
504 outStream << std::endl;
505 TypeG::Algorithm<Real>::writeExitStatus(outStream);
506 }
507}
