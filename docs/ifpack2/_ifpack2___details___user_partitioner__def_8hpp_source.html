30template <class GraphType>
31void UserPartitioner<GraphType>::
32 setPartitionParameters(Teuchos::ParameterList& List) {
33 int partCount = 0;
34 bool globalProvidedParts;
35 userProvidedParts_ = List.isParameter("partitioner: parts");
36 globalProvidedParts = List.isParameter("partitioner: global ID parts");
37 if (userProvidedParts_) partCount++;
38 if (globalProvidedParts) {
39 partCount++;
40 userProvidedParts_ = true;
41 }
42
43 userProvidedMap_ = List.isParameter("partitioner: map");
44 if (userProvidedMap_) partCount++;
45 if (partCount > 1) {
46 TEUCHOS_TEST_FOR_EXCEPTION(true, std::runtime_error,
47 "Ifpack2::UserPartitioner::setPartitionParameters: "
48 "you may specify only one of \"partitioner: parts\", \"partitioner: global ID parts\" and \"partitioner: map\","
49 " not two of them");
50 }
51 if (userProvidedMap_) {
52 map_ = List.get("partitioner: map", map_);
53 TEUCHOS_TEST_FOR_EXCEPTION(
54 map_.is_null(), std::runtime_error,
55 "Ifpack2::UserPartitioner::"
56 "setPartitionParameters: map_ is null.");
57 }
58 if (userProvidedParts_) {
59 typedef Teuchos::Array<typename Teuchos::ArrayRCP<typename GraphType::local_ordinal_type>> parts_type;
60 typedef Teuchos::Array<typename Teuchos::ArrayRCP<typename GraphType::global_ordinal_type>> gparts_type;
61 // FIXME JJH 9-Dec-2015 - This is a potentially expensive deep copy. Use an ArrayRCP<ArrayRCP<int>> instead.
62
63 // convert glboal ids to local ids before storing in Parts_
64 if (globalProvidedParts) {
65 gparts_type gparts;
66 gparts = List.get<gparts_type>("partitioner: global ID parts");
67 typedef Teuchos::RCP<Tpetra::Map<typename GraphType::local_ordinal_type, typename GraphType::global_ordinal_type, typename GraphType::node_type> const> map_type;
68 map_type OverlapMap = List.get<map_type>("OverlapRowMap");
69 this->Parts_.resize(gparts.size());
70 for (int ii = 0; ii < (int)gparts.size(); ii++) {
71 this->Parts_[ii].resize(gparts[ii].size());
72 for (int jj = 0; jj < (int)gparts[ii].size(); jj++) {
73 local_ordinal_type itmp = (int)OverlapMap->getLocalElement(gparts[ii][jj]);
74 if (itmp == Teuchos::OrdinalTraits<local_ordinal_type>::invalid()) printf("global id %d (%d,%d) not found\n", (int)gparts[ii][jj], (int)ii, (int)jj);
75 TEUCHOS_TEST_FOR_EXCEPTION(itmp == Teuchos::OrdinalTraits<local_ordinal_type>::invalid(), std::runtime_error, " \"partitioner: global ID parts\" requires that all global IDs within a block reside on the MPI rank owning the block. This can be done using overlapping Schwarz with a BLOCK_RELAXATION subdomain solver making sure that \"schwarz: overlap level\" is sufficient. Note: zeroed out Dirichlet columns will never be included in overlap parts of domains.");
76 this->Parts_[ii][jj] = itmp;
77 }
78 }
79 List.remove("partitioner: global ID parts"); // JJH I observed that iterating through the ParameterList is much more
80 // expensive when "partitioner: collection" is a Parameter object.
81 // Thus, I remove it once it's fetched from the list.
82 } else {
83 this->Parts_ = List.get<parts_type>("partitioner: parts");
84 List.remove("partitioner: parts"); // JJH I observed that iterating through the ParameterList is much more
85 // expensive when "partitioner: collection" is a Parameter object.
86 // Thus, I remove it once it's fetched from the list.
87 }
88 }
89}

91template <class GraphType>
92void UserPartitioner<GraphType>::computePartitions() {
93 if (userProvidedParts_) {
94 this->NumLocalParts_ = this->Parts_.size();
95 // The user has explicitly defined Parts_, so there is no need to for Partition_.
96 this->Partition_.resize(0);
97 } else {
98 TEUCHOS_TEST_FOR_EXCEPTION(
99 map_.is_null(), std::logic_error,
100 "Ifpack2::UserPartitioner::"
101 "computePartitions: map_ is null.");
102 const size_t localNumRows = this->Graph_->getLocalNumRows();
103 for (size_t ii = 0; ii < localNumRows; ++ii) {
104 this->Partition_[ii] = map_[ii];
105 }
106 }
