177 template <typename T>
178 Teuchos::RCP<panzer::ResponseEvaluatorFactoryBase> build() const
179 {
180 Teuchos::RCP<ResponseEvaluatorFactory_SolutionWriter<T> > ref =
181 Teuchos::rcp(new panzer_stk::ResponseEvaluatorFactory_SolutionWriter<T>(mesh));
182
183 // disable/enable the solution fields
184 ref->setAddSolutionFields(addSolutionFields_);
185
186 // disable/enable the coordinate fields
187 ref->setAddCoordinateFields(addCoordinateFields_);
188
189 // add all additional fields
190 for(std::size_t i=0;i<additionalFields_.size();i++)
191 ref->addAdditionalField(additionalFields_[i].first,additionalFields_[i].second);
192
193 for(std::size_t i=0;i<removedFields_.size();i++)
194 ref->removeField(removedFields_[i]);
195
196 // set all scaled field values
197 for(std::unordered_map<std::string,double>::const_iterator itr=fieldToScalar_.begin();
198 itr!=fieldToScalar_.end();++itr)
199 ref->scaleField(itr->first,itr->second);
200
201 return ref;
202 }
