159 template <typename EvalT>
160 Teuchos::RCP<panzer::CloneableEvaluator> buildScatter() const
161 {
162 if(!hasColProvider_)
163 return Teuchos::rcp(new ScatterResidual_BlockedTpetra<EvalT,Traits,LocalOrdinalT,GlobalOrdinalT,NodeT>(blockedDOFManager_));
164
165 // Non-square: the columns carry the derivative components, so the scatter
166 // needs the column indexers to place them.
167 return Teuchos::rcp(new ScatterResidual_BlockedTpetra<EvalT,Traits,LocalOrdinalT,GlobalOrdinalT,NodeT>(blockedDOFManager_,colGidProviders_));
168 }

181 template <typename EvalT>
182 Teuchos::RCP<panzer::CloneableEvaluator > buildGatherDomain() const
183 {
184 if(!hasColProvider_)
185 return Teuchos::rcp(new GatherSolution_BlockedTpetra<EvalT,Traits,ScalarT,LocalOrdinalT,GlobalOrdinalT,NodeT>(blockedDOFManager_));
186
187 // A blocked column side gathers into a product vector; a flat one is a
188 // single vector, which only the non-blocked gather knows how to fill.
189 if(colBlockedDOFManager_!=Teuchos::null)
190 return Teuchos::rcp(new GatherSolution_BlockedTpetra<EvalT,Traits,ScalarT,LocalOrdinalT,GlobalOrdinalT,NodeT>(colBlockedDOFManager_));
191
192 return Teuchos::rcp(new GatherSolution_Tpetra<EvalT,Traits,LocalOrdinalT,GlobalOrdinalT,NodeT>(colGidProviders_[0]));
193 }

201 template <typename EvalT>

204 if(!hasColProvider_)
205 return Teuchos::rcp(new ScatterDirichletResidual_BlockedTpetra<EvalT,Traits,LocalOrdinalT,GlobalOrdinalT,NodeT>(blockedDOFManager_));
206

208 }
