100 //
101 // x/dxdt/f are *stored* as MultiVectorType (not VectorType) so that this same container
102 // class can hold either a plain Tpetra::Vector (the traditional path) or a
103 // Tpetra::FEMultiVector (the opt-in FE-assembly path) -- Tpetra::FEMultiVector derives
104 // from Tpetra::MultiVector directly, not from Tpetra::Vector, so a Vector-typed field
105 // could not hold one.
106 //
107 // The getters come in two flavors:
108 // get_x() -- returns RCP<VectorType>, preserving the long-standing API for all
109 // existing callers. Narrows via rcp_dynamic_cast, which returns null for
110 // an unset field and throws loudly if the field actually holds an
111 // FEMultiVector (rather than silently misbehaving).
112 // get_x_mv() -- returns the stored RCP<MultiVectorType>, valid for BOTH a plain Vector
113 // and an FEMultiVector. Assembly code that must work in either mode
114 // (the gather/scatter evaluators, the LOF's ghost<->global helpers)
115 // uses this form.
116 // Setters take MultiVectorType, which accepts either kind via implicit upcast, so
117 // existing set_x(rcp_to_a_Vector) call sites are unaffected.
119 inline void set_x(const Teuchos::RCP<MultiVectorType> & in) { x = in; }
120 inline const Teuchos::RCP<VectorType> get_x() const { return Teuchos::rcp_dynamic_cast<VectorType>(x,true); }
121 inline const Teuchos::RCP<MultiVectorType> get_x_mv() const { return x; }
122
123 inline void set_dxdt(const Teuchos::RCP<MultiVectorType> & in) { dxdt = in; }
124 inline const Teuchos::RCP<VectorType> get_dxdt() const { return Teuchos::rcp_dynamic_cast<VectorType>(dxdt,true); }
125 inline const Teuchos::RCP<MultiVectorType> get_dxdt_mv() const { return dxdt; }
126
127 inline void set_f(const Teuchos::RCP<MultiVectorType> & in) { f = in; }
128 inline const Teuchos::RCP<VectorType> get_f() const { return Teuchos::rcp_dynamic_cast<VectorType>(f,true); }
129 inline const Teuchos::RCP<MultiVectorType> get_f_mv() const { return f; }
130
131 inline void set_A(const Teuchos::RCP<CrsMatrixType> & in) { A = in; }
132 inline const Teuchos::RCP<CrsMatrixType> get_A() const { return A; }

202 Teuchos::RCP<Thyra::VectorBase<ScalarT> >
203 toThyraVector(const Teuchos::RCP<MultiVectorType> & mv,
204 const Teuchos::RCP<const Thyra::VectorSpaceBase<ScalarT> > & space) const
205 {
206 if(mv==Teuchos::null) return Teuchos::null;
207
208 Teuchos::RCP<VectorType> v = Teuchos::rcp_dynamic_cast<VectorType>(mv);
209 if(v!=Teuchos::null)
210 return Thyra::createVector(v,space);
211
212 // explicitly typed: a bare Teuchos::null cannot deduce the space template parameter
213 const Teuchos::RCP<const Thyra::VectorSpaceBase<ScalarT> > deriveFromMap = Teuchos::null;
214 return Thyra::createVector(mv->getVectorNonConst(0),deriveFromMap);
215 }
