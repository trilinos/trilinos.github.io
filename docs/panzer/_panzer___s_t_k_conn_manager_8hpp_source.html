299 struct CacheKeyCompare {
300 const panzer::FieldPattern& fp_;

303 CacheKeyCompare(const panzer::FieldPattern& fp,
304 const STK_Interface* mesh,
305 const std::vector<std::string>& sidesets)
306 : fp_(fp), mesh_(mesh), sidesets_(sidesets) {}
307 bool inline operator()(const CachedEntry& entry_to_compare) const
308 {
309 const STKConnManager& cm = *entry_to_compare.second;
310 return (cm.stkMeshDB_.get() == mesh_) &&
311 (cm.sidesetsToAssociate_ == sidesets_) &&
312 fp_.equals(*entry_to_compare.first);
313 }
