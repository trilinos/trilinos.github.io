25template<typename EvalT, typename Traits>
26CrossProduct<EvalT, Traits>::
27CrossProduct(
28 const Teuchos::ParameterList& p)
29{
30 std::string result_name = p.get<std::string>("Result Name");
31 std::string vec_a_name = p.get<std::string>("Vector A Name");
32 std::string vec_b_name = p.get<std::string>("Vector B Name");
33
34 const Teuchos::RCP<const panzer::PointRule> pr =
35 p.get< Teuchos::RCP<const panzer::PointRule> >("Point Rule");
36
37 // use a scalar field only if dimension is 2D
38 useScalarField = (pr->spatial_dimension==2);
39
40 if(!useScalarField)
41 vec_a_cross_vec_b = PHX::MDField<ScalarT>(result_name, pr->dl_vector);
42 else
43 vec_a_cross_vec_b = PHX::MDField<ScalarT>(result_name, pr->dl_scalar);
44
45 vec_a = PHX::MDField<const ScalarT>(vec_a_name, pr->dl_vector);
46 vec_b = PHX::MDField<const ScalarT>(vec_b_name, pr->dl_vector);
47
48 this->addEvaluatedField(vec_a_cross_vec_b);
49 this->addDependentField(vec_a);
50 this->addDependentField(vec_b);
51
52 std::string n = "CrossProduct: " + result_name + " = " + vec_a_name + " . " + vec_b_name;
53 this->setName(n);
54}

72template<typename EvalT, typename Traits>
73void
74CrossProduct<EvalT, Traits>::
75evaluateFields(
76 typename Traits::EvalData workset)
77{
78 auto a = vec_a.get_static_view();
79 auto b = vec_b.get_static_view();
80 auto a_cross_b = vec_a_cross_vec_b.get_static_view();
81 const int l_num_pts = num_pts;
82
83 auto policy = panzer::HP::inst().teamPolicy<ScalarT,PHX::ExecutionSpace>(workset.num_cells);
84
85 if(useScalarField) {
86 Kokkos::parallel_for("panzer::CrossProduct 2D",policy,KOKKOS_LAMBDA (const Kokkos::TeamPolicy<PHX::ExecutionSpace>::member_type& team) {
87 const int cell = team.league_rank();
88 Kokkos::parallel_for(Kokkos::TeamThreadRange(team,0,l_num_pts), [&] (const int p) {
89 a_cross_b(cell,p) = a(cell,p,0)*b(cell,p,1)-a(cell,p,1)*b(cell,p,0);
90 });
91 });
92 }
93 else {
94 Kokkos::parallel_for("panzer::CrossProduct 3D",policy,KOKKOS_LAMBDA (const Kokkos::TeamPolicy<PHX::ExecutionSpace>::member_type& team) {
95 const int cell = team.league_rank();
96 Kokkos::parallel_for(Kokkos::TeamThreadRange(team,0,l_num_pts), [&] (const int p) {
97 a_cross_b(cell,p,0) = a(cell,p,1)*b(cell,p,2)-a(cell,p,2)*b(cell,p,1);
98 a_cross_b(cell,p,1) = -(a(cell,p,0)*b(cell,p,2)-a(cell,p,2)*b(cell,p,0));
99 a_cross_b(cell,p,2) = a(cell,p,0)*b(cell,p,1)-a(cell,p,1)*b(cell,p,0);
100 });
101 });
102 }
103}
