62 virtual ~CGSingleRedIterationState() = default;

64 void initialize(Teuchos::RCP<const MV> tmp, int _numVectors) {
65 using MVT = MultiVecTraits<ScalarType, MV, DM>;
66
67 TEUCHOS_ASSERT(_numVectors == 1);
68
69 // W = (AZ, R, Z)
70 W = MVT::Clone( *tmp, 3 );
71 std::vector<int> index2(2,0);
72 std::vector<int> index(1,0);
73
74 // S = (AZ, R)
75 index2[0] = 0;
76 index2[1] = 1;
77 S = MVT::CloneViewNonConst( *W, index2 );
78
79 // U = (AZ, Z)
80 index2[0] = 0;
81 index2[1] = 2;
82 U = MVT::CloneViewNonConst( *W, index2 );
83
84 index[0] = 1;
85 this->R = MVT::CloneViewNonConst( *W, index );
86 index[0] = 0;
87 AZ = MVT::CloneViewNonConst( *W, index );
88 index[0] = 2;
89 this->Z = MVT::CloneViewNonConst( *W, index );
90
91 // T = (R, Z)
92 index2[0] = 1;
93 index2[1] = 2;
94 T = MVT::CloneViewNonConst( *W, index2 );
95
96 // V = (AP, P)
97 V = MVT::Clone( *tmp, 2 );
98 index[0] = 0;
99 this->AP = MVT::CloneViewNonConst( *V, index );
100 index[0] = 1;
101 this->P = MVT::CloneViewNonConst( *V, index );
102
103 CGIterationStateBase<ScalarType, MV, DM>::initialize(tmp, _numVectors);
104 }
105
106 bool matches(Teuchos::RCP<const MV> tmp, int _numVectors=1) const {
107 return (CGIterationStateBase<ScalarType, MV, DM>::matches(tmp, _numVectors) &&
108 !W.is_null() &&
109 !V.is_null() &&
110 !U.is_null() &&
111 !S.is_null() &&
112 !T.is_null() &&
113 !AZ.is_null());
114 }

118 Teuchos::RCP<MV> U;
119 Teuchos::RCP<MV> S;
120 Teuchos::RCP<MV> T;
121 Teuchos::RCP<MV> AZ;

371 template<class ScalarType, class MV, class OP, class DM>
372 void CGSingleRedIter<ScalarType,MV,OP,DM>::initializeCG(Teuchos::RCP<CGIterationStateBase<ScalarType,MV,DM> > newstate, Teuchos::RCP<MV> R_0)
373 {
374 // Initialize the state storage if it isn't already.
375 Teuchos::RCP<const MV> lhsMV = lp_->getLHS();
376 Teuchos::RCP<const MV> rhsMV = lp_->getRHS();
377 Teuchos::RCP<const MV> tmp = ( (rhsMV!=Teuchos::null)? rhsMV: lhsMV );
378 TEUCHOS_ASSERT(!newstate.is_null());
379 if (!Teuchos::rcp_dynamic_cast<CGSingleRedIterationState<ScalarType,MV,DM> >(newstate, true)->matches(tmp, 1))
380 newstate->initialize(tmp, 1);
381 setState(newstate);
382
383 std::string errstr("Belos::CGSingleRedIter::initialize(): Specified multivectors must have a consistent length and width.");
384
385 {
386
387 TEUCHOS_TEST_FOR_EXCEPTION( MVT::GetGlobalLength(*newstate->R) != MVT::GetGlobalLength(*R_),
388 std::invalid_argument, errstr );
389 TEUCHOS_TEST_FOR_EXCEPTION( MVT::GetNumberVecs(*newstate->R) != 1,
390 std::invalid_argument, errstr );
391
392 // Copy basis vectors from newstate into V
393 if (R_0 != R_) {
394 // copy over the initial residual (unpreconditioned).
395 MVT::Assign( *R_0, *R_ );
396 }
397
398 // Compute initial direction vectors
399 // Initially, they are set to the preconditioned residuals
400 //
401 if ( lp_->getLeftPrec() != Teuchos::null ) {
402 lp_->applyLeftPrec( *R_, *Z_ );
403 if ( lp_->getRightPrec() != Teuchos::null ) {
404 Teuchos::RCP<MV> tmp2 = MVT::Clone( *Z_, 1 );
405 lp_->applyRightPrec( *Z_, *tmp2 );
406 MVT::Assign( *tmp2, *Z_ );
407 }
408 }
409 else if ( lp_->getRightPrec() != Teuchos::null ) {
410 lp_->applyRightPrec( *R_, *Z_ );
411 }
412 else {
413 MVT::Assign( *R_, *Z_ );
414 }
415
416 // Multiply the current preconditioned residual vector by A and store in AZ_
417 lp_->applyOp( *Z_, *AZ_ );
418
419 // P_ := Z_
420 // Logically, AP_ := AZ_
421 MVT::Assign( *U_, *V_);
422 }
423
424 // The solver is initialized
425 initialized_ = true;
426 }

431 template<class ScalarType, class MV, class OP, class DM>
432 Teuchos::RCP<const MV>
433 CGSingleRedIter<ScalarType,MV,OP,DM>::getNativeResiduals( std::vector<MagnitudeType> *norms ) const {
434 if (convTest_->getResNormType() == Belos::PreconditionerNorm) {
435 (*norms)[0] = std::sqrt(Teuchos::ScalarTraits<ScalarType>::magnitude(rHz_));
436 return Teuchos::null;
437 } else if (foldConvergenceDetectionIntoAllreduce_ && convTest_->getResNormType() == Belos::TwoNorm) {
438 (*norms)[0] = std::sqrt(Teuchos::ScalarTraits<ScalarType>::magnitude(rHr_));
439 return Teuchos::null;
440 } else
441 return R_;
442 }

446 // Iterate until the status test informs us we should stop.
447 template<class ScalarType, class MV, class OP, class DM>
448 void CGSingleRedIter<ScalarType,MV,OP,DM>::iterate()
449 {
450 //
451 // Allocate/initialize data structures
452 //
453 if (!initialized_) {
454 initialize();
455 }
456
457 // Allocate memory for scalars.
458 Teuchos::RCP<DM> sHz = DMT::Create( 2, 1 );
459 Teuchos::RCP<DM> sHt = DMT::Create( 2, 2 );
460 ScalarType rHz_old, alpha, beta, delta;
461
462 // Create convenience variables for zero and one.
463 const ScalarType one = Teuchos::ScalarTraits<ScalarType>::one();
464 const MagnitudeType zero = Teuchos::ScalarTraits<MagnitudeType>::zero();
465
466 // Get the current solution vector.
467 Teuchos::RCP<MV> cur_soln_vec = lp_->getCurrLHSVec();
468
469 // Check that the current solution vector only has one column.
470 TEUCHOS_TEST_FOR_EXCEPTION( MVT::GetNumberVecs(*cur_soln_vec) != 1, CGIterateFailure,
471 "Belos::CGSingleRedIter::iterate(): current linear system has more than one vector!" );
472
473 if (foldConvergenceDetectionIntoAllreduce_ && convTest_->getResNormType() == Belos::TwoNorm) {
474 // Compute first <S_,T_> a.k.a. <R_,Z_>, <AZ_,Z_> and <R_,R_> combined (also computes unneeded <AZ_,R_>)
475 MVT::MvTransMv( one, *S_, *T_, *sHt );
476 DMT::SyncDeviceToHost( *sHt );
477 rHz_ = DMT::ValueConst(*sHt,1,1);
478 delta = DMT::ValueConst(*sHt,0,1);
479 rHr_ = DMT::ValueConst(*sHt,1,0);
480 } else {
481 // Compute first <s,z> a.k.a. <r,z> and <Az,z> combined
482 MVT::MvTransMv( one, *S_, *Z_, *sHz );
483 DMT::SyncDeviceToHost( *sHz );
484 rHz_ = DMT::ValueConst(*sHz,1,0);
485 delta = DMT::ValueConst(*sHz,0,0);
486 }
487 if ((Teuchos::ScalarTraits<ScalarType>::magnitude(delta) < Teuchos::ScalarTraits<ScalarType>::eps()) &&
488 (stest_->checkStatus(this) == Passed))
489 return;
490 alpha = rHz_ / delta;
491
492 // Check that alpha is a positive number!
493 TEUCHOS_TEST_FOR_EXCEPTION( SCT::real(alpha) <= zero, CGPositiveDefiniteFailure,
494 "Belos::CGSingleRedIter::iterate(): non-positive value for p^H*A*p encountered!" );
495
497 // Iterate until the status test tells us to stop.
498 //
499 if (foldConvergenceDetectionIntoAllreduce_ && convTest_->getResNormType() == Belos::TwoNorm) {
501 // Iterate until the status test tells us to stop.
502 //
503 while (true) {
504
505 // Update the solution vector x := x + alpha * P_
506 //
507 MVT::MvAddMv( one, *cur_soln_vec, alpha, *P_, *cur_soln_vec );
508 //
509 // Compute the new residual R_ := R_ - alpha * AP_
510 //
511 MVT::MvAddMv( one, *R_, -alpha, *AP_, *R_ );
512 //
513 // Apply preconditioner to new residual to update Z_
514 //
515 if ( lp_->getLeftPrec() != Teuchos::null ) {
516 lp_->applyLeftPrec( *R_, *Z_ );
517 if ( lp_->getRightPrec() != Teuchos::null ) {
518 Teuchos::RCP<MV> tmp = MVT::CloneCopy( *Z_ );
519 lp_->applyRightPrec( *tmp, *Z_ );
520 }
521 }
522 else if ( lp_->getRightPrec() != Teuchos::null ) {
523 lp_->applyRightPrec( *R_, *Z_ );
524 }
525 else {
526 MVT::Assign( *R_, *Z_ );
527 }
528 //
529 // Multiply the current preconditioned residual vector by A and store in AZ_
530 lp_->applyOp( *Z_, *AZ_ );
531 //
532 // Compute <S_,T_> a.k.a. <R_,Z_>, <AZ_,Z_> and <R_,R_> combined (also computes unneeded <AZ_,R_>)
533 MVT::MvTransMv( one, *S_, *T_, *sHt );
534 DMT::SyncDeviceToHost( *sHt );
535 //
536 // Update scalars.
537 rHz_old = rHz_;
538 rHz_ = DMT::ValueConst(*sHt,1,1);
539 delta = DMT::ValueConst(*sHt,0,1);
540 rHr_ = DMT::ValueConst(*sHt,1,0);
541
542 // Increment the iteration
543 iter_++;
544 //
545 // Check the status test, now that the solution and residual have been updated
546 //
547 if (stest_->checkStatus(this) == Passed) {
548 break;
549 }
550 //
551 beta = rHz_ / rHz_old;
552 alpha = rHz_ / (delta - (beta*rHz_ / alpha));
553 //
554 // Check that alpha is a positive number!
555 TEUCHOS_TEST_FOR_EXCEPTION( SCT::real(alpha) <= zero, CGPositiveDefiniteFailure,
556 "Belos::CGSingleRedIter::iterate(): non-positive value for p^H*A*p encountered!" );
557
558 //
559 // Update the direction vector P_ := Z_ + beta * P_
560 // Update AP_ through recurrence relation AP_ := AZ_ + beta * AP_
561 // Hence: V_ = (AP_, P_) := (AZ_, Z_) + beta (AP_, P_) = U_ + beta * V_
562 //
563 MVT::MvAddMv( one, *U_, beta, *V_, *V_ );
564
565 } // end while (1)
566 } else {
568 // Iterate until the status test tells us to stop.
569 //
570 while (true) {
571
572 // Update the solution vector x := x + alpha * P_
573 //
574 MVT::MvAddMv( one, *cur_soln_vec, alpha, *P_, *cur_soln_vec );
575 //
576 // Compute the new residual R_ := R_ - alpha * AP_
577 //
578 MVT::MvAddMv( one, *R_, -alpha, *AP_, *R_ );
579
580 // Increment the iteration
581 iter_++;
582 //
583 // Check the status test, now that the solution and residual have been updated
584 //
585 if (stest_->checkStatus(this) == Passed) {
586 break;
587 }
588 //
589 // Apply preconditioner to new residual to update Z_
590 //
591 if ( lp_->getLeftPrec() != Teuchos::null ) {
592 lp_->applyLeftPrec( *R_, *Z_ );
593 if ( lp_->getRightPrec() != Teuchos::null ) {
594 Teuchos::RCP<MV> tmp = MVT::CloneCopy( *Z_ );
595 lp_->applyRightPrec( *tmp, *Z_ );
596 }
597 }
598 else if ( lp_->getRightPrec() != Teuchos::null ) {
599 lp_->applyRightPrec( *R_, *Z_ );
600 }
601 else {
602 MVT::Assign( *R_, *Z_ );
603 }
604 //
605 // Multiply the current preconditioned residual vector by A and store in AZ_
606 lp_->applyOp( *Z_, *AZ_ );
607 //
608 // Compute <S_,Z_> a.k.a. <R_,Z_> and <AZ_,Z_> combined
609 MVT::MvTransMv( one, *S_, *Z_, *sHz );
610 DMT::SyncDeviceToHost( *sHz );
611 //
612 // Update scalars.
613 rHz_old = rHz_;
614 rHz_ = DMT::ValueConst(*sHz,1,0);
615 delta = DMT::ValueConst(*sHz,0,0);
616 //
617 beta = rHz_ / rHz_old;
618 alpha = rHz_ / (delta - (beta*rHz_ / alpha));
619 //
620 // Check that alpha is a positive number!
621 TEUCHOS_TEST_FOR_EXCEPTION( SCT::real(alpha) <= zero, CGPositiveDefiniteFailure,
622 "Belos::CGSingleRedIter::iterate(): non-positive value for p^H*A*p encountered!" );
623
624 //
625 // Update the direction vector P_ := Z_ + beta * P_
626 // Update AP_ through recurrence relation AP_ := AZ_ + beta * AP_
627 // Hence: V_ = (AP_, P_) := (AZ_, Z_) + beta (AP_, P_) = U_ + beta * V_
628 //
629 MVT::MvAddMv( one, *U_, beta, *V_, *V_ );
630
631 } // end while (1)
632 }
633 }
