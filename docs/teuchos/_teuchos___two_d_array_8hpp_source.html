252 size_type colLimit = (numberOfCols < _numCols ? numberOfCols : _numCols);
253 for(size_type i = 0; i<_numRows; i++){
254 for(size_type j = 0; j<colLimit; j++){

279template<class T>
280TwoDArray<T> TwoDArray<T>::fromString(const std::string& string_in){
281 std::string curString = string_in;
282 std::string metaSeperator = TwoDArray<T>::getMetaSeperator();
283 size_t curPos = curString.find(metaSeperator);
284 std::string dimString = curString.substr(0, curPos);
285 curString = curString.substr(curPos+1);
286
287 //process dimensions
288 size_t dimCharPos =
289 dimString.find(TwoDArray<T>::getDimensionsDelimiter());
290 std::istringstream numRowsStream(dimString.substr(0,dimCharPos));
291 std::istringstream numColsStream(dimString.substr(dimCharPos+1));
292 size_t numRows, numCols;
293 numRowsStream >> numRows;
294 numColsStream >> numCols;
295
296 //determine symetry state
297 bool symmetrical = false;
298 curPos = curString.find(metaSeperator);
299 if(curPos != std::string::npos){
300 symmetrical = true;
301 curString = curString.substr(curPos+1);
302 }
303
304 //Get actual data
305 Array<T> array = fromStringToArray<T>(curString);
306
307 // Check for overflow in numRows*numCols
308 const size_t maxSize =
309 static_cast<size_t>(std::numeric_limits<typename Array<T>::size_type>::max());
310 TEUCHOS_TEST_FOR_EXCEPTION(numRows != 0 && numCols > maxSize / numRows,
311 InvalidArrayStringRepresentation,
312 "Error: TwoDArray dimensions " << numRows << "x" << numCols
313 << " exceed the maximum array size of " << maxSize << "."
314 )
315 typename Array<T>::size_type expectedSize = numRows * numCols;
316 TEUCHOS_TEST_FOR_EXCEPTION(array.size() != expectedSize,
317 InvalidArrayStringRepresentation,
318 "Error: You've specified an TwoDArray as having the dimensions of "
319 << numRows << "x" << numCols << ". This means you should have " <<
320 expectedSize << " entries specified in your array. However you "
321 "only specified " << array.size() << " entries."
322 )
323
324 //Construct object to return
325 TwoDArray<T> toReturn(numRows, numCols, array);
326 toReturn.setSymmetrical(symmetrical);
327 return toReturn;
328}

358template<typename T>
359bool symmetricCompare(const TwoDArray<T> &a1, const TwoDArray<T> &a2 ){
360 if(a1.getNumRows() != a2.getNumRows() ||
361 a1.getNumRows() != a2.getNumRows())
362 {
363 return false;
364 }
365 else{
366 typedef typename TwoDArray<T>::size_type ST;
367 for(ST i=0;i<a1.getNumRows(); ++i){
368 for(ST j=0;j<a1.getNumCols()-a1.getNumRows()+i; ++j){
369 if(a1(i,j) != a2(i,j)){
370 return false;
371 }
372 }
373 }
374 return true;
375 }
376}

381/* \brief Returns true of the two TwoDArrays have the same contents and
382 * their dimensions are the same.
383 *
384 * \note If the arrays are symmetrical, only the values in the upper half
385 * of the array are compared. For example: in a 4x4 array, only the values
386 * indicated with x's in the figure below would be compared.
387 *
388 * o o o o
389 * x o o o
390 * x x o o
391 * x x x o
392 *
393 * \relates TwoDArray
394 */
395template<typename T>
396bool operator==( const TwoDArray<T> &a1, const TwoDArray<T> &a2 ){
397 if(a1.isSymmetrical() != a2.isSymmetrical()){
398 return false;
399 }
400 if(a1.isSymmetrical()){
401 return TwoDDetails::symmetricCompare(a1,a2);
402 }
403 else{
404 return a1.getDataArray() == a2.getDataArray() &&
405 a1.getNumRows() == a2.getNumRows() &&
406 a1.getNumCols() == a2.getNumCols();
407 }
