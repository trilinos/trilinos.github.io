171 virtual std::pair<bool, std::vector<size_t> >
172 read (std::istream& in,
173 const size_t startingLineNumber,
174 const bool tolerant,
175 const bool debug = false)
176 {
177 (void) debug; // silence unused input argument warning
178 TEUCHOS_TEST_FOR_EXCEPTION(! in, std::invalid_argument,
179 "Input stream is invalid.");
180
181 std::string line;
182 size_t lineNumber = startingLineNumber;
183 bool allSucceeded = true;
184 std::vector<size_t> badLineNumbers;
185 while (getline (in, line)) {
186 size_t start, size;
187 if (checkCommentLine (line, start, size, lineNumber, tolerant)) {
188 ++lineNumber;
189 continue; // it's a comment line
190 }
191 const std::string theLine = line.substr (start, size);
192
193 const bool localSuccess = readLine (theLine, lineNumber, tolerant);
194 ++lineNumber;
195 allSucceeded = allSucceeded && localSuccess;
196 if (! localSuccess) {
197 badLineNumbers.push_back (lineNumber);
198 }
199 }
200 return std::make_pair (allSucceeded, badLineNumbers);
201 }

226 std::pair<Teuchos::Tuple<Ordinal, 3>, bool>
227 readDimensions (std::istream& in,
228 size_t& lineNumber,
229 const bool tolerant = false)
230 {
231 Teuchos::Tuple<Ordinal, 3> dims;
232 // Fill in (numRows, numCols, numNonzeros) with reasonable
233 // defaults. If we don't succeed in reading all the data
234 // from the current line of the input stream, the remaining
235 // values not read will be these default values.
236 dims[0] = 0;
237 dims[1] = 0;
238 dims[2] = 0;
239
240 // Keep reading lines from the input stream until we find a
241 // non-comment line, or until we run out of lines. The latter
242 // is an error, since every "coordinate" format Matrix Market
243 // file must have a dimensions line after the banner (even if
244 // the matrix has zero rows or columns, or zero entries).
245 std::string line;
246 bool commentLine = true;
247 while (commentLine) {
248 // Is it even valid to read from the input stream?
249 if (in.eof() || in.fail()) {
250 if (tolerant) {
251 return std::make_pair (dims, false);
252 }
253 else {
254 std::ostringstream os;
255 os << "Unable to get coordinate dimensions line (at all) "
256 "from (line " << lineNumber << ") of input stream; the "
257 "input stream claims that it is at \"end-of-file\" or has "
258 "an otherwise \"fail\"ed state.";
259 throw std::invalid_argument(os.str());
260 }
261 }
262 // Try to get the next line from the input stream.
263 if (getline(in, line)) {
264 lineNumber++; // We did actually read a line
265 }
266 else {
267 if (tolerant) {
268 return std::make_pair (dims, false);
269 }
270 else {
271 std::ostringstream os;
272 os << "Failed to read coordinate dimensions line (at all) "
273 "from (line " << lineNumber << " from input stream. The "
274 "line should contain the coordinate matrix dimensions in "
275 << " the form \"<numRows> <numCols> <numNonzeros>\".";
276 throw std::invalid_argument (os.str());
277 }
278 }
279 // Is the current line a comment line? Ignore start and
280 // size; they are only useful for reading the actual matrix
281 // entries. (We could use them here as an optimization, but
282 // we've chosen not to.)
283 size_t start = 0, size = 0;
284 commentLine = checkCommentLine (line, start, size,
285 lineNumber, tolerant);
286 }
287 //
288 // Read in <numRows> <numCols> <numNonzeros> from input line
289 //
290 std::istringstream istr (line);
291 // Does line contain anything at all? Can we safely read from
292 // the input stream wrapping the line?
293 if (istr.eof() || istr.fail()) {
294 if (tolerant) {
295 return std::make_pair (dims, false);
296 }
297 std::ostringstream os;
298 os << "Unable to read any data from line " << lineNumber
299 << " of input; the line should contain the coordinate matrix "
300 << "dimensions \"<numRows> <numCols> <numNonzeros>\".";
301 throw std::invalid_argument(os.str());
302 }
303 // Read in <numRows>
304 {
305 Ordinal theNumRows = 0;
306 istr >> theNumRows;
307 if (istr.fail()) {
308 if (tolerant) {
309 return std::make_pair (dims, false);
310 }
311 std::ostringstream os;
312 os << "Failed to get number of rows from line " << lineNumber
313 << " of input; the line should contain the coordinate matrix "
314 << " dimensions \"<numRows> <numCols> <numNonzeros>\".";
315 throw std::invalid_argument(os.str());
316 }
317 else { // Capture the validly read result before checking for eof.
318 dims[0] = theNumRows;
319 }
320 }
321 // There should be two more things to read.
322 if (istr.eof()) {
323 if (tolerant) {
324 return std::make_pair (dims, false);
325 }
326 std::ostringstream os;
327 os << "No more data after number of rows on line " << lineNumber
328 << " of input; the line should contain the coordinate matrix "
329 << " dimensions \"<numRows> <numCols> <numNonzeros>\".";
330 throw std::invalid_argument(os.str());
331 }
332 // Read in <numCols>
333 {
334 Ordinal theNumCols = 0;
335 istr >> theNumCols;
336 if (istr.fail()) {
337 if (tolerant) {
338 return std::make_pair (dims, false);
339 }
340 std::ostringstream os;
341 os << "Failed to get number of columns from line " << lineNumber
342 << " of input; the line should contain the coordinate matrix "
343 << " dimensions \"<numRows> <numCols> <numNonzeros>\".";
344 throw std::invalid_argument(os.str());
345 }
346 else { // Capture the validly read result before checking for eof.
347 dims[1] = theNumCols;
348 }
349 }
350 // There should be one more thing to read.
351 if (istr.eof()) {
352 if (tolerant) {
353 return std::make_pair (dims, false);
354 }
355 std::ostringstream os;
356 os << "No more data after number of columns on line " << lineNumber
357 << " of input; the line should contain the coordinate matrix "
358 << " dimensions \"<numRows> <numCols> <numNonzeros>\".";
359 throw std::invalid_argument(os.str());
360 }
361 // Read in <numNonzeros>
362 {
363 Ordinal theNumNonzeros = 0;
364 istr >> theNumNonzeros;
365 if (istr.fail()) {
366 if (tolerant) {
367 return std::make_pair (dims, false);
368 }
369 std::ostringstream os;
370 os << "Failed to get number of (structural) nonzeros from line "
371 << lineNumber
372 << " of input; the line should contain the coordinate matrix "
373 << " dimensions \"<numRows> <numCols> <numNonzeros>\".";
374 throw std::invalid_argument(os.str());
375 }
376 else { // Capture the validly read result
377 dims[2] = theNumNonzeros;
378 }
379 }
380 // Validate that dimensions are non-negative. Only signed
381 // Ordinal types can be negative; skipping the test for
382 // unsigned types avoids tautological-comparison warnings.
383 if constexpr (std::is_signed_v<Ordinal>) {
384 if (dims[0] < 0 || dims[1] < 0 || dims[2] < 0) {
385 if (tolerant) {
386 return std::make_pair (dims, false);
387 }
388 std::ostringstream os;
389 os << "Invalid Matrix Market dimensions on line " << lineNumber
390 << ": dimensions must be non-negative, got "
391 << dims[0] << " " << dims[1] << " " << dims[2];
392 throw std::invalid_argument(os.str());
393 }
394 }
395 return std::make_pair (dims, true);
396 }

489 bool
490 readLine (const std::string& theLine,
491 const size_t lineNumber,
492 const bool tolerant)
493 {
494 typedef Teuchos::ScalarTraits<Scalar> STS;
495 typedef typename STS::magnitudeType Real;
496
497 Ordinal rowIndex;
498 Ordinal colIndex;
499 Scalar value;
500
501 Real realPart, imagPart;
502 const bool localSuccess =
503 readComplexLine (theLine, rowIndex, colIndex, realPart, imagPart,
504 lineNumber, tolerant);
505 if (localSuccess) {
506 // Assume that assignment from std::complex<Real> to Scalar
507 // (which itself is complex-valued) is valid. We have to do
508 // this, since the C++ compiler may not be smart enough to
509 // assume here (when it instantiates the templates) that
510 // Scalar is an std::complex<Real> -- even though it has to
511 // be, if STS::isComplex is true (which as of 31 Jan 2011,
512 // only holds for std::complex<T>).
513 value = std::complex<Real> (realPart, imagPart);
514
515 // Now that we've read in the (i, j, A_ij) triple
516 // successfully, we can add the entry to the sparse matrix.
517 (*(this->adder_)) (rowIndex, colIndex, value);
518 }
519 return localSuccess;
520 }
521 };
522#endif // HAVE_TEUCHOS_COMPLEX
