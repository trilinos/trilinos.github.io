79template<typename Scalar>
80bool Amesos2LinearOpWithSolveFactory<Scalar>::isCompatible(
81 const LinearOpSourceBase<Scalar> &fwdOpSrc
82 ) const
83{
84 Teuchos::RCP<const LinearOpBase<Scalar> >
85 fwdOp = fwdOpSrc.getOp();
86 auto tpetraFwdOp = ConverterT::getConstTpetraOperator(fwdOp);
87 if ( ! dynamic_cast<const MAT * >(&*tpetraFwdOp) )
88 return false;
89 return true;

99template<typename Scalar>
100void Amesos2LinearOpWithSolveFactory<Scalar>::initializeOp(
101 const RCP<const LinearOpSourceBase<Scalar> > &fwdOpSrc,
102 LinearOpWithSolveBase<Scalar> *Op,
103 const ESupportSolveUse /* supportSolveUse */
104 ) const
105{
106 THYRA_FUNC_TIME_MONITOR("Stratimikos: Amesos2LOWSF");
107
108 TEUCHOS_TEST_FOR_EXCEPT(Op==NULL);
109 TEUCHOS_TEST_FOR_EXCEPT(fwdOpSrc.get()==NULL);
110 TEUCHOS_TEST_FOR_EXCEPT(fwdOpSrc->getOp().get()==NULL);
111 RCP<const LinearOpBase<Scalar> > fwdOp = fwdOpSrc->getOp();
112
113 Teuchos::RCP<Teuchos::FancyOStream>
114 out = Teuchos::VerboseObjectBase::getDefaultOStream();
115
116 //
117 // Unwrap and get the forward Tpetra::Operator object
118 //
119 auto tpetraFwdOp = ConverterT::getConstTpetraOperator(fwdOp);
120 auto tpetraCrsMat = Teuchos::rcp_dynamic_cast<const MAT>(tpetraFwdOp);
121 // Get the Amesos2LinearOpWithSolve object
122 Amesos2LinearOpWithSolve<Scalar>
123 *amesos2Op = &Teuchos::dyn_cast<Amesos2LinearOpWithSolve<Scalar>>(*Op);
124
125 //
126 // Determine if we must start over or not
127 //
128 bool startOver = ( amesos2Op->get_amesos2Solver()==Teuchos::null );
129 if (!startOver) {
130 auto oldTpetraFwdOp = ConverterT::getConstTpetraOperator(amesos2Op->get_fwdOp());
131 startOver =
132 (
133 tpetraFwdOp.get() != oldTpetraFwdOp.get()
134 // Assuming that, like Amesos, Amesos2 must start over if the matrix changes
135 );
136 }
137 //
138 // Update the amesos2 solver
139 //
140 if (startOver) {
141 //
142 // This LOWS object has not be initialized yet or is not compatible with the existing
143 //
144 // so this is where we setup everything from the ground up.
145
146 // Create the concrete solver
147 Teuchos::RCP<Solver> amesos2Solver;
148 {
149 THYRA_FUNC_TIME_MONITOR_DIFF("Stratimikos: Amesos2LOWSF:InitConstruct",
150 InitConstruct);
151 switch(solverType_) {
152 case Thyra::Amesos2::KLU2:
153 amesos2Solver = ::Amesos2::create<MAT,MV>("klu2", tpetraCrsMat);
154 break;
155#ifdef HAVE_AMESOS2_LAPACK
156 case Thyra::Amesos2::LAPACK:
157 amesos2Solver = ::Amesos2::create<MAT,MV>("lapack", tpetraCrsMat);
158 break;
159#endif
160#ifdef HAVE_AMESOS2_SUPERLU
161 case Thyra::Amesos2::SUPERLU:
162 amesos2Solver = ::Amesos2::create<MAT,MV>("superlu", tpetraCrsMat);
163 break;
164#endif
165#ifdef HAVE_AMESOS2_SUPERLUMT
166 case Thyra::Amesos2::SUPERLUMT:
167 amesos2Solver = ::Amesos2::create<MAT,MV>("superlumt", tpetraCrsMat);
168 break;
169#endif
170#ifdef HAVE_AMESOS2_SUPERLUDIST
171 case Thyra::Amesos2::SUPERLUDIST:
172 amesos2Solver = ::Amesos2::create<MAT,MV>("superludist", tpetraCrsMat);
173 break;
174# endif
175#ifdef HAVE_AMESOS2_PARDISO_MKL
176 case Thyra::Amesos2::PARDISO_MKL:
177 amesos2Solver = ::Amesos2::create<MAT,MV>("pardiso_mkl", tpetraCrsMat);
178 break;
179#endif
180#ifdef HAVE_AMESOS2_CSS_MKL
181 case Thyra::Amesos2::CSS_MKL:
182 amesos2Solver = ::Amesos2::create<MAT,MV>("css_mkl", tpetraCrsMat);
183 break;
184#endif
185#ifdef HAVE_AMESOS2_CHOLMOD
186 case Thyra::Amesos2::CHOLMOD:
187 amesos2Solver = ::Amesos2::create<MAT,MV>("cholmod", tpetraCrsMat);
188 break;
189#endif
190#ifdef HAVE_AMESOS2_BASKER
191 case Thyra::Amesos2::BASKER:
192 amesos2Solver = ::Amesos2::create<MAT,MV>("basker", tpetraCrsMat);
193 break;
194#endif
195#ifdef HAVE_AMESOS2_MUMPS
196 case Thyra::Amesos2::MUMPS:
197 amesos2Solver = ::Amesos2::create<MAT,MV>("mumps", tpetraCrsMat);
198 break;
199#endif
200#ifdef HAVE_AMESOS2_UMFPACK
201 case Thyra::Amesos2::UMFPACK:
202 amesos2Solver = ::Amesos2::create<MAT,MV>("umfpack", tpetraCrsMat);
203 break;
204#endif
205#ifdef HAVE_AMESOS2_SHYLU_NODETACHO
206 case Thyra::Amesos2::TACHO:
207 amesos2Solver = ::Amesos2::create<MAT,MV>("tacho", tpetraCrsMat);
208 break;
209#endif
210#ifdef HAVE_AMESOS2_STRUMPACK
211 case Thyra::Amesos2::STRUMPACK:
212 amesos2Solver = ::Amesos2::create<MAT,MV>("strumpack", tpetraCrsMat);
213 break;
214#endif
215#ifdef HAVE_AMESOS2_CUSOLVER
216 case Thyra::Amesos2::CUSOLVER:
217 amesos2Solver = ::Amesos2::create<MAT,MV>("cusolver", tpetraCrsMat);
218 break;
219#endif
220 default:
221 TEUCHOS_TEST_FOR_EXCEPTION(
222 true, std::logic_error
223 ,"Error, the solver type ID = " << solverType_ << " is invalid!"
224 );
225 }
226 }
227
228 // Extract and set Amesos2 Parameters
229 if( paramList_->isSublist(Amesos2_Settings_name) ){
230 auto amesos2Params = Teuchos::rcp(new Teuchos::ParameterList(paramList_->sublist(Amesos2_Settings_name)));
231 amesos2Params->setName("Amesos2");
232 amesos2Solver->setParameters(amesos2Params);
233 } else {
234 auto amesos2Params = Teuchos::rcp(new Teuchos::ParameterList());
235 amesos2Params->setName("Amesos2");
236 amesos2Params->sublist(amesos2Solver->name());
237 amesos2Solver->setParameters(amesos2Params);
238 }
239
240 // Do the initial factorization
241 {
242 THYRA_FUNC_TIME_MONITOR_DIFF("Stratimikos: Amesos2LOWSF:Symbolic", Symbolic);
243 amesos2Solver->symbolicFactorization();
244 }
245 {
246 THYRA_FUNC_TIME_MONITOR_DIFF("Stratimikos: Amesos2LOWSF:Factor", Factor);
247 amesos2Solver->numericFactorization();
248 }
249
250 // Initialize the LOWS object and we are done!
251 amesos2Op->initialize(fwdOp,fwdOpSrc,amesos2Solver);
252 }
253 else {
254 //
255 // This LOWS object has already be initialized once so we must just reset
256 // the matrix and refactor it.
257 auto amesos2Solver = amesos2Op->get_amesos2Solver();
258
259 // set
260 amesos2Solver->setA(tpetraCrsMat);
261
262 // Do the initial factorization
263 if(refactorizationPolicy_ == Amesos2::REPIVOT_ON_REFACTORIZATION) {
264 THYRA_FUNC_TIME_MONITOR_DIFF("Stratimikos: Amesos2LOWSF:Symbolic", Symbolic);
265 amesos2Solver->symbolicFactorization();
266 }
267 {
268 THYRA_FUNC_TIME_MONITOR_DIFF("Stratimikos: Amesos2LOWSF::Factor", Factor);
269 amesos2Solver->numericFactorization();
270 }
271
272 // Initialize the LOWS object and we are done!
273 amesos2Op->initialize(fwdOp,fwdOpSrc,amesos2Solver);
274 }
275 amesos2Op->setOStream(this->getOStream());
276 amesos2Op->setVerbLevel(this->getVerbLevel());
277}

285template<typename Scalar>
286void Amesos2LinearOpWithSolveFactory<Scalar>::initializePreconditionedOp(
287 const RCP<const LinearOpSourceBase<Scalar> > &fwdOpSrc,
288 const RCP<const PreconditionerBase<Scalar> > &/* prec */,
289 LinearOpWithSolveBase<Scalar> *Op,
290 const ESupportSolveUse supportSolveUse
291 ) const
292{
293 TEUCHOS_TEST_FOR_EXCEPTION(
294 this->throwOnPrecInput_, std::logic_error,
295 "Error, the concrete implementation described as \'"<<this->description()
296 <<"\' does not support preconditioners"
297 " and has been configured to throw this exception when the"
298 " initializePreconditionedOp(...) function is called!"
299 );
300 this->initializeOp(fwdOpSrc,Op,supportSolveUse); // Ignore the preconditioner!
301}

321template<typename Scalar>
322void Amesos2LinearOpWithSolveFactory<Scalar>::uninitializeOp(
323 LinearOpWithSolveBase<Scalar> *Op,
324 RCP<const LinearOpSourceBase<Scalar> > *fwdOpSrc,
325 RCP<const PreconditionerBase<Scalar> > *prec,
326 RCP<const LinearOpSourceBase<Scalar> > *approxFwdOpSrc,
327 ESupportSolveUse * /* supportSolveUse */
328 ) const
329{
330#ifdef TEUCHOS_DEBUG
331 TEUCHOS_TEST_FOR_EXCEPT(Op==NULL);
332#endif
333 Amesos2LinearOpWithSolve<Scalar>
334 *amesos2Op = &Teuchos::dyn_cast<Amesos2LinearOpWithSolve<Scalar>>(*Op);
335 RCP<const LinearOpSourceBase<Scalar> >
336 _fwdOpSrc = amesos2Op->extract_fwdOpSrc(); // Will be null if uninitialized!
337 if(fwdOpSrc) *fwdOpSrc = _fwdOpSrc; // It is fine if the client does not want this object back!
338 if(prec) *prec = Teuchos::null; // We never keep a preconditioner!
339 if(approxFwdOpSrc) *approxFwdOpSrc = Teuchos::null; // never keep approx fwd op!
340}

418template<typename Scalar>
419RCP<const Teuchos::ParameterList>
420Amesos2LinearOpWithSolveFactory<Scalar>::generateAndGetValidParameters()
421{
422 static RCP<Teuchos::ParameterList> validParamList;
423 if (validParamList.get()==NULL) {
424 validParamList = Teuchos::rcp(new Teuchos::ParameterList("Amesos2"));
425 Teuchos::Array<std::string> solverTypeNames(Amesos2::numSolverTypes);
426 for (int k = 0; k<Amesos2::numSolverTypes; ++k)
427 solverTypeNames[k] = Thyra::Amesos2::solverTypeNames[k];
428 auto validator = Teuchos::rcp(new Teuchos::StringValidator(solverTypeNames));
429 validParamList->set(SolverType_name,
430 Thyra::Amesos2::solverTypeNames[0],
431 "Type of Amesos2 solver",
432 validator
433 );
434 validParamList->set(RefactorizationPolicy_name,
435 Amesos2::toString(Amesos2::REPIVOT_ON_REFACTORIZATION));
436 validParamList->set(ThrowOnPreconditionerInput_name,bool(true));
437 Teuchos::setupVerboseObjectSublist(&*validParamList);
438
439 // empty Amesos2_Settings parameter list
440 // (Stratimikos won't validate, but Amesos2 will when a user actually try to set parameters)
441 RCP<Teuchos::ParameterList> amesos2Params = rcp(new ParameterList("Amesos2"));
442 validParamList->sublist(Amesos2_Settings_name).setParameters(*amesos2Params);
443 }
444 return validParamList;
445}
