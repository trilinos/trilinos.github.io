24def check_bounds(porder, rate):
25 if (porder==1):
26 if rate>.9:
27 return True
28 else:
29 return False
30 else:
31 if rate>float(porder)-1.2:
32 return True
33 else:
34 return False

36# operators built on second derivatives cannot converge until the polynomial order is at least 2,
37# since degree 1 polynomials have no second derivatives
38second_derivative_operators = ("Laplace-Beltrami", "Gaussian Curvature")
39def rate_is_checked(porder, operator):
40 return not (porder <= 1 and any(name in operator for name in second_derivative_operators))

42num_target_sites = 100
