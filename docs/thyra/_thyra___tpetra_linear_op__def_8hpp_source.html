22template <class Scalar>
23inline
24Teuchos::ETransp
25convertConjNoTransToTeuchosTransMode()
26{
27 TEUCHOS_TEST_FOR_EXCEPTION(
28 Teuchos::ScalarTraits<Scalar>::isComplex,
29 Exceptions::OpNotSupported,
30 "For complex scalars such as " + Teuchos::TypeNameTraits<Scalar>::name() +
31 ", Tpetra does not support conjugation without transposition."
32 );
33 return Teuchos::NO_TRANS; // For non-complex scalars, CONJ is equivalent to NOTRANS.
34}

37template <class Scalar>
38inline
39Teuchos::ETransp
40convertToTeuchosTransMode(const Thyra::EOpTransp transp)
41{
42 switch (transp) {
43 case NOTRANS: return Teuchos::NO_TRANS;
44 case CONJ: return convertConjNoTransToTeuchosTransMode<Scalar>();
45 case TRANS: return Teuchos::TRANS;
46 case CONJTRANS: return Teuchos::CONJ_TRANS;
47 }
48
49 // Should not escape the switch
50 TEUCHOS_TEST_FOR_EXCEPT(true);
51}

124template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
125bool TpetraLinearOp<Scalar,LocalOrdinal,GlobalOrdinal,Node>::opSupportedImpl(
126 Thyra::EOpTransp M_trans) const
127{
128 if (is_null(tpetraOperator_))
129 return false;
130
131 if (M_trans == NOTRANS)
132 return true;
133
134 if (M_trans == CONJ) {
135 // For non-complex scalars, CONJ is always supported since it is equivalent to NO_TRANS.
136 // For complex scalars, Tpetra does not support conjugation without transposition.
137 return !Teuchos::ScalarTraits<Scalar>::isComplex;
138 }
139
140 return tpetraOperator_->hasTransposeApply();
141}

144template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
145void TpetraLinearOp<Scalar,LocalOrdinal,GlobalOrdinal,Node>::applyImpl(
146 const Thyra::EOpTransp M_trans,
147 const Thyra::MultiVectorBase<Scalar> &X_in,
148 const Teuchos::Ptr<Thyra::MultiVectorBase<Scalar> > &Y_inout,
149 const Scalar alpha,
150 const Scalar beta
151 ) const
152{
153 using Teuchos::rcpFromRef;
154 using Teuchos::rcpFromPtr;
155 typedef TpetraOperatorVectorExtraction<Scalar,LocalOrdinal,GlobalOrdinal,Node>
156 ConverterT;
157 typedef Tpetra::MultiVector<Scalar,LocalOrdinal,GlobalOrdinal,Node>
158 TpetraMultiVector_t;
159
160 // Get Tpetra::MultiVector objects for X and Y
161
162 const RCP<const TpetraMultiVector_t> tX =
163 ConverterT::getConstTpetraMultiVector(rcpFromRef(X_in));
164
165 const RCP<TpetraMultiVector_t> tY =
166 ConverterT::getTpetraMultiVector(rcpFromPtr(Y_inout));
167
168 const Teuchos::ETransp tTransp = convertToTeuchosTransMode<Scalar>(M_trans);
169
170 // Apply the operator
171
172 tpetraOperator_->apply(*tX, *tY, tTransp, alpha, beta);
173 // CAG: Commented out since the purpose seems unclear.
174 // Tpetra apply should do all the necessary fencing.
175 // Kokkos::fence();
176}

252template <class Scalar, class LocalOrdinal, class GlobalOrdinal, class Node>
253void TpetraLinearOp<Scalar,LocalOrdinal,GlobalOrdinal,Node>::getRowStatImpl(
254 const RowStatLinearOpBaseUtils::ERowStat rowStat,
255 const Ptr<VectorBase<Scalar> > &rowStatVec_in
256 ) const
257{
258 typedef Tpetra::Vector<Scalar,LocalOrdinal,GlobalOrdinal,Node>
259 TpetraVector_t;
260 typedef Teuchos::ScalarTraits<Scalar> STS;
261 typedef typename STS::magnitudeType MT;
262 typedef Teuchos::ScalarTraits<MT> STM;
263
264 if ( (rowStat == RowStatLinearOpBaseUtils::ROW_STAT_INV_ROW_SUM) ||
265 (rowStat == RowStatLinearOpBaseUtils::ROW_STAT_ROW_SUM) ) {
266
267 TEUCHOS_ASSERT(nonnull(tpetraOperator_));
268 TEUCHOS_ASSERT(nonnull(rowStatVec_in));
269
270 // Currently we only support the case of row sums for a concrete
271 // Tpetra::CrsMatrix where (1) the entire row is stored on a
272 // single process and (2) that the domain map, the range map and
273 // the row map are the SAME. These checks enforce that. Later on
274 // we hope to add complete support for any mapping to the concrete
275 // tpetra matrix types.
276
277 const RCP<TpetraVector_t> tRowSumVec =
278 TpetraOperatorVectorExtraction<Scalar,LocalOrdinal,GlobalOrdinal,Node>::getTpetraVector(rcpFromPtr(rowStatVec_in));
279
280 const RCP<const typename Tpetra::CrsMatrix<Scalar,LocalOrdinal,GlobalOrdinal,Node> > tCrsMatrix =
281 Teuchos::rcp_dynamic_cast<const Tpetra::CrsMatrix<Scalar,LocalOrdinal,GlobalOrdinal,Node> >(tpetraOperator_.getConstObj(),true);
282
283 // EGP: The following assert fails when row sum scaling is applied to blocked Tpetra operators, but without the assert, the correct row sum scaling is obtained.
284 // Furthermore, no valgrind memory errors occur in this case when the assert is removed.
285 //TEUCHOS_ASSERT(tCrsMatrix->getRowMap()->isSameAs(*tCrsMatrix->getDomainMap()));
286 TEUCHOS_ASSERT(tCrsMatrix->getRowMap()->isSameAs(*tCrsMatrix->getRangeMap()));
287 TEUCHOS_ASSERT(tCrsMatrix->getRowMap()->isSameAs(*tRowSumVec->getMap()));
288
289 size_t numMyRows = tCrsMatrix->getLocalNumRows();
290
291 using crs_t = Tpetra::CrsMatrix<Scalar,LocalOrdinal,GlobalOrdinal,Node>;
292 typename crs_t::local_inds_host_view_type indices;
293 typename crs_t::values_host_view_type values;
294
295
296 for (size_t row=0; row < numMyRows; ++row) {
297 MT sum = STM::zero ();
298 tCrsMatrix->getLocalRowView (row, indices, values);
299
300 for (int col = 0; col < (int) values.size(); ++col) {
301 sum += STS::magnitude (values[col]);
302 }
303
304 if (rowStat == RowStatLinearOpBaseUtils::ROW_STAT_INV_ROW_SUM) {
305 if (sum < STM::sfmin ()) {
306 TEUCHOS_TEST_FOR_EXCEPTION(sum == STM::zero (), std::runtime_error,
307 "Error - Thyra::TpetraLinearOp::getRowStatImpl() - Inverse row sum "
308 << "requested for a matrix where one of the rows has a zero row sum!");
309 sum = STM::one () / STM::sfmin ();
310 }
311 else {
312 sum = STM::one () / sum;
313 }
314 }
315
316 tRowSumVec->replaceLocalValue (row, Scalar (sum));
317 }
318
319 }
320 else {
321 TEUCHOS_TEST_FOR_EXCEPTION(true,std::runtime_error,
322 "Error - Thyra::TpetraLinearOp::getRowStatImpl() - Column sum support not implemented!");
323 }
324 Kokkos::fence();
325}
