218template<class Scalar>
219void DefaultBlockedTriangularLinearOpWithSolve<Scalar>::endBlockFill()
220{
221 assertBlockFillIsActive(true);
222 Array<RCP<const VectorSpaceBase<Scalar> > > rangeSpaces;
223 Array<RCP<const VectorSpaceBase<Scalar> > > domainSpaces;
224 for ( int k = 0; k < numDiagBlocks_; ++k ) {
225 const RCP<const LinearOpWithSolveBase<Scalar> > lows_k =
226 diagonalBlocks_[k].getConstObj();
227 TEUCHOS_TEST_FOR_EXCEPTION(is_null(lows_k), std::logic_error,
228 "Error, the block diagonal k="<<k<<" can not be null when ending block fill!"
229 );
230 if (is_null(productRange_)) {
231 rangeSpaces.push_back(lows_k->range());
232 domainSpaces.push_back(lows_k->domain());
233 }
234 }
235 if (is_null(productRange_)) {
236 productRange_ = productVectorSpace<Scalar>(rangeSpaces());
237 productDomain_ = productVectorSpace<Scalar>(domainSpaces());
238 }
239 blockFillIsActive_ = false;
240}

423template<class Scalar>
424bool DefaultBlockedTriangularLinearOpWithSolve<Scalar>::opSupportedImpl(
425 EOpTransp M_trans
426 ) const
427{
428 using Thyra::opSupported;
429 assertBlockFillIsActive(false);
430 for ( int k = 0; k < numDiagBlocks_; ++k ) {
431 if ( !opSupported(*diagonalBlocks_[k].getConstObj(),M_trans) )
432 return false;
433 }
434 return offDiagonalBlocksSupport(M_trans);
435 // ToDo: To be safe we really should do a collective reduction of all
436 // clusters of processes. However, for the typical use case, every block
437 // will return the same info and we should be safe!
438}

441template<class Scalar>
442void DefaultBlockedTriangularLinearOpWithSolve<Scalar>::applyImpl(
443 const EOpTransp M_trans,
444 const MultiVectorBase<Scalar> &X_in,
445 const Ptr<MultiVectorBase<Scalar> > &Y_inout,
446 const Scalar alpha,
447 const Scalar beta
448 ) const
449{
450
451 using Teuchos::RCP;
452 using Teuchos::dyn_cast;
453 using Thyra::apply;
454
455#ifdef THYRA_DEBUG
456 THYRA_ASSERT_LINEAR_OP_MULTIVEC_APPLY_SPACES(
457 "DefaultBlockedTriangularLinearOpWithSolve<Scalar>::apply(...)",
458 *this, M_trans, X_in, &*Y_inout
459 );
460#endif // THYRA_DEBUG
461
462 //
463 // Y = alpha * op(M) * X + beta*Y
464 //
465 // =>
466 //
467 // Y[i] = beta*Y[i] + alpha*sum_j op(M)[i,j]*X[j], for i=0...numDiagBlocks-1
468 //
469 // where op(M)[i,j] is op(M[i,j]) for op = NOTRANS or CONJ, and op(M[j,i])
470 // for op = TRANS or CONJTRANS (see getOpOffDiagonalBlock()).
471 //
472
473 typedef Teuchos::ScalarTraits<Scalar> ST;
474
475 const ProductMultiVectorBase<Scalar>
476 &X = dyn_cast<const ProductMultiVectorBase<Scalar> >(X_in);
477 ProductMultiVectorBase<Scalar>
478 &Y = dyn_cast<ProductMultiVectorBase<Scalar> >(*Y_inout);
479
480 for ( int i = 0; i < numDiagBlocks_; ++ i ) {
481 const RCP<MultiVectorBase<Scalar> > Y_i = Y.getNonconstMultiVectorBlock(i);
482 Thyra::apply( *diagonalBlocks_[i].getConstObj(), M_trans,
483 *X.getMultiVectorBlock(i), Y_i.ptr(), alpha, beta
484 );
485 for ( int j = 0; j < numDiagBlocks_; ++j ) {
486 const RCP<const LinearOpBase<Scalar> >
487 opM_ij = getOpOffDiagonalBlock(M_trans,i,j);
488 if (nonnull(opM_ij)) {
489 Thyra::apply( *opM_ij, M_trans, *X.getMultiVectorBlock(j), Y_i.ptr(),
490 alpha, ST::one() );
491 }
492 }
493 }
494
495}

543template<class Scalar>
544SolveStatus<Scalar>
545DefaultBlockedTriangularLinearOpWithSolve<Scalar>::solveImpl(
546 const EOpTransp M_trans,
547 const MultiVectorBase<Scalar> &B_in,
548 const Ptr<MultiVectorBase<Scalar> > &X_inout,
549 const Ptr<const SolveCriteria<Scalar> > solveCriteria
550 ) const
551{
552
553 using Teuchos::RCP;
554 using Teuchos::dyn_cast;
555 using Thyra::solve;
556
557#ifdef THYRA_DEBUG
558 THYRA_ASSERT_LINEAR_OP_MULTIVEC_APPLY_SPACES(
559 "DefaultBlockedTriangularLinearOpWithSolve<Scalar>::apply(...)",
560 *this, M_trans, *X_inout, &B_in
561 );
562 TEUCHOS_TEST_FOR_EXCEPT(!this->solveSupportsImpl(M_trans));
563 // TEUCHOS_TEST_FOR_EXCEPTION(
564 // nonnull(solveCriteria) && !solveCriteria->solveMeasureType.useDefault(),
565 // std::logic_error,
566 // "Error, we can't handle any non-default solve criteria yet!"
567 // );
568 // ToDo: If solve criteria is to be handled, then we will have to be very
569 // carefull how it it interpreted in terms of the individual period solves!
570#endif // THYRA_DEBUG
571
572 //
573 // X = inv(op(M)) * B
574 //
575 // => block substitution. If op(M) is block lower triangular (M block lower
576 // triangular and op = NOTRANS or CONJ, or M block upper triangular and
577 // op = TRANS or CONJTRANS):
578 //
579 // X[i] = inv(op(M)[i,i])*( B[i] - sum_{j<i} op(M)[i,j]*X[j] ),
580 // for i=0...numDiagBlocks-1
581 //
582 // and if op(M) is block upper triangular:
583 //
584 // X[i] = inv(op(M)[i,i])*( B[i] - sum_{j>i} op(M)[i,j]*X[j] ),
585 // for i=numDiagBlocks-1...0
586 //
587 // (op(M)[i,j] as in applyImpl()). With only diagonal blocks, the blocks
588 // are solved independently. The solve criteria are passed to the solves
589 // of the diagonal blocks.
590 //
591
592 typedef Teuchos::ScalarTraits<Scalar> ST;
593
594 const ProductMultiVectorBase<Scalar>
595 &B = dyn_cast<const ProductMultiVectorBase<Scalar> >(B_in);
596 ProductMultiVectorBase<Scalar>
597 &X = dyn_cast<ProductMultiVectorBase<Scalar> >(*X_inout);
598
599 const bool opIsUpper =
600 ( real_trans(M_trans) == NOTRANS ? hasUpperBlocks_ : hasLowerBlocks_ );
601
602 bool converged = true;
603 for ( int k = 0; k < numDiagBlocks_; ++ k ) {
604 const int i = ( opIsUpper ? numDiagBlocks_-1-k : k );
605 const RCP<const LinearOpWithSolveBase<Scalar> >
606 Op_k = diagonalBlocks_[i].getConstObj();
607 Op_k->setOStream(this->getOStream());
608 Op_k->setVerbLevel(this->getVerbLevel());
609 // B[i] minus the off-diagonal blocks times the solved blocks of X
610 RCP<const MultiVectorBase<Scalar> > B_i = B.getMultiVectorBlock(i);
611 RCP<MultiVectorBase<Scalar> > R_i;
612 const int j_begin = ( opIsUpper ? i+1 : 0 );
613 const int j_end = ( opIsUpper ? numDiagBlocks_ : i );
614 for ( int j = j_begin; j < j_end; ++j ) {
615 const RCP<const LinearOpBase<Scalar> >
616 opM_ij = getOpOffDiagonalBlock(M_trans,i,j);
617 if (nonnull(opM_ij)) {
618 if (is_null(R_i))
619 R_i = B_i->clone_mv();
620 Thyra::apply( *opM_ij, M_trans, *X.getMultiVectorBlock(j), R_i.ptr(),
621 -ST::one(), ST::one() );
622 }
623 }
624 if (nonnull(R_i))
625 B_i = R_i;
626 SolveStatus<Scalar> status =
627 Thyra::solve( *Op_k, M_trans, *B_i,
628 X.getNonconstMultiVectorBlock(i).ptr(), solveCriteria );
629 if (status.solveStatus != SOLVE_STATUS_CONVERGED)
630 converged = false;
631 }
632
633 SolveStatus<Scalar> solveStatus;
634 solveStatus.solveStatus =
635 converged ? SOLVE_STATUS_CONVERGED : SOLVE_STATUS_UNCONVERGED;
636
637 return solveStatus;
638
639}

705template<class Scalar>
706void DefaultBlockedTriangularLinearOpWithSolve<Scalar>::assertAndSetBlockStructure(
707 const PhysicallyBlockedLinearOpBase<Scalar>& blocks
708 )
709{
710 // Checked in all builds: apply and solve access the blocks (i,j) of blocks
711 // for i,j < numDiagBlocks_
712 TEUCHOS_TEST_FOR_EXCEPTION(
713 blocks.productRange()->numBlocks() != numDiagBlocks_, std::logic_error,
714 "Error, DefaultBlockedTriangularLinearOpWithSolve<Scalar>::"
715 "assertAndSetBlockStructure(blocks): the blocked operator has "
716 << blocks.productRange()->numBlocks() << " block rows, but this operator has "
717 << numDiagBlocks_ << " diagonal blocks!"
718 );
719#ifdef THYRA_DEBUG
720 THYRA_ASSERT_VEC_SPACES(
721 "DefaultBlockedTriangularLinearOpWithSolve<Scalar>::assertAndSetBlockStructure(blocks)",
722 *blocks.range(), *this->range()
723 );
724 THYRA_ASSERT_VEC_SPACES(
725 "DefaultBlockedTriangularLinearOpWithSolve<Scalar>::assertAndSetBlockStructure(blocks)",
726 *blocks.domain(), *this->domain()
727 );
728#endif
729 // Sets the flags only if blocks is block triangular, so that a rejected
730 // operator leaves this object unchanged
731 getBlockStructure(blocks, hasLowerBlocks_, hasUpperBlocks_);
732}

735template<class Scalar>
736void DefaultBlockedTriangularLinearOpWithSolve<Scalar>::getBlockStructure(
737 const PhysicallyBlockedLinearOpBase<Scalar> &blocks,
738 bool &hasLower, bool &hasUpper
739 )
740{
741 const int numBlocks = blocks.productRange()->numBlocks();
742 TEUCHOS_TEST_FOR_EXCEPTION(
743 blocks.productDomain()->numBlocks() != numBlocks, std::logic_error,
744 "Error, DefaultBlockedTriangularLinearOpWithSolve<Scalar>: the blocked "
745 "operator has " << numBlocks << " block rows and "
746 << blocks.productDomain()->numBlocks() << " block columns!"
747 );
748 // The off-diagonal blocks must all be above or all be below the diagonal
749 bool lower = false, upper = false;
750 for ( int i = 0; i < numBlocks; ++i ) {
751 for ( int j = 0; j < numBlocks; ++j ) {
752 if ( i != j && nonnull(blocks.getBlock(i,j)) ) {
753 if (i > j)
754 lower = true;
755 else
756 upper = true;
757 }
758 }
759 }
760 TEUCHOS_TEST_FOR_EXCEPTION( lower && upper, std::logic_error,
761 "Error, DefaultBlockedTriangularLinearOpWithSolve<Scalar>: the blocked "
762 "operator has nonzero blocks both below and above the diagonal, it is not "
763 "block triangular!"
764 );
765 hasLower = lower;
766 hasUpper = upper;
767}
