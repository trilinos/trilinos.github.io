57 AlgebraicDAEModel(const double kappa = 0.5, const double h0 = 1.0,
58 const double p0 = 0.3, const double q0 = 0.1)
59 : kappa_(kappa), h0_(h0), p0_(p0), q0_(q0)
60 {
61 space_ = Thyra::defaultSpmdVectorSpace<double>(2);
62 p_names_.push_back("p");
63 p_names_.push_back("q");
64 g_names_.push_back("g = x");
65
66 // The descriptions must be this->description(), which is checked by
67 // Thyra::ModelEvaluatorDefaultBase in debug builds
68 MEB::InArgsSetup<double> inArgs;
69 inArgs.setModelEvalDescription(this->description());
70 inArgs.setSupports(MEB::IN_ARG_t);
71 inArgs.setSupports(MEB::IN_ARG_x);
72 inArgs.setSupports(MEB::IN_ARG_x_dot);
73 inArgs.setSupports(MEB::IN_ARG_alpha);
74 inArgs.setSupports(MEB::IN_ARG_beta);
75 inArgs.set_Np(1);
76 inArgs_ = inArgs;
77
78 MEB::OutArgsSetup<double> outArgs;
79 outArgs.setModelEvalDescription(this->description());
80 outArgs.setSupports(MEB::OUT_ARG_f);
81 outArgs.setSupports(MEB::OUT_ARG_W_op);
82 outArgs.set_Np_Ng(1, 1);
83 outArgs.setSupports(MEB::OUT_ARG_DfDp, 0, MEB::DERIV_MV_JACOBIAN_FORM);
84 outArgs.setSupports(MEB::OUT_ARG_DgDx, 0, MEB::DERIV_MV_GRADIENT_FORM);
85 outArgs.setSupports(MEB::OUT_ARG_DgDp, 0, 0, MEB::DERIV_MV_JACOBIAN_FORM);
86 outArgs_ = outArgs;
87
88 nominal_ = inArgs_;
89 nominal_.set_t(0.0);
90 RCP<Thyra::VectorBase<double> > x = Thyra::createMember(space_);
91 RCP<Thyra::VectorBase<double> > x_dot = Thyra::createMember(space_);
92 RCP<Thyra::VectorBase<double> > p = Thyra::createMember(space_);
93 {
94 Thyra::DetachedVectorView<double> xv(*x), xdv(*x_dot), pv(*p);
95 xv[0] = kappa_ * h0_ + p0_;
96 xv[1] = h0_;
97 xdv[0] = 0.0;
98 xdv[1] = a() * h0_ + p0_ + 2.0 * q0_;
99 pv[0] = p0_;
100 pv[1] = q0_;
101 }
102 nominal_.set_x(x);
103 nominal_.set_x_dot(x_dot);
104 nominal_.set_p(0, p);
105 }
106
107 double a() const { return kappa_ - 1.0; }
108 double kappa() const { return kappa_; }

117 RCP<Thyra::MultiVectorBase<double> > getInitialSensitivity() const

119 RCP<Thyra::MultiVectorBase<double> > dxdp = Thyra::createMembers(space_, 2);
120 Thyra::assign(dxdp.ptr(), 0.0);
121 Thyra::DetachedMultiVectorView<double> v(*dxdp);
122 v(0, 0) = 1.0;

127 void exactSensitivity(const double t, double s[2][2]) const
128 {
129 const double st = (std::exp(a() * t) - 1.0) / a();
130 s[0][0] = kappa_ * st + 1.0;
131 s[0][1] = 2.0 * kappa_ * st;
132 s[1][0] = st;
133 s[1][1] = 2.0 * st;
134 }
135
137 void exactIntegratedSensitivity(const double t, double s[2][2]) const
138 {
139 const double St = ((std::exp(a() * t) - 1.0) / a() - t) / a();
140 s[0][0] = kappa_ * St + t;
141 s[0][1] = 2.0 * kappa_ * St;
142 s[1][0] = St;
143 s[1][1] = 2.0 * St;
144 }

189 void evalModelImpl(const MEB::InArgs<double>& inArgs,
190 const MEB::OutArgs<double>& outArgs) const
191 {
192 Thyra::ConstDetachedVectorView<double> x(*inArgs.get_x().assert_not_null());
193 RCP<const Thyra::VectorBase<double> > p_in = inArgs.get_p(0);
194 if (p_in == Teuchos::null) p_in = nominal_.get_p(0);
195 Thyra::ConstDetachedVectorView<double> p(*p_in);
196 double h_dot = 0.0;
197 if (inArgs.get_x_dot() != Teuchos::null)
198 h_dot = Thyra::get_ele(*inArgs.get_x_dot(), 1);
199
200 RCP<Thyra::VectorBase<double> > f = outArgs.get_f();
201 if (f != Teuchos::null) {
202 Thyra::DetachedVectorView<double> fv(*f);
203 fv[0] = x[0] - kappa_ * x[1] - p[0];
204 fv[1] = h_dot + x[1] - x[0] - 2.0 * p[1];
205 }
206
207 // W = alpha df/dx_dot + beta df/dx, with df/dx_dot = [0 0; 0 1]
208 RCP<Thyra::LinearOpBase<double> > W_op = outArgs.get_W_op();
209 if (W_op != Teuchos::null) {
210 const double alpha = inArgs.get_alpha();
211 const double beta = inArgs.get_beta();
212 RCP<Thyra::MultiVectorBase<double> > W =
213 Teuchos::rcp_dynamic_cast<Thyra::MultiVectorBase<double> >(W_op, true);
214 Thyra::DetachedMultiVectorView<double> Wv(*W);
215 Wv(0, 0) = beta;
216 Wv(0, 1) = -kappa_ * beta;
217 Wv(1, 0) = -beta;
218 Wv(1, 1) = alpha + beta;
219 }
220
221 RCP<Thyra::MultiVectorBase<double> > dfdp = outArgs.get_DfDp(0).getMultiVector();
222 if (dfdp != Teuchos::null) {
223 Thyra::DetachedMultiVectorView<double> v(*dfdp);
224 v(0, 0) = -1.0;
225 v(0, 1) = 0.0;
226 v(1, 0) = 0.0;
227 v(1, 1) = -2.0;
228 }
229
230 RCP<Thyra::VectorBase<double> > g = outArgs.get_g(0);
231 if (g != Teuchos::null) Thyra::assign(g.ptr(), *inArgs.get_x());
232
233 RCP<Thyra::MultiVectorBase<double> > dgdx = outArgs.get_DgDx(0).getMultiVector();
234 if (dgdx != Teuchos::null) {
235 Thyra::DetachedMultiVectorView<double> v(*dgdx);
236 v(0, 0) = 1.0;
237 v(0, 1) = 0.0;
238 v(1, 0) = 0.0;
239 v(1, 1) = 1.0;
240 }
241
242 RCP<Thyra::MultiVectorBase<double> > dgdp = outArgs.get_DgDp(0, 0).getMultiVector();
243 if (dgdp != Teuchos::null) Thyra::assign(dgdp.ptr(), 0.0);
244 }

246 double kappa_, h0_, p0_, q0_;
247 RCP<const Thyra::VectorSpaceBase<double> > space_;
248 Teuchos::Array<std::string> p_names_;
249 Teuchos::Array<std::string> g_names_;
250 MEB::InArgs<double> inArgs_;
251 MEB::OutArgs<double> outArgs_;
252 MEB::InArgs<double> nominal_;
253};

351RCP<Thyra::MultiVectorBase<double> > runAdjoint(
352 const RCP<AlgebraicDAEModel>& model, const std::string& responseType,
353 const std::string& responseTimes, const double dt,
354 const RCP<const Thyra::VectorBase<double> >& mask, bool& status)
355{
356 RCP<ParameterList> pl = getTempusParameters(dt);
357 ParameterList& sens_pl = pl->sublist("Sensitivities");
358 sens_pl.set("Mass Matrix Is Identity", false);
359 sens_pl.set("Response Type", responseType);
360 if (!responseTimes.empty()) sens_pl.set("Response Time List", responseTimes);
361 sens_pl.set(Tempus::DAEStructure<double>::algebraicMaskName(), mask);
362 sens_pl.sublist("Algebraic Jump Solver").set("Relative Tolerance", 1.0e-12);
363
364 RCP<Tempus::IntegratorAdjointSensitivity<double> > integrator =
365 Tempus::createIntegratorAdjointSensitivity<double>(pl, model);
366 integrator->initializeSolutionHistory(
367 0.0, model->getInitialCondition(), Teuchos::null, Teuchos::null,
368 model->getInitialSensitivity(), Teuchos::null, Teuchos::null);
369 status = integrator->advanceTime();
370
371 // getDgDp() is in gradient form: (p, g)
372 RCP<const Thyra::MultiVectorBase<double> > DgDp = integrator->getDgDp();
373 RCP<Thyra::MultiVectorBase<double> > dgdp = Thyra::createMembers(model->get_g_space(0), 2);
374 Thyra::ConstDetachedMultiVectorView<double> grad(*DgDp);
375 Thyra::DetachedMultiVectorView<double> jac(*dgdp);
376 for (int i = 0; i < 2; ++i)
377 for (int j = 0; j < 2; ++j) jac(i, j) = grad(j, i);
378 return dgdp;
379}

382void checkConvergence(const std::string& responseType,
383 Teuchos::FancyOStream& out, bool& success)
384{
385 RCP<AlgebraicDAEModel> model = rcp(new AlgebraicDAEModel());
386 const std::vector<double> times = {0.25, 0.5, 1.0};
387 const std::string timeList = "0.25, 0.5, 1.0";
388
389 // Exact sensitivity of the response
390 double exact[2][2] = {{0.0, 0.0}, {0.0, 0.0}};
391 if (responseType == "Final Time")
392 model->exactSensitivity(1.0, exact);
393 else if (responseType == "Time Integral")
394 model->exactIntegratedSensitivity(1.0, exact);
395 else {
396 for (const double t : times) {
397 double s[2][2];
398 model->exactSensitivity(t, s);
399 for (int i = 0; i < 2; ++i)
400 for (int j = 0; j < 2; ++j) exact[i][j] += s[i][j];
401 }
402 }
403
404 std::vector<double> stepSize, errorNorm;
405 for (int n = 0; n < 5; ++n) {
406 const double dt = 1.0 / (16 << n); // exact in binary
407 bool status = false;
408 RCP<Thyra::MultiVectorBase<double> > dgdp =
409 runAdjoint(model, responseType,
410 responseType == "Sum Over Times" ? timeList : "", dt,
411 createMask(*model, 1.0, 0.0), status);
412 TEST_ASSERT(status);
413
414 // Backward Euler sensitivity s_k of h at t_k = k dt (exact for the
415 // discrete solution, for comparison only)
416 double s_BE = 0.0, sum_BE = 0.0;
417 const int N = 16 << n;
418 for (int k = 1; k <= N; ++k) {
419 s_BE = (s_BE + dt) / (1.0 - model->a() * dt);
420 for (const double t : times)
421 if (std::abs(k * dt - t) < 1.0e-12) sum_BE += s_BE;
422 }
423 const double s_disc = responseType == "Final Time" ? s_BE : sum_BE;
424 const double ns = responseType == "Final Time" ? 1.0 : 3.0;
425
426 Thyra::ConstDetachedMultiVectorView<double> v(*dgdp);
427 double err2 = 0.0;
428 for (int i = 0; i < 2; ++i)
429 for (int j = 0; j < 2; ++j)
430 err2 += (v(i, j) - exact[i][j]) * (v(i, j) - exact[i][j]);
431 stepSize.push_back(dt);
432 errorNorm.push_back(std::sqrt(err2));
433 out << " " << responseType << ", dt = " << dt << ": du/dp = " << v(0, 0)
434 << ", du/dq = " << v(0, 1) << ", dh/dp = " << v(1, 0)
435 << ", dh/dq = " << v(1, 1) << ", |error| = " << std::sqrt(err2);
436 if (responseType != "Time Integral")
437 out << ", du/dp - (discrete BE) = "
438 << v(0, 0) - (model->kappa() * s_disc + ns);
439 out << std::endl;
440 }
441
442 const double slope = computeLinearRegressionLogLog<double>(stepSize, errorNorm);
443 out << " " << responseType << ": order = " << slope << std::endl;
444 TEST_FLOATING_EQUALITY(slope, 1.0, 0.05);
445 TEST_COMPARE(errorNorm.back(), <, 1.0e-2);
446}

466// ************************************************************
467// The algebraic component of the jump: for g = u at the final time, the
468// sensitivity to p jumps by 1 (the ODE formula would need df/dx_dot^{-T},
469// which does not exist).
470TEUCHOS_UNIT_TEST(BackwardEuler, ASA_AlgebraicEquations_AlgebraicJump)
471{
472 RCP<AlgebraicDAEModel> model = rcp(new AlgebraicDAEModel());
473 bool status = false;
474 const double dt = 1.0 / 64;
475 RCP<Thyra::MultiVectorBase<double> > dgdp = runAdjoint(
476 model, "Final Time", "", dt, createMask(*model, 1.0, 0.0), status);
477 TEST_ASSERT(status);
478 Thyra::ConstDetachedMultiVectorView<double> v(*dgdp);
479 // du/dp = kappa dh/dp + 1 and du/dq = kappa dh/dq hold exactly for the
480 // discrete solution
481 TEST_FLOATING_EQUALITY(v(0, 0), model->kappa() * v(1, 0) + 1.0, 1.0e-10);
482 TEST_FLOATING_EQUALITY(v(0, 1), model->kappa() * v(1, 1), 1.0e-10);
483}
