54inline AdjointResponseType parseResponseType(const std::string& name)
55{
56 if (name == "Final Time") return AdjointResponseType::FinalTime;
57 if (name == "Time Integral") return AdjointResponseType::TimeIntegral;
58 if (name == "Sum Over Times") return AdjointResponseType::SumOverTimes;
59 TEUCHOS_TEST_FOR_EXCEPTION(
60 true, std::invalid_argument,
61 "Error - invalid \"Response Type\" = \""
62 << name
63 << "\". Valid values are \"Final Time\", \"Time Integral\" and "
64 "\"Sum Over Times\".\n");
65 return AdjointResponseType::FinalTime; // not reached
66}

70template <class Scalar>
71std::vector<Scalar> parseTimeList(const std::string& str)
72{
73 std::vector<Scalar> times;
74 std::string::size_type lastPos = str.find_first_not_of(", ", 0);
75 std::string::size_type pos = str.find_first_of(",", lastPos);
76 while (lastPos != std::string::npos) {
77 std::string token = str.substr(lastPos, pos - lastPos);
78 token.erase(token.find_last_not_of(" \t\n\r") + 1);
79 if (!token.empty()) {
80 std::size_t numParsed = 0;
81 double value = 0.0;
82 try {
83 value = std::stod(token, &numParsed);
84 }
85 catch (const std::exception&) {
86 numParsed = 0;
87 }
88 TEUCHOS_TEST_FOR_EXCEPTION(
89 numParsed != token.size(), std::invalid_argument,
90 "Error - could not parse \"" << token << "\" in the time list \""
91 << str << "\".\n");
92 times.push_back(Scalar(value));
93 }
94 if (pos == std::string::npos) break;
95 lastPos = str.find_first_not_of(", ", pos);
96 pos = str.find_first_of(",", lastPos);
97 }
98 std::sort(times.begin(), times.end());
99 return times;
100}

131template <class Scalar>
132Scalar timeTolerance(const Scalar t_init, const Scalar t_final,
133 const Teuchos::ParameterList* sensPL = nullptr)
134{
135 using std::abs;
136 const auto tolerance = [sensPL](const std::string& name,
137 const double defaultValue) {
138 if (sensPL == nullptr || !sensPL->isParameter(name))
139 return defaultValue;
140 else
141 return sensPL->get<double>(name);
142 };
143 const Scalar relTol(tolerance("Response Time Relative Tolerance", 1.0e-9));
144 const Scalar absTol(tolerance("Response Time Absolute Tolerance", 1.0e-15));
145 return std::max(relTol * std::max(abs(t_init), abs(t_final)), absTol);
146}

156template <class Scalar>
157void evalResponseGradientX(const Thyra::ModelEvaluator<Scalar>& model,
158 const int g_index,
159 const Thyra::ModelEvaluatorBase::InArgs<Scalar>& inArgs,
160 Thyra::MultiVectorBase<Scalar>& dgdx,
161 const bool dgdx_dot_check, const std::string& caller)
162{
163 typedef Thyra::ModelEvaluatorBase MEB;
164 MEB::OutArgs<Scalar> outArgs = model.createOutArgs();
165 TEUCHOS_TEST_FOR_EXCEPTION(
166 !outArgs.supports(MEB::OUT_ARG_DgDx, g_index)
167 .supports(MEB::DERIV_MV_GRADIENT_FORM),
168 std::logic_error,
169 "Error - " << caller << ": the model must support dg/dx in gradient "
170 << "(DERIV_MV_GRADIENT_FORM) form for response " << g_index
171 << ".\n");
172 outArgs.set_DgDx(g_index, MEB::Derivative<Scalar>(Teuchos::rcpFromRef(dgdx),
173 MEB::DERIV_MV_GRADIENT_FORM));
174
175 Teuchos::RCP<Thyra::MultiVectorBase<Scalar>> dgdx_dot;
176 if (dgdx_dot_check &&
177 outArgs.supports(MEB::OUT_ARG_DgDx_dot, g_index).supports(MEB::DERIV_MV_GRADIENT_FORM)) {
178 dgdx_dot = Thyra::createMembers(model.get_x_space(), dgdx.domain()->dim());
179 outArgs.set_DgDx_dot(g_index, MEB::Derivative<Scalar>(dgdx_dot, MEB::DERIV_MV_GRADIENT_FORM));
180 }
181 model.evalModel(inArgs, outArgs);
182
183 if (dgdx_dot != Teuchos::null) {
184 typedef typename Teuchos::ScalarTraits<Scalar>::magnitudeType Mag;
185 Teuchos::Array<Mag> norms(dgdx_dot->domain()->dim());
186 Thyra::norms_inf(*dgdx_dot, norms());
187 const Mag maxNorm = *std::max_element(norms.begin(), norms.end());
188 TEUCHOS_TEST_FOR_EXCEPTION(
189 maxNorm != Teuchos::ScalarTraits<Mag>::zero(), std::logic_error,
190 "Error - " << caller << ": response " << g_index
191 << " depends on x_dot (|dg/dx_dot| = " << maxNorm
192 << "). The \"Time Integral\" and \"Sum Over Times\" "
193 << "adjoint responses require g = g(x,p,t).\n");
194 }
195}

203template <class Scalar>
204void evalResponseGradientP(const Thyra::ModelEvaluator<Scalar>& model,
205 const int g_index, const int p_index,
206 const Thyra::ModelEvaluatorBase::InArgs<Scalar>& inArgs,
207 Thyra::MultiVectorBase<Scalar>& dgdp)
208{
209 typedef Thyra::ModelEvaluatorBase MEB;
210 MEB::OutArgs<Scalar> outArgs = model.createOutArgs();
211 MEB::DerivativeSupport dgdp_support =
212 outArgs.supports(MEB::OUT_ARG_DgDp, g_index, p_index);
213 if (dgdp_support.supports(MEB::DERIV_MV_GRADIENT_FORM)) {
214 outArgs.set_DgDp(g_index, p_index,
215 MEB::Derivative<Scalar>(Teuchos::rcpFromRef(dgdp),
216 MEB::DERIV_MV_GRADIENT_FORM));
217 model.evalModel(inArgs, outArgs);
218 }
219 else if (dgdp_support.supports(MEB::DERIV_MV_JACOBIAN_FORM)) {
220 const int num_g = dgdp.domain()->dim();
221 const int num_p = dgdp.range()->dim();
222 Teuchos::RCP<Thyra::MultiVectorBase<Scalar>> dgdp_trans =
223 Thyra::createMembers(model.get_g_space(g_index), num_p);
224 outArgs.set_DgDp(g_index, p_index, MEB::Derivative<Scalar>(dgdp_trans, MEB::DERIV_MV_JACOBIAN_FORM));
225 model.evalModel(inArgs, outArgs);
226 Thyra::DetachedMultiVectorView<Scalar> dgdp_view(dgdp);
227 Thyra::ConstDetachedMultiVectorView<Scalar> dgdp_trans_view(*dgdp_trans);
228 for (int i = 0; i < num_p; ++i)
229 for (int j = 0; j < num_g; ++j) dgdp_view(i, j) = dgdp_trans_view(j, i);
230 }
231 else
232 TEUCHOS_TEST_FOR_EXCEPTION(true, std::logic_error, "Invalid dg/dp support");
233}

243template <class Scalar>
244void applyDfDpTranspose(const Thyra::ModelEvaluator<Scalar>& model,
245 const int p_index,
246 const Thyra::ModelEvaluatorBase::InArgs<Scalar>& inArgs,
247 const Thyra::MultiVectorBase<Scalar>& v,
248 Thyra::MultiVectorBase<Scalar>& y, const Scalar alpha,
249 const Scalar beta,
250 Teuchos::RCP<Thyra::LinearOpBase<Scalar>>& dfdp_op,
251 Teuchos::RCP<Thyra::MultiVectorBase<Scalar>>& dfdp_mv)
252{
253 typedef Thyra::ModelEvaluatorBase MEB;
254 MEB::OutArgs<Scalar> outArgs = model.createOutArgs();
255 MEB::DerivativeSupport dfdp_support = outArgs.supports(MEB::OUT_ARG_DfDp, p_index);
256 Thyra::EOpTransp trans = Thyra::CONJTRANS;
257 if (dfdp_support.supports(MEB::DERIV_LINEAR_OP)) {
258 if (dfdp_op == Teuchos::null) dfdp_op = model.create_DfDp_op(p_index);
259 outArgs.set_DfDp(p_index, MEB::Derivative<Scalar>(dfdp_op));
260 }
261 else if (dfdp_support.supports(MEB::DERIV_MV_JACOBIAN_FORM)) {
262 if (dfdp_mv == Teuchos::null)
263 dfdp_mv = Thyra::createMembers(model.get_f_space(), model.get_p_space(p_index)->dim());
264 outArgs.set_DfDp(p_index, MEB::Derivative<Scalar>(dfdp_mv, MEB::DERIV_MV_JACOBIAN_FORM));
265 dfdp_op = dfdp_mv;
266 }
267 else if (dfdp_support.supports(MEB::DERIV_MV_GRADIENT_FORM)) {
268 if (dfdp_mv == Teuchos::null)
269 dfdp_mv = Thyra::createMembers(model.get_p_space(p_index), model.get_f_space()->dim());
270 outArgs.set_DfDp(p_index, MEB::Derivative<Scalar>(dfdp_mv, MEB::DERIV_MV_GRADIENT_FORM));
271 dfdp_op = dfdp_mv;
272 trans = Thyra::CONJ;
273 }
274 else
275 TEUCHOS_TEST_FOR_EXCEPTION(true, std::logic_error, "Invalid df/dp support");
276 model.evalModel(inArgs, outArgs);
277 dfdp_op->apply(trans, v, Teuchos::ptrFromRef(y), alpha, beta);
278}

285template <class Scalar>
286Teuchos::RCP<Thyra::LinearOpWithSolveBase<Scalar>>
287createAdjointLinearOpWithSolve(
288 const Thyra::ModelEvaluator<Scalar>& adjoint_model,
289 const Thyra::ModelEvaluatorBase::InArgs<Scalar>& inArgs, const Scalar alpha,
290 const Scalar beta)
291{
292 using Teuchos::RCP;
293 using Thyra::LinearOpBase;
294 using Thyra::LinearOpWithSolveBase;
295 using Thyra::LinearOpWithSolveFactoryBase;
296 using Thyra::PreconditionerBase;
297 using Thyra::PreconditionerFactoryBase;
298 typedef Thyra::ModelEvaluatorBase MEB;
299
300 MEB::InArgs<Scalar> inargs = inArgs;
301 inargs.set_alpha(alpha);
302 inargs.set_beta(beta);
303 MEB::OutArgs<Scalar> adj_outargs = adjoint_model.createOutArgs();
304 RCP<LinearOpWithSolveBase<Scalar>> W;
305 if (adj_outargs.supports(MEB::OUT_ARG_W)) {
306 // Model supports W
307 W = adjoint_model.create_W();
308 adj_outargs.set_W(W);
309 adjoint_model.evalModel(inargs, adj_outargs);
310 }
311 else {
312 // Otherwise model must support a W_op and W factory
313 RCP<const LinearOpWithSolveFactoryBase<Scalar>> lowsfb =
314 adjoint_model.get_W_factory();
315 TEUCHOS_TEST_FOR_EXCEPTION(lowsfb == Teuchos::null, std::logic_error,
316 "Adjoint ME must support W out-arg or provide "
317 "a W_factory for non-identity mass matrix");
318
319 // Compute W_op (and W_prec if supported)
320 RCP<LinearOpBase<Scalar>> W_op = adjoint_model.create_W_op();
321 adj_outargs.set_W_op(W_op);
322 RCP<PreconditionerFactoryBase<Scalar>> prec_factory = lowsfb->getPreconditionerFactory();
323 RCP<PreconditionerBase<Scalar>> W_prec;
324 if (prec_factory != Teuchos::null)
325 W_prec = prec_factory->createPrec();
326 else if (adj_outargs.supports(MEB::OUT_ARG_W_prec)) {
327 W_prec = adjoint_model.create_W_prec();
328 adj_outargs.set_W_prec(W_prec);
329 }
330 adjoint_model.evalModel(inargs, adj_outargs);
331
332 // Create and initialize W
333 W = lowsfb->createOp();
334 if (W_prec != Teuchos::null) {
335 if (prec_factory != Teuchos::null)
336 prec_factory->initializePrec(Thyra::defaultLinearOpSource<Scalar>(W_op), W_prec.get());
337 Thyra::initializePreconditionedOp<Scalar>(*lowsfb, W_op, W_prec, W.ptr());
338 }
339 else
340 Thyra::initializeOp<Scalar>(*lowsfb, W_op, W.ptr());
341 }
342 TEUCHOS_TEST_FOR_EXCEPTION(
343 W == Teuchos::null, std::logic_error,
344 "A null W has been encountered in "
345 "Tempus::AdjointSensitivityUtils::createAdjointLinearOpWithSolve!\n");
346 return W;
347}
