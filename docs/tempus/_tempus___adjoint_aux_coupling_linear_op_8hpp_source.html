98 bool opSupportedImpl(Thyra::EOpTransp M_trans) const
99 {
100 if (Teuchos::is_null(dfdp_)) return true;
101 // A complex conjugation of the scalings is not implemented
102 if (ST::isComplex && isConjugate(M_trans) && isScaled()) return false;
103 return Thyra::opSupported(*dfdp_, Thyra::trans_trans(dfdp_trans_, M_trans));
104 }

108 void applyImpl(const Thyra::EOpTransp M_trans,
109 const Thyra::MultiVectorBase<Scalar>& X,
110 const Teuchos::Ptr<Thyra::MultiVectorBase<Scalar> >& Y,
111 const Scalar alpha, const Scalar beta) const
112 {
113 using Teuchos::RCP;
114 typedef Thyra::MultiVectorBase<Scalar> MV;
115
116 if (Teuchos::is_null(dfdp_)) {
117 scaleOrZero(beta, *Y);
118 return;
119 }
120 TEUCHOS_TEST_FOR_EXCEPTION(
121 ST::isComplex && isConjugate(M_trans) && isScaled(), std::logic_error,
122 "Error - Tempus::AdjointAuxCouplingLinearOp::apply(): a conjugate "
123 "apply of the scaled operator is not supported.\n");
124
125 // op_M(D_l*s*op(A)*D_r) = D_l*s*op_M(op(A))*D_r for NOTRANS and CONJ,
126 // and D_r*s*op_M(op(A))*D_l for TRANS and CONJTRANS (real scalings)
127 const bool notrans = (Thyra::real_trans(M_trans) == Thyra::NOTRANS);
128 const RCP<const Thyra::VectorBase<Scalar> > pre_scaling =
129 notrans ? right_scaling_ : left_scaling_;
130 const RCP<const Thyra::VectorBase<Scalar> > post_scaling =
131 notrans ? left_scaling_ : right_scaling_;
132 const Scalar s = isConjugate(M_trans) ? ST::conjugate(s_) : s_;
133 const Thyra::EOpTransp trans = Thyra::trans_trans(dfdp_trans_, M_trans);
134
135 RCP<const MV> Xs = Teuchos::rcpFromRef(X);
136 if (Teuchos::nonnull(pre_scaling)) {
137 RCP<MV> X_scaled = X.clone_mv();
138 X_scaled->scaleLeft(*pre_scaling);
139 Xs = X_scaled;
140 }
141 if (Teuchos::is_null(post_scaling)) {
142 Thyra::apply(*dfdp_, trans, *Xs, Y, alpha * s, beta);
143 }
144 else {
145 RCP<MV> T = Thyra::createMembers(notrans ? range_ : domain_,
146 X.domain()->dim());
147 Thyra::apply(*dfdp_, trans, *Xs, T.ptr(), s, ST::zero());
148 T->scaleLeft(*post_scaling);
149 scaleOrZero(beta, *Y);
150 Thyra::update(alpha, *T, Y);
151 }
152 }
