148 void getRowStatImpl(const RowStatLinearOpBaseUtils::ERowStat rowStat,
149 const Ptr<VectorBase<Scalar> >& rowStatVec) const
150 {
151 typedef Teuchos::ScalarTraits<Scalar> ST;
152 TEUCHOS_ASSERT(this->rowStatIsSupported(rowStat));
153 const Scalar abs_s = ST::magnitude(s_);
154 if (rowStat == RowStatLinearOpBaseUtils::ROW_STAT_INV_ROW_SUM ||
155 rowStat == RowStatLinearOpBaseUtils::ROW_STAT_INV_COL_SUM)
156 put_scalar(ST::one() / abs_s, rowStatVec);
157 else
158 put_scalar(abs_s, rowStatVec);
159 }

193 static Scalar uniformValue(const VectorBase<Scalar>& v,
194 const std::string& caller)
195 {
196 typedef Teuchos::ScalarTraits<Scalar> ST;
197 typedef typename ST::magnitudeType Mag;
198 if (v.space()->dim() == 0) return ST::one();
199 const Scalar d = get_ele(v, 0);
200 RCP<VectorBase<Scalar> > diff = v.clone_v();
201 add_scalar(-d, diff.ptr());
202 TEUCHOS_TEST_FOR_EXCEPTION(
203 norm_inf(*diff) > Mag(100) * ST::eps() * ST::magnitude(d),
204 std::logic_error,
205 "Error - Thyra::ScaledIdentityLinearOpWithSolve::"
206 << caller
207 << ": only a scaling by a uniform vector (all entries equal) "
208 "keeps the operator a scaled identity.\n");
209 return d;
210 }
