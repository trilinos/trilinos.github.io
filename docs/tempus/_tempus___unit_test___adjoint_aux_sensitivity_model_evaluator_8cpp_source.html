41// Model that counts the evaluations of df/dx (W with alpha = 0, beta = 1)
42// and of df/dp of the underlying model
43class CountingModel : public Thyra::ModelEvaluatorDelegatorBase<double> {
44 public:
45 CountingModel(const RCP<const Thyra::ModelEvaluator<double> >& model)
46 : num_dfdx(0), num_dfdp(0)
47 {
48 this->initialize(model);
49 }
50 mutable int num_dfdx;
51 mutable int num_dfdp;
52
53 private:
54 void evalModelImpl(const MEB::InArgs<double>& inArgs,
55 const MEB::OutArgs<double>& outArgs) const
56 {
57 if (nonnull(outArgs.get_W_op()) && inArgs.get_alpha() == 0.0 &&
58 inArgs.get_beta() == 1.0)
59 ++num_dfdx;
60 if (!outArgs.supports(MEB::OUT_ARG_DfDp, 0).none() &&
61 !outArgs.get_DfDp(0).isEmpty())
62 ++num_dfdp;
63 this->getUnderlyingModel()->evalModel(inArgs, outArgs);
64 }
65};

67// SinCos model, its adjoint, and a forward solution history with the exact
68// solution at t = 0, 0.5, 1
69struct SinCosSetup {
70 RCP<SinCosModel<double> > sincos;
71 RCP<CountingModel> model;
72 RCP<CountingModel> adjoint_model;
73 RCP<Tempus::SolutionHistory<double> > sh;
74
75 SinCosSetup()
76 {
77 RCP<ParameterList> scm_pl = Teuchos::parameterList();
78 scm_pl->set("Accept model parameters", true);
79 scm_pl->set("Provide nominal values", true);
80 scm_pl->set("Coeff a", 0.5);
81 scm_pl->set("Coeff f", 2.0);
82 scm_pl->set("Coeff L", 1.5);
83 sincos = rcp(new SinCosModel<double>(scm_pl));
84 model = rcp(new CountingModel(sincos));
85 adjoint_model = rcp(new CountingModel(rcp(new SinCosModelAdjoint<double>(scm_pl))));
86
87 sh = rcp(new Tempus::SolutionHistory<double>());
88 sh->setStorageTypeString("Unlimited");
89 const double times[3] = {0.0, 0.5, 1.0};
90 for (int i = 0; i < 3; ++i) {
91 MEB::InArgs<double> exact = sincos->getExactSolution(times[i]);
92 RCP<Tempus::SolutionState<double> > state =
93 Tempus::createSolutionStateX<double>(exact.get_x()->clone_v(),
94 exact.get_x_dot()->clone_v());
95 state->setTime(times[i]);
96 state->setIndex(i);
97 sh->addState(state);
98 }
99 sh->initialize();
100 }
101
102 RCP<AuxME> createAuxME(const std::string& responseType) const
103 {
104 RCP<ParameterList> pl = Teuchos::parameterList();
105 pl->set("Response Type", responseType);
106 RCP<AuxME> aux = rcp(new AuxME(model, adjoint_model, 0.0, 1.0, pl));
107 aux->setForwardSolutionHistory(sh);
108 return aux;
109 }
110};

112// The inArgs of the adjoint model at the adjoint time t, with random adjoint
113// state (y, z) and time derivative (x_dot null if !set_x_dot)
114MEB::InArgs<double> randomInArgs(const AuxME& aux, const double t,
115 const double alpha, const double beta,
116 const bool set_x_dot = true)
117{
118 MEB::InArgs<double> inArgs = aux.getNominalValues();
119 RCP<Vector> x = Thyra::createMember(aux.get_x_space());
120 Thyra::randomize(-1.0, 1.0, x.ptr());
121 inArgs.set_x(x);
122 if (set_x_dot) {
123 RCP<Vector> x_dot = Thyra::createMember(aux.get_x_space());
124 Thyra::randomize(-1.0, 1.0, x_dot.ptr());
125 inArgs.set_x_dot(x_dot);
126 }
127 else
128 inArgs.set_x_dot(Teuchos::null);
129 inArgs.set_t(t);
130 inArgs.set_alpha(alpha);
131 inArgs.set_beta(beta);
132 return inArgs;
133}

136{
137 RCP<Vector> f = Thyra::createMember(aux.get_f_space());

139 outArgs.set_f(f);

144RCP<Thyra::LinearOpBase<double> > evalW(const AuxME& aux,
145 const MEB::InArgs<double>& inArgs)
146{

154// The residual is affine in (x, x_dot), so W*v = alpha*dF/dx_dot*v +
155// beta*dF/dx*v = alpha*(F(x, x_dot+v) - F(x, x_dot)) + beta*(F(x+v, x_dot) -
156// F(x, x_dot)) up to rounding.
157void checkWIsJacobian(const std::string& responseType,
158 Teuchos::FancyOStream& out, bool& success)
159{
160 SinCosSetup setup;
161 RCP<AuxME> aux = setup.createAuxME(responseType);
162
163 const double alpha = 3.0, beta = 0.7;
164 for (const double t : {0.0, 0.5}) {
165 MEB::InArgs<double> inArgs = randomInArgs(*aux, t, alpha, beta);
166 RCP<Vector> v = Thyra::createMember(aux->get_x_space());
167 Thyra::randomize(-1.0, 1.0, v.ptr());
168
169 RCP<Vector> F0 = evalResidual(*aux, inArgs);
170 MEB::InArgs<double> inArgs_x = inArgs;
171 RCP<Vector> x_plus_v = inArgs.get_x()->clone_v();
172 Thyra::Vp_V(x_plus_v.ptr(), *v);
173 inArgs_x.set_x(x_plus_v);
174 RCP<Vector> Fx = evalResidual(*aux, inArgs_x);
175 MEB::InArgs<double> inArgs_xdot = inArgs;
176 RCP<Vector> x_dot_plus_v = inArgs.get_x_dot()->clone_v();
177 Thyra::Vp_V(x_dot_plus_v.ptr(), *v);
178 inArgs_xdot.set_x_dot(x_dot_plus_v);
179 RCP<Vector> Fxdot = evalResidual(*aux, inArgs_xdot);
180
181 // Jv = beta*(Fx - F0) + alpha*(Fxdot - F0)
182 RCP<Vector> Jv = Thyra::createMember(aux->get_f_space());
183 Thyra::V_StVpStV(Jv.ptr(), beta, *Fx, alpha, *Fxdot);
184 Thyra::Vp_StV(Jv.ptr(), -(alpha + beta), *F0);
185
186 RCP<Thyra::LinearOpBase<double> > W = evalW(*aux, inArgs);
187 RCP<Vector> Wv = Thyra::createMember(aux->get_f_space());
188 Thyra::apply(*W, Thyra::NOTRANS, *v, Wv.ptr());
189
190 RCP<Vector> diff = Wv->clone_v();
191 Thyra::Vp_StV(diff.ptr(), -1.0, *Jv);
192 out << "t = " << t << ": |W*v| = " << Thyra::norm_2(*Wv)
193 << ", |W*v - J*v| = " << Thyra::norm_2(*diff) << std::endl;
194 TEST_COMPARE(Thyra::norm_2(*diff), <=, 1.0e-12 * Thyra::norm_2(*Jv));
195 }
196}

211// The solver of W from get_W_factory() solves the coupled system.
212TEUCHOS_UNIT_TEST(AdjointAuxSensitivityModelEvaluator, W_Solve)
213{
214 SinCosSetup setup;
215 RCP<AuxME> aux = setup.createAuxME("Final Time");
216 MEB::InArgs<double> inArgs = randomInArgs(*aux, 0.5, 3.0, 0.7);
217 RCP<Thyra::LinearOpBase<double> > W = evalW(*aux, inArgs);
218 RCP<Thyra::LinearOpWithSolveBase<double> > W_lows =
219 Thyra::linearOpWithSolve<double>(*aux->get_W_factory(), W);
220
221 RCP<Vector> b = Thyra::createMember(aux->get_f_space());
222 Thyra::randomize(-1.0, 1.0, b.ptr());
223 RCP<Vector> x = Thyra::createMember(aux->get_x_space());

229 out << "|W*x - b| = " << Thyra::norm_2(*r) << ", |b| = " << Thyra::norm_2(*b)
230 << std::endl;
231 TEST_COMPARE(Thyra::norm_2(*r), <=, 1.0e-12 * Thyra::norm_2(*b));
232}

234// ************************************************************
235// The factory of W rejects, when reinitializing, an operator that is not
236// block triangular or has another number of blocks, and leaves the solver
237// unchanged (it checks the operator before reinitializing the diagonal
238// blocks).
239TEUCHOS_UNIT_TEST(BlockedTriangularLinearOpWithSolveFactory,
240 ReinitializeRejected)
241{
242 typedef Thyra::LinearOpBase<double> LinearOp;
243 typedef Thyra::PhysicallyBlockedLinearOpBase<double> PBLOB;
244 SinCosSetup setup;
245 RCP<AuxME> aux = setup.createAuxME("Time Integral");
246 RCP<const Thyra::LinearOpWithSolveFactoryBase<double> > lowsf =
247 aux->get_W_factory();
248 // The two W differ in their diagonal blocks
249 RCP<LinearOp> W = evalW(*aux, randomInArgs(*aux, 0.5, 3.0, 0.7));
250 RCP<LinearOp> W2 = evalW(*aux, randomInArgs(*aux, 0.5, 1.5, 0.4));
251 RCP<Thyra::LinearOpWithSolveBase<double> > W_lows =
252 Thyra::linearOpWithSolve<double>(*lowsf, W);
253
254 // Relative residual of the solve of op*x = b with W_lows
255 auto solveResidual = [&](const LinearOp& op) {
256 RCP<Vector> b = Thyra::createMember(op.range());
257 Thyra::randomize(-1.0, 1.0, b.ptr());
258 RCP<Vector> x = Thyra::createMember(op.domain());
259 Thyra::assign(x.ptr(), 0.0);
260 Thyra::solve<double>(*W_lows, Thyra::NOTRANS, *b, x.ptr());
261 RCP<Vector> r = b->clone_v();
262 Thyra::apply(op, Thyra::NOTRANS, *x, r.ptr(), 1.0, -1.0);
263 return Thyra::norm_2(*r) / Thyra::norm_2(*b);
264 };
265
266 // W2 with a block above the diagonal, and its first diagonal block alone
267 RCP<const PBLOB> W2_blo = Teuchos::rcp_dynamic_cast<const PBLOB>(W2, true);
268 RCP<const LinearOp> W2_10 = W2_blo->getBlock(1, 0).assert_not_null();
269 RCP<const LinearOp> notTriangular =
270 Thyra::block2x2<double>(W2_blo->getBlock(0, 0), Thyra::adjoint(W2_10),
271 W2_10, W2_blo->getBlock(1, 1));
272 RCP<const LinearOp> oneBlock = Thyra::block1x1<double>(W2_blo->getBlock(0, 0));
273 TEST_THROW(Thyra::initializeOp<double>(*lowsf, notTriangular, W_lows.ptr()),
274 std::logic_error);
275 TEST_THROW(Thyra::initializeOp<double>(*lowsf, oneBlock, W_lows.ptr()),
276 std::logic_error);
277
278 // W_lows still solves W, and can be reinitialized with W2
279 TEST_COMPARE(solveResidual(*W), <=, 1.0e-12);
280 Thyra::initializeOp<double>(*lowsf, W2, W_lows.ptr());
281 TEST_COMPARE(solveResidual(*W2), <=, 1.0e-12);
282 TEST_COMPARE(solveResidual(*W), >, 1.0e-6);
283}

285// ************************************************************
286// df/dx and df/dp are evaluated once for each time, parameters and form
287// (x_dot given or not), and the residual does not change.
288TEUCHOS_UNIT_TEST(AdjointAuxSensitivityModelEvaluator, DerivativesReused)
289{
290 SinCosSetup setup;
291 RCP<AuxME> aux = setup.createAuxME("Time Integral");
292 CountingModel& fm = *setup.model;
293 CountingModel& am = *setup.adjoint_model;
294
295 // Residual and W at t = 0.5
296 MEB::InArgs<double> inArgs = randomInArgs(*aux, 0.5, 3.0, 0.7);
297 RCP<Vector> F0 = evalResidual(*aux, inArgs);
298 TEST_EQUALITY(am.num_dfdx, 1);
299 TEST_EQUALITY(fm.num_dfdp, 1);
300 MEB::InArgs<double> inArgs2 = randomInArgs(*aux, 0.5, 3.0, 0.7);
301 evalResidual(*aux, inArgs2);
302 evalW(*aux, inArgs2);
303 TEST_EQUALITY(am.num_dfdx, 1);
304 TEST_EQUALITY(fm.num_dfdp, 1);
305
306 // Another time
307 MEB::InArgs<double> inArgs3 = randomInArgs(*aux, 0.0, 3.0, 0.7);
308 evalW(*aux, inArgs3); // df/dp for the coupling block of W
309 TEST_EQUALITY(fm.num_dfdp, 2);
310 evalResidual(*aux, inArgs3);
311 TEST_EQUALITY(am.num_dfdx, 2);
312 TEST_EQUALITY(fm.num_dfdp, 2);
313
314 // Other parameter values (changed in place), then the nominal ones
315 RCP<Vector> p = inArgs3.get_p(0)->clone_v();
316 inArgs3.set_p(0, p);
317 Thyra::scale(1.1, p.ptr());
318 evalResidual(*aux, inArgs3);
319 TEST_EQUALITY(am.num_dfdx, 3);
320 TEST_EQUALITY(fm.num_dfdp, 3);
321 Thyra::assign(p.ptr(), *inArgs.get_p(0));
322 evalResidual(*aux, inArgs3);
323 TEST_EQUALITY(am.num_dfdx, 4);
324 TEST_EQUALITY(fm.num_dfdp, 4);
325
326 // Explicit form (no x_dot)
327 MEB::InArgs<double> inArgs4 = randomInArgs(*aux, 0.0, 3.0, 0.7, false);
328 evalResidual(*aux, inArgs4);
329 TEST_EQUALITY(am.num_dfdx, 5);
330 TEST_EQUALITY(fm.num_dfdp, 5);
331
332 // The residual is the one of a new model evaluator
333 RCP<Vector> F1 = evalResidual(*aux, inArgs);
334 RCP<Vector> F2 = evalResidual(*setup.createAuxME("Time Integral"), inArgs);
335 RCP<Vector> diff = F1->clone_v();
336 Thyra::Vp_StV(diff.ptr(), -1.0, *F0);
337 TEST_COMPARE(Thyra::norm_2(*diff), <=, 1.0e-14 * Thyra::norm_2(*F0));
338 Thyra::V_VmV(diff.ptr(), *F2, *F0);
339 TEST_COMPARE(Thyra::norm_2(*diff), <=, 1.0e-14 * Thyra::norm_2(*F0));
340}

342// ************************************************************
343// A W keeps the df/dp of its evaluation point: later evaluations of another W
344// or of the residual at other times (the df/dp of SinCos depends on the
345// forward state) do not change it.
346TEUCHOS_UNIT_TEST(AdjointAuxSensitivityModelEvaluator, W_KeepsItsEvaluationPoint)
347{
348 SinCosSetup setup;
349 RCP<AuxME> aux = setup.createAuxME("Final Time");
350 RCP<Vector> v = Thyra::createMember(aux->get_x_space());
351 Thyra::randomize(-1.0, 1.0, v.ptr());
352 auto applyW = [&](const Thyra::LinearOpBase<double>& W) {
353 RCP<Vector> Wv = Thyra::createMember(aux->get_f_space());
354 Thyra::apply(W, Thyra::NOTRANS, *v, Wv.ptr());
355 return Wv;
356 };
357 auto relDiff = [](const Vector& a, const Vector& b) {
358 RCP<Vector> d = a.clone_v();
359 Thyra::Vp_StV(d.ptr(), -1.0, b);
360 return Thyra::norm_2(*d) / Thyra::norm_2(b);
361 };
362
363 // Same alpha and beta: the two W differ in their coupling blocks
364 RCP<Thyra::LinearOpBase<double> > W1 =
365 evalW(*aux, randomInArgs(*aux, 0.0, 3.0, 0.7));
366 RCP<Vector> W1v = applyW(*W1);
367 RCP<Thyra::LinearOpBase<double> > W2 =
368 evalW(*aux, randomInArgs(*aux, 0.5, 3.0, 0.7));
369 RCP<Vector> W2v = applyW(*W2);
370 evalResidual(*aux, randomInArgs(*aux, 1.0, 3.0, 0.7));
371 out << "|W2*v - W1*v|/|W1*v| = " << relDiff(*W2v, *W1v) << std::endl;
372 TEST_COMPARE(relDiff(*W2v, *W1v), >, 1.0e-3);
373 TEST_COMPARE(relDiff(*applyW(*W1), *W1v), <=, 1.0e-14);
374 TEST_COMPARE(relDiff(*applyW(*W2), *W2v), <=, 1.0e-14);
375
376 // df/dp is still evaluated once per evaluation point
377 const int num_dfdp = setup.model->num_dfdp;
378 evalW(*aux, randomInArgs(*aux, 1.0, 1.5, 0.4));
379 evalResidual(*aux, randomInArgs(*aux, 1.0, 3.0, 0.7));
380 TEST_EQUALITY(setup.model->num_dfdp, num_dfdp);
381}

383// ************************************************************
384// The coupling operator s*df/dp^T, with left and right scalings.
385TEUCHOS_UNIT_TEST(AdjointAuxCouplingLinearOp, ApplyScalingRowStats)
386{
387 typedef Tempus::AdjointAuxCouplingLinearOp<double> CouplingOp;
388 RCP<const Thyra::VectorSpaceBase<double> > f_space =
389 Thyra::defaultSpmdVectorSpace<double>(5);
390 RCP<const Thyra::VectorSpaceBase<double> > p_space =
391 Thyra::defaultSpmdVectorSpace<double>(3);
392 RCP<MultiVector> dfdp = Thyra::createMembers(f_space, 3);
393 Thyra::seed_randomize<double>(41);
394 Thyra::randomize(-1.0, 1.0, dfdp.ptr());
395
396 RCP<CouplingOp> op = rcp(new CouplingOp(p_space, f_space));
397 RCP<Vector> v = Thyra::createMember(f_space);
398 Thyra::randomize(-1.0, 1.0, v.ptr());
399 RCP<Vector> y = Thyra::createMember(p_space);
400
401 // Zero before initialize()
402 Thyra::assign(y.ptr(), 1.0);
403 Thyra::apply(*op, Thyra::NOTRANS, *v, y.ptr());
404 TEST_EQUALITY(Thyra::norm_inf(*y), 0.0);
405
406 // s*df/dp^T*v, scaled left and right
407 const double s = -0.7;
408 op->initialize(dfdp, Thyra::CONJTRANS, s);
409 RCP<Vector> dl = Thyra::createMember(p_space);
410 RCP<Vector> dr = Thyra::createMember(f_space);
411 Thyra::randomize(0.5, 2.0, dl.ptr());
412 Thyra::randomize(0.5, 2.0, dr.ptr());
413 op->scaleLeft(*dl);
414 op->scaleRight(*dr);
415 op->scaleLeft(*dl);
416
417 RCP<Vector> drv = v->clone_v();
418 Thyra::ele_wise_scale(*dr, drv.ptr());
419 RCP<Vector> y_ref = Thyra::createMember(p_space);
420 Thyra::apply(*dfdp, Thyra::CONJTRANS, *drv, y_ref.ptr(), s);
421 Thyra::ele_wise_scale(*dl, y_ref.ptr());
422 Thyra::ele_wise_scale(*dl, y_ref.ptr());
423 Thyra::apply(*op, Thyra::NOTRANS, *v, y.ptr());
424 Thyra::Vp_StV(y_ref.ptr(), -1.0, *y);
425 TEST_COMPARE(Thyra::norm_inf(*y_ref), <=, 1.0e-14 * Thyra::norm_inf(*y));
426
427 // Consistency of the apply and its transpose
428 Thyra::LinearOpTester<double> tester;
429 TEST_ASSERT(tester.check(*op, Teuchos::inOutArg(out)));
430
431 // The row statistics are zero
432 RCP<Vector> rs = Thyra::createMember(p_space);
433 op->getRowStat(Thyra::RowStatLinearOpBaseUtils::ROW_STAT_ROW_SUM, rs.ptr());
434 TEST_EQUALITY(Thyra::norm_inf(*rs), 0.0);
435 TEST_ASSERT(!op->rowStatIsSupported(Thyra::RowStatLinearOpBaseUtils::ROW_STAT_INV_ROW_SUM));
436
437 // initialize() removes the scalings
438 op->initialize(dfdp, Thyra::CONJTRANS, s);
439 Thyra::apply(*dfdp, Thyra::CONJTRANS, *v, y_ref.ptr(), s);
440 Thyra::apply(*op, Thyra::NOTRANS, *v, y.ptr());
441 Thyra::Vp_StV(y_ref.ptr(), -1.0, *y);
442 TEST_COMPARE(Thyra::norm_inf(*y_ref), <=, 1.0e-14 * Thyra::norm_inf(*y));
443}
